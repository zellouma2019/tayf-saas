          {/* Status Distribution Bar (R79) */}
          {safeOrders.length > 0 && (
            <div className="status-dist-bar-container">
              <div className="flex items-center justify-between mb-1.5">
                <h4 className="text-[11px] font-bold text-foreground flex items-center gap-1.5">
                  <BarChart3 className="h-3.5 w-3.5 text-primary" />
                  توزيع حالات الطلبات
                </h4>
                <span className="text-[10px] text-muted-foreground">{safeOrders.length} طلب</span>
              </div>
              <div className="status-dist-bar-track">
                {['pending','confirmed','printing','ready','delivered','cancelled'].map(status => {
                  const count = safeOrders.filter(o => o.status === status).length;
                  const pct = safeOrders.length > 0 ? (count / safeOrders.length) * 100 : 0;
                  if (count === 0) return null;
                  return (
                    <div
                      key={status}
                      className="status-dist-bar-segment"
                      style={{
                        width: `${Math.max(pct, 1.5)}%`,
                        backgroundColor: STATUS_COLORS[status] || '#94a3b8',
                      }}
                      title={`${STATUS_META[status as keyof typeof STATUS_META]?.label || status}: ${count} (${pct.toFixed(1)}%)`}
                    />
                  );
                })}
              </div>
              <div className="flex flex-wrap gap-x-3 gap-y-1 mt-1.5">
                {['pending','confirmed','printing','ready','delivered','cancelled'].map(status => {
                  const count = safeOrders.filter(o => o.status === status).length;
                  if (count === 0) return null;
                  return (
                    <button
                      key={status}
                      onClick={() => { setStatusFilter(status); setActiveTab('orders'); }}
                      className="status-dist-legend-item"
                    >
                      <span className="status-dist-legend-dot" style={{ backgroundColor: STATUS_COLORS[status] || '#94a3b8' }} />
                      <span className="text-[10px] text-muted-foreground">{STATUS_META[status as keyof typeof STATUS_META]?.label || status}</span>
                      <span className="text-[10px] font-bold text-foreground tabular-data">{count}</span>
                    </button>
                  );
                })}
              </div>
            </div>
          )}
