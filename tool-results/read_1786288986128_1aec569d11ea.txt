    53→    53→// ===== Inline Login Gate =====
    54→    54→function InlineLoginGate({ onUnlock }: { onUnlock: () => void }) {
    55→    55→  const [password, setPassword] = useState('');
    56→    56→  const [error, setError] = useState(false);
    57→    57→  const [loading, setLoading] = useState(false);
    58→    58→
    59→    59→  async function handleSubmit(e: React.FormEvent) {
    60→    60→    e.preventDefault();
    61→    61→    if (!password) return;
    62→    62→    setLoading(true);
    63→    63→    setError(false);
    64→    64→    try {
    65→    65→      const res = await fetch('/api/super-admin/auth', {
    66→    66→        method: 'POST',
    67→    67→        headers: { 'Content-Type': 'application/json' },
    68→    68→        body: JSON.stringify({ password }),
    69→    69→      });
    70→    70→      const data = await res.json();
    71→    71→      if (data.success) {
    72→    72→        markAuthenticated(data.token, data.ts);
    73→    73→        onUnlock();
    74→    74→      } else {
    75→    75→        setError(true);
    76→    76→      }
    77→    77→    } catch {
    78→    78→      setError(true);
    79→    79→    } finally {
    80→    80→      setLoading(false);
    81→    81→    }
    82→    82→  }
    83→    83→
    84→    84→  return (
    85→    85→    <div className="min-h-screen flex items-center justify-center bg-background p-4" dir="rtl">
    86→    86→      <Card className="max-w-sm w-full">
    87→    87→        <CardContent className="p-6 space-y-4">
    88→    88→          <div className="flex flex-col items-center gap-3">
    89→    89→            <img src="/n.png" alt="طيف" className="w-16 h-16 rounded-2xl object-contain shadow-lg" />
    90→    90→            <div className="text-center">
    91→    91→              <h1 className="text-xl font-bold">لوحة تحكم المنصة</h1>
    92→    92→              <p className="text-sm text-muted-foreground">أدخل كلمة المرور للدخول</p>
    93→    93→            </div>
    94→    94→          </div>
    95→    95→          <form onSubmit={handleSubmit} className="space-y-3">
    96→    96→            <div className="relative">
    97→    97→              <Lock className="absolute right-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
    98→    98→              <Input type="password" placeholder="كلمة المرور" value={password} onChange={(e) => { setPassword(e.target.value); setError(false); }} className="pr-9" dir="ltr" />
    99→    99→            </div>
   100→   100→            {error && <p className="text-destructive text-xs text-center">كلمة المرور غير صحيحة</p>}
   101→   101→            <Button type="submit" className="w-full bg-gradient-to-l from-amber-500 to-amber-600 text-white" disabled={loading}>
   102→   102→              {loading ? 'جاري التحقق...' : 'دخول'}
   103→   103→            </Button>
   104→   104→          </form>
   105→   105→          <div className="flex items-center justify-center gap-2 pt-2">
   106→   106→            <ThemeToggle />
   107→   107→          </div>
   108→   108→        </CardContent>
   109→   109→      </Card>
   110→   110→    </div>
   111→   111→  );
   112→   112→}

   115→   115→function InlineCreateShop({ open, onClose, onCreated }: { open: boolean; onClose: () => void; onCreated: () => void }) {
   116→   116→  const [name, setName] = useState('');
   117→   117→  const [slug, setSlug] = useState('');
   118→   118→  const [pin, setPin] = useState('');
   119→   119→  const [ownerName, setOwnerName] = useState('');
   120→   120→  const [ownerPhone, setOwnerPhone] = useState('');
   121→   121→  const [country, setCountry] = useState('DZ');
   122→   122→  const [submitting, setSubmitting] = useState(false);
   123→   123→  const [created, setCreated] = useState<{ name: string; slug: string; pin: string } | null>(null);
   124→   124→
   125→   125→  function genSlug(t: string) {
   126→   126→    return t.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/^-|-$/g, '');
   127→   127→  }
   128→   128→
   129→   129→  function handleName(v: string) {
   130→   130→    setName(v);
   131→   131→    if (!slug || slug === genSlug(name)) setSlug(genSlug(v));
   132→   132→  }
   133→   133→
   134→   134→  async function handleSubmit(e: React.FormEvent) {
   135→   135→    e.preventDefault();
   136→   136→    if (!name || !slug || !pin) return;
   137→   137→    setSubmitting(true);
   138→   138→    try {
   139→   139→      const res = await fetch('/api/shops', {
   140→   140→        method: 'POST',
   141→   141→        headers: { 'Content-Type': 'application/json' },
   142→   142→        body: JSON.stringify({ name, slug, adminPin: pin, ownerName, ownerPhone, country }),
   143→   143→      });
   144→   144→      if (res.ok) {
   145→   145→        setCreated({ name, slug, pin });
   146→   146→        onCreated();
   147→   147→      } else {
   148→   148→        const data = await res.json().catch(() => ({}));
   149→   149→        toast.error(data.error || 'فشل إنشاء المتجر');
   150→   150→      }
   151→   151→    } catch {
   152→   152→      toast.error('خطأ في الاتصال');
   153→   153→    } finally {
   154→   154→      setSubmitting(false);
   155→   155→    }
   156→   156→  }
   157→   157→
   158→   158→  function handleClose() {
   159→   159→    setName(''); setSlug(''); setPin(''); setOwnerName(''); setOwnerPhone('');
   160→   160→    setCreated(null); onClose();
   161→   161→  }
   162→   162→
   163→   163→  return (
   164→   164→    <Dialog open={open} onOpenChange={(o) => !o && handleClose()}>
   165→   165→      <DialogContent className="sm:max-w-md" dir="rtl">
   166→   166→        <DialogTitle>{created ? 'تم إنشاء المتجر ✓' : 'إنشاء متجر جديد'}</DialogTitle>
   167→   167→        {created ? (
   168→   168→          <div className="space-y-4 py-4">
   169→   169→            <div className="text-center space-y-2">
   170→   170→              <p className="font-bold text-lg">{created.name}</p>
   171→   171→              <div className="space-y-2 text-sm">
   172→   172→                <div className="flex items-center justify-between p-2 rounded-lg bg-muted">
   173→   173→                  <span className="text-muted-foreground">رابط المتجر:</span>
   174→   174→                  <button onClick={() => robustCopy(`/s/${created.slug}`, 'تم نسخ الرابط', '')} className="font-mono text-xs text-primary hover:underline">/s/{created.slug}</button>
   175→   175→                </div>
   176→   176→                <div className="flex items-center justify-between p-2 rounded-lg bg-muted">
   177→   177→                  <span className="text-muted-foreground">كلمة المرور:</span>
   178→   178→                  <span className="font-mono text-xs font-bold">{created.pin}</span>
   179→   179→                </div>
   180→   180→              </div>
   181→   181→            </div>
   182→   182→            <Button onClick={handleClose} className="w-full">تم</Button>
   183→   183→          </div>
   184→   184→        ) : (
   185→   185→          <form onSubmit={handleSubmit} className="space-y-4">
   186→   186→            <div className="space-y-2">
   187→   187→              <Label>اسم المتجر *</Label>
   188→   188→              <Input value={name} onChange={(e) => handleName(e.target.value)} placeholder="مثال: مطبعة النور" />
   189→   189→            </div>
   190→   190→            <div className="space-y-2">
   191→   191→              <Label>المعرّف (slug) *</Label>
   192→   192→              <Input value={slug} onChange={(e) => setSlug(e.target.value)} placeholder="matbaa-alnoor" dir="ltr" />
   193→   193→            </div>
   194→   194→            <div className="space-y-2">
   195→   195→              <Label>كلمة مرور لوحة التحكم (PIN) *</Label>
   196→   196→              <Input value={pin} onChange={(e) => setPin(e.target.value)} placeholder="1234" maxLength={10} dir="ltr" type="text" />
   197→   197→            </div>
   198→   198→            <div className="grid grid-cols-2 gap-3">
   199→   199→              <div className="space-y-2">
   200→   200→                <Label>اسم المالك</Label>
   201→   201→                <Input value={ownerName} onChange={(e) => setOwnerName(e.target.value)} placeholder="اختياري" />
   202→   202→              </div>
   203→   203→              <div className="space-y-2">
   204→   204→                <Label>هاتف المالك</Label>
   205→   205→                <Input value={ownerPhone} onChange={(e) => setOwnerPhone(e.target.value)} placeholder="اختياري" dir="ltr" />
   206→   206→              </div>
   207→   207→            </div>
   208→   208→            <div className="space-y-2">
   209→   209→              <Label>الدولة</Label>
   210→   210→              <Select value={country} onValueChange={setCountry}>
   211→   211→                <SelectTrigger><SelectValue /></SelectTrigger>
   212→   212→                <SelectContent>
   213→   213→                  {ARAB_COUNTRIES.map((c) => (
   214→   214→                    <SelectItem key={c.code} value={c.code}>{c.flag} {c.name_ar}</SelectItem>
   215→   215→                  ))}
   216→   216→                </SelectContent>
   217→   217→              </Select>
   218→   218→            </div>
   219→   219→            <div className="flex gap-2 pt-2">
   220→   220→              <Button type="button" variant="outline" onClick={handleClose} className="flex-1">إلغاء</Button>
   221→   221→              <Button type="submit" disabled={!name || !slug || !pin || submitting} className="flex-1 bg-gradient-to-l from-amber-500 to-amber-600 text-white">
   222→   222→                {submitting ? 'جاري الإنشاء...' : 'إنشاء'}
   223→   223→              </Button>
   224→   224→            </div>
   225→   225→          </form>
   226→   226→        )}
   227→   227→      </DialogContent>
   228→   228→    </Dialog>
   229→   229→  );
   230→   230→}
