   262→
   263→function AnimatedCounter({
   264→  value,
   265→  duration = 1.2,
   266→  suffix = "",
   267→}: {
   268→  value: number;
   269→  duration?: number;
   270→  suffix?: string;
   271→}) {
   272→  const [display, setDisplay] = useState(0);
   273→  const prevValue = useRef(0);
   274→
   275→  useEffect(() => {
   276→    const start = prevValue.current;
   277→    const end = value;
   278→    const startTime = performance.now();
   279→
   280→    function animate(now: number) {
   281→      const elapsed = (now - startTime) / (duration * 1000);
   282→      const progress = Math.min(elapsed, 1);
   283→      // easeOutExpo
   284→      const eased = progress === 1 ? 1 : 1 - Math.pow(2, -10 * progress);
   285→      setDisplay(Math.round(start + (end - start) * eased));
   286→      if (progress < 1) requestAnimationFrame(animate);
   287→    }
   288→
   289→    requestAnimationFrame(animate);
   290→    prevValue.current = value;
   291→  }, [value, duration]);
   292→
   293→  return <>{display.toLocaleString("en-US")}{suffix}</>;
   294→}

   296→/* ═══════════════════════════════════════════════════════
   297→   Sub-Components: CircularGauge
   298→   ═══════════════════════════════════════════════════════ */
   299→
   300→function CircularGauge({
   301→  value,
   302→  max,
   303→  size = 64,
   304→  smSize,
   305→  strokeWidth = 5,
   306→  color,
   307→  label,
   308→  tooltipText,
   309→}: {
   310→  value: number;
   311→  max: number;
   312→  size?: number;
   313→  smSize?: number;
   314→  strokeWidth?: number;
   315→  color: string;
   316→  label: string;
   317→  tooltipText?: string;
   318→}) {
   319→  const radius = (size - strokeWidth) / 2;
   320→  const circumference = 2 * Math.PI * radius;
   321→  const percentage = Math.min(value / max, 1);
   322→  const offset = circumference - percentage * circumference;
   323→  // Use CSS to make the gauge responsive: smaller on mobile, larger on desktop (smSize)
   324→  const responsiveClass = smSize ? "w-14 h-14 sm:w-[72px] sm:h-[72px]" : "";
   325→
   326→  const gauge = (
   327→    <div className="flex flex-col items-center gap-1">
   328→      <div className={`relative ${responsiveClass}`} style={!smSize ? { width: size, height: size } : undefined}>
   329→        <svg width="100%" height="100%" viewBox={`0 0 ${size} ${size}`} className="-rotate-90">
   330→          <circle
   331→            cx={size / 2}
   332→            cy={size / 2}
   333→            r={radius}
   334→            fill="none"
   335→            stroke="currentColor"
   336→            strokeWidth={strokeWidth}
   337→            className="text-muted/40"
   338→          />
   339→          <motion.circle
   340→            cx={size / 2}
   341→            cy={size / 2}
   342→            r={radius}
   343→            fill="none"
   344→            stroke={color}
   345→            strokeWidth={strokeWidth}
   346→            strokeLinecap="round"
   347→            strokeDasharray={circumference}
   348→            initial={{ strokeDashoffset: circumference }}
   349→            animate={{ strokeDashoffset: offset }}
   350→            transition={{ duration: 1.4, ease: "easeOut", delay: 0.3 }}
   351→          />
   352→        </svg>
   353→        <div className="absolute inset-0 flex items-center justify-center">
   354→          <span className="text-xs sm:text-sm font-bold" style={{ color }}>
   355→            <AnimatedCounter value={value} />
   356→          </span>
   357→        </div>
   358→      </div>
   359→      <span className="text-[10px] text-muted-foreground font-medium">{label}</span>
   360→    </div>
   361→  );
   362→
   363→  if (tooltipText) {
   364→    return (
   365→      <Tooltip>
   366→        <TooltipTrigger asChild>
   367→          <div className="relative cursor-help">{gauge}</div>
   368→        </TooltipTrigger>
   369→        <TooltipContent side="top" className="max-w-[220px] text-center">
   370→          {tooltipText}
   371→        </TooltipContent>
   372→      </Tooltip>
   373→    );
   374→  }
   375→
   376→  return <div className="relative">{gauge}</div>;
   377→}

   476→  const [dragCounter, setDragCounter] = useState(0);

   483→  const [elapsedSec, setElapsedSec] = useState(0);

   493→      analysisStartRef.current = Date.now();
   494→    }
   495→    const interval = setInterval(() => {
   496→      setElapsedSec(Math.floor((Date.now() - (analysisStartRef.current || Date.now())) / 1000));
   497→    }, 500);
   498→    return () => clearInterval(interval);
   499→  }, [analysisPhase]);
   500→
   501→  // ─── Validation helpers ───
   502→  function validateFile(file: File): string | null {
   503→    const ext = file.name.split(".").pop()?.toLowerCase() || "";
   504→    if (!ACCEPTED_TYPES.includes(`.${ext}`)) {
   505→      return `صيغة الملف "${ext}" غير مدعومة. الصيغ المدعومة: ${ACCEPTED_TYPES.join(", ")}`;
   506→    }
   507→    if (file.size > MAX_FILE_SIZE) {
   508→      return `حجم الملف ${(file.size / (1024 * 1024)).toFixed(1)} ميغابايت يتجاوز الحد الأقصى (50 ميغابايت)`;
   509→    }
   510→    if (file.size === 0) {
   511→      return "الملف فارغ — يرجى اختيار ملف آخر";
   512→    }
   513→    return null;
   514→  }

   516→  // ─── File processing ───
   517→  function processFile(file: File) {
   518→    const error = validateFile(file);

   521→      // The parent's handleFile will handle validation too
   522→    }
   523→    currentFileRef.current = file;
   524→    onFileSelected(file);
   525→  }
   526→
   527→  // ─── Drag & Drop handlers ───
   528→  const handleDragEnter = useCallback((e: React.DragEvent) => {
   529→    e.preventDefault();
   530→    e.stopPropagation();
   531→    setDragCounter((c) => c + 1);
   532→    setIsDragOver(true);
   533→  }, []);
   534→
   535→  const handleDragLeave = useCallback((e: React.DragEvent) => {
   536→    e.preventDefault();
   537→    e.stopPropagation();
   538→    setDragCounter((c) => {
   539→      const next = c - 1;
   540→      if (next === 0) setIsDragOver(false);
   541→      return next;
   542→    });
   543→  }, []);
   544→
   545→  const handleDragOver = useCallback((e: React.DragEvent) => {
   546→    e.preventDefault();
   547→    e.stopPropagation();
   548→  }, []);
   549→
   550→  const handleDrop = useCallback(
   551→    (e: React.DragEvent) => {
   552→      e.preventDefault();
   553→      e.stopPropagation();
