    82→interface ShopData {
    83→  slug: string; name: string; phone: string | null;
    84→  ownerName: string | null; ownerPhone: string | null;
    85→  whatsapp: string | null; email: string | null; address: string | null;
    86→  country: string | null; language: string | null; customCurrency: string | null;
    87→  isActive: boolean; plan: string;
    88→  primaryColor: string | null; adminPin: string;
    89→  themeId: number | null;
    90→  logoIcon: string | null;
    91→  trialDays: number | null; trialStartsAt: string | null;
    92→  features: Record<string, boolean>;
    93→  ownerNotes: string | null;
    94→  paymentInfo: string | null;
    95→}
    96→
    97→interface AdminShopManagementProps {
    98→  open: boolean;
    99→  onClose: () => void;
   100→  shop: { slug: string; name: string } | null;
   101→  onSaved: () => void;
   102→}
   103→
   104→export function AdminShopManagement({ open, onClose, shop, onSaved }: AdminShopManagementProps) {
   105→  const [data, setData] = useState<ShopData | null>(null);
   106→  const [loading, setLoading] = useState(false);
   107→  const [saving, setSaving] = useState(false);
   108→  const [tab, setTab] = useState("basic");
   109→  const [hasChanges, setHasChanges] = useState(false);
   110→  const [originalData, setOriginalData] = useState<ShopData | null>(null);
   111→
   112→  useEffect(() => {
   113→    if (!open || !shop) return;
   114→    let cancelled = false;
   115→    (async () => {
   116→      setLoading(true);
   117→      setTab("basic");
   118→      try {
   119→        const res = await fetch(`/api/admin/shops/${shop.slug}`);
   120→        if (res.ok && !cancelled) {
   121→          const d = await res.json();
   122→          const s = d.shop;
   123→          const parsed: ShopData = {
   124→            slug: s.slug, name: s.name, phone: s.phone,
   125→            ownerName: s.ownerName, ownerPhone: s.ownerPhone,
   126→            whatsapp: s.whatsapp, email: s.email, address: s.address,
   127→            country: s.country, language: s.language || "ar",
   128→            customCurrency: s.customCurrency || null,
   129→            isActive: s.isActive,
   130→            plan: s.plan || "free", primaryColor: s.primaryColor,
   131→            adminPin: s.adminPin, themeId: s.themeId || 1,
   132→            logoIcon: s.logoIcon || "Printer",
   133→            trialDays: s.trialDays, trialStartsAt: s.trialStartsAt,
   134→            features: s.features ? (typeof s.features === "string" ? JSON.parse(s.features) : s.features) : {},
   135→            ownerNotes: s.ownerNotes, paymentInfo: s.paymentInfo,
   136→          };
   137→          setData(parsed);
   138→          setOriginalData(JSON.parse(JSON.stringify(parsed)));
   139→          setHasChanges(false);
   140→        } else if (!cancelled) {
   141→          toast.error("فشل تحميل بيانات المتجر");
   142→          onClose();
   143→        }
   144→      } catch { if (!cancelled) { toast.error("خطأ في الاتصال"); onClose(); } }
   145→      finally { if (!cancelled) setLoading(false); }
   146→    })();
   147→    return () => { cancelled = true; };
   148→  }, [open, shop, onClose]);
   149→
   150→  function setField<K extends keyof ShopData>(key: K, val: ShopData[K]) {
   151→    setData((prev) => {
   152→      if (!prev) return prev;
   153→      const next = { ...prev, [key]: val };
   154→      // عند تغيير الدولة، حدّث العملة تلقائياً إن لم يكن هناك عملة مخصصة
   155→      if (key === "country" && !prev.customCurrency) {
   156→        const country = ARAB_COUNTRIES.find((c) => c.code === val);
   157→        if (country) {
   158→          next.customCurrency = country.currencyCode;
   159→        }
   160→      }
   161→      // عند تغيير الخطة إلى مدفوعة، فعّل كل الميزات
   162→      if (key === "plan" && val === "paid") {
   163→        const allFeatures: Record<string, boolean> = {};
   164→        SHOP_FEATURES.forEach((f) => { allFeatures[f.key] = true; });
   165→        MERCHANT_ADMIN_TABS.forEach((t) => { allFeatures[t.key] = true; });
   166→        MERCHANT_PERMISSIONS.forEach((p) => { allFeatures[p.key] = true; });
   167→        next.features = allFeatures;
   168→      }
   169→      // عند تغيير الخطة إلى مجانية، أوقف الميزات المدفوعة فقط
   170→      if (key === "plan" && val === "free") {
   171→        const paidOnly = ["aiAssistant", "loyaltyProgram", "priceCalculator", "multiBranch", "couponSystem", "customerSupport"];
   172→        const updated = { ...prev.features };
   173→        paidOnly.forEach((k) => { updated[k] = false; });
   174→        next.features = updated;
   175→      }
   176→      return next;
   177→    });
   178→    setHasChanges(true);
   179→  }
   180→
   181→  function toggleFeature(key: string) {
   182→    setData((prev) => {
   183→      if (!prev) return prev;
   184→      return { ...prev, features: { ...prev.features, [key]: !prev.features[key] } };
   185→    });
   186→    setHasChanges(true);
   187→  }
   188→
   189→  async function handleSave() {
   190→    if (!data) return;
   191→    setSaving(true);
   192→    try {
   193→      const payload: Record<string, unknown> = {
   194→        name: data.name, phone: data.phone, whatsapp: data.whatsapp,
   195→        email: data.email, address: data.address, country: data.country,
   196→        language: data.language, customCurrency: data.customCurrency,
   197→        ownerName: data.ownerName, ownerPhone: data.ownerPhone,
   198→        adminPin: data.adminPin, isActive: data.isActive, plan: data.plan,
   199→        primaryColor: data.primaryColor, themeId: data.themeId,
   200→        logoIcon: data.logoIcon, trialDays: data.trialDays,
   201→        features: data.features, ownerNotes: data.ownerNotes,
   202→        paymentInfo: data.paymentInfo,
   203→      };
   204→      const res = await fetch(`/api/admin/shops/${data.slug}`, {
   205→        method: "PUT",
   206→        headers: { "Content-Type": "application/json" },
   207→        body: JSON.stringify(payload),
   208→      });
   209→      if (res.ok) {
   210→        toast.success("تم حفظ إعدادات المتجر");
   211→        setOriginalData(JSON.parse(JSON.stringify(data)));
   212→        setHasChanges(false);
   213→        onSaved();
   214→      } else {
   215→        toast.error("فشل حفظ الإعدادات");
   216→      }
   217→    } catch { toast.error("خطأ في الاتصال"); }
   218→    finally { setSaving(false); }
   219→  }
   220→
   221→  const tabs = [
   222→    { key: "basic", label: "الأساسيات", icon: Store },
   223→    { key: "plan", label: "الخطة والتجربة", icon: CreditCard },
   224→    { key: "features", label: "ميزات المتجر", icon: Sparkles },
   225→    { key: "merchant", label: "لوحة التحكم", icon: LayoutDashboard },
   226→    { key: "appearance", label: "المظهر", icon: Palette },
   227→    { key: "notes", label: "ملاحظات", icon: FileText },
   228→  ];
   229→
   230→  // Trial status
   231→  const trialStatus = (() => {
   232→    if (!data?.trialDays || !data?.trialStartsAt) return null;
   233→    const start = new Date(data.trialStartsAt).getTime();
   234→    const end = start + data.trialDays * 86400000;
   235→    const now = Date.now();
   236→    const remaining = Math.max(0, Math.ceil((end - now) / 86400000));
   237→    const active = now < end;
   238→    return { active, remaining, total: data.trialDays, start: data.trialStartsAt, end: new Date(end).toISOString() };
   239→  })();
   240→
   241→  return (
   242→    <Dialog open={open} onOpenChange={(v) => { if (!v && hasChanges) { /* could warn */ } onClose(); }}>
   243→      <DialogContent className="max-w-3xl max-h-[90vh] overflow-hidden flex flex-col p-0" dir="rtl">
   244→        <div className="flex items-center justify-between px-6 py-4 border-b border-border/50 shrink-0">
   245→          <div>
   246→            <DialogTitle className="text-lg font-bold">إدارة المتجر</DialogTitle>
   247→            <DialogDescription className="text-xs mt-0.5">تحكم شامل بإعدادات وميزات المتجر</DialogDescription>
   248→          </div>
   249→          {hasChanges && (
   250→            <Badge variant="outline" className="text-amber-600 border-amber-300 bg-amber-50 dark:bg-amber-500/10 dark:border-amber-500/30 dark:text-amber-400">
   251→              تعديلات غير محفوظة
   252→            </Badge>
   253→          )}
   254→        </div>
   255→
   256→        {/* Tabs */}
   257→        <div className="flex gap-1 px-4 pt-3 pb-0 overflow-x-auto no-scrollbar shrink-0">
   258→          {tabs.map((t) => (
   259→            <button
   260→              key={t.key}
   261→              onClick={() => setTab(t.key)}
   262→              className={`flex items-center gap-1.5 px-3 py-2 rounded-lg text-xs font-medium whitespace-nowrap transition-all ${
   263→                tab === t.key
   264→                  ? "bg-amber-500/10 text-amber-600 dark:text-amber-400"
   265→                  : "text-muted-foreground hover:text-foreground hover:bg-muted/50"
   266→              }`}
   267→            >
   268→              <t.icon className="h-3.5 w-3.5" />
   269→              <span className="hidden sm:inline">{t.label}</span>
   270→            </button>
   271→          ))}
   272→        </div>
   273→
   274→        {/* Content */}
   275→        <div className="flex-1 overflow-y-auto px-6 py-4">
   276→          {loading ? (
   277→            <div className="space-y-4">
   278→              {[1,2,3].map((i) => <div key={i} className="h-10 rounded-lg bg-muted animate-pulse" />)}
   279→            </div>
   280→          ) : !data ? null : (
   281→            <>
   282→              {/* ===== BASIC TAB ===== */}
   283→              {tab === "basic" && (
   284→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   285→                  <div className="sm:col-span-2">
   286→                    <Label>اسم المتجر</Label>
   287→                    <Input value={data.name} onChange={(e) => setField("name", e.target.value)} className="mt-1" />
   288→                  </div>
   289→                  <div>
   290→                    <Label>المعرف (slug)</Label>
   291→                    <Input value={data.slug} disabled className="mt-1 bg-muted/50" />
   292→                    <p className="text-[10px] text-muted-foreground mt-1">لا يمكن تغييره بعد الإنشاء</p>
   293→                  </div>
   294→                  <div>
   295→                    <Label>هاتف المتجر</Label>
   296→                    <Input value={data.phone || ""} onChange={(e) => setField("phone", e.target.value || null)} className="mt-1" dir="ltr" />
   297→                  </div>
   298→                  <div>
   299→                    <Label>واتساب</Label>
   300→                    <Input value={data.whatsapp || ""} onChange={(e) => setField("whatsapp", e.target.value || null)} className="mt-1" dir="ltr" />
   301→                  </div>
   302→                  <div>
   303→                    <Label>البريد الإلكتروني</Label>
   304→                    <Input type="email" value={data.email || ""} onChange={(e) => setField("email", e.target.value || null)} className="mt-1" dir="ltr" />
   305→                  </div>
   306→                  <div className="sm:col-span-2">
   307→                    <Label>العنوان</Label>
   308→                    <Input value={data.address || ""} onChange={(e) => setField("address", e.target.value || null)} className="mt-1" />
   309→                  </div>
   310→                  <div>
   311→                    <Label>الدولة</Label>
   312→                    <Select value={data.country || "DZ"} onValueChange={(v) => setField("country", v)}>
   313→                      <SelectTrigger className="mt-1"><SelectValue /></SelectTrigger>
   314→                      <SelectContent>
   315→                        {ARAB_COUNTRIES.map((c) => (
   316→                          <SelectItem key={c.code} value={c.code}>
   317→                            <span className="flex items-center gap-2">
   318→                              <span>{c.flag}</span>
   319→                              <span>{c.nameAr}</span>
   320→                              <span className="text-muted-foreground text-xs">({c.code})</span>
   321→                            </span>
   322→                          </SelectItem>
   323→                        ))}
   324→                      </SelectContent>
   325→                    </Select>
   326→                  </div>
   327→                  <div>
   328→                    <Label>اللغة</Label>
   329→                    <Select value={data.language || "ar"} onValueChange={(v) => setField("language", v)}>
   330→                      <SelectTrigger className="mt-1"><SelectValue /></SelectTrigger>
   331→                      <SelectContent>
   332→                        <SelectItem value="ar">العربية</SelectItem>
   333→                        <SelectItem value="fr">الفرنسية</SelectItem>
   334→                        <SelectItem value="en">الإنجليزية</SelectItem>
   335→                        <SelectItem value="tr">التركية</SelectItem>
   336→                        <SelectItem value="es">الإسبانية</SelectItem>
   337→                      </SelectContent>
   338→                    </Select>
   339→                  </div>
   340→                  <div>
   341→                    <Label>العملة</Label>
   342→                    <Select
   343→                      value={data.customCurrency || ARAB_COUNTRIES.find(c => c.code === data.country)?.currencyCode || "DZD"}
   344→                      onValueChange={(v) => setField("customCurrency", v)}
   345→                    >
   346→                      <SelectTrigger className="mt-1"><SelectValue /></SelectTrigger>
   347→                      <SelectContent>
   348→                        {(() => {
   349→                          const country = ARAB_COUNTRIES.find(c => c.code === data.country);
   350→                          const countryCurrency = country ? [{ code: country.currencyCode, label: `${country.currencySymbol} ${country.currencyEn} (${country.nameAr})`, isDefault: true }] : [];
   351→                          const otherCurrencies = COUNTRY_CURRENCIES.filter(c => c.code !== (country?.currencyCode));
   352→                          return [
   353→                            ...countryCurrency.map(c => (
   354→                              <SelectItem key={c.code} value={c.code}>
   355→                                <span className="flex items-center gap-2">
   356→                                  <span className="text-xs bg-amber-100 dark:bg-amber-500/15 text-amber-700 dark:text-amber-400 px-1.5 py-0.5 rounded">افتراضي</span>
   357→                                  <span>{c.label}</span>
   358→                                </span>
   359→                              </SelectItem>
   360→                            )),
   361→                            ...otherCurrencies.map(c => (
   362→                              <SelectItem key={c.code} value={c.code}>
   363→                                {c.label}
   364→                              </SelectItem>
   365→                            )),
   366→                          ];
   367→                        })()}
   368→                      </SelectContent>
   369→                    </Select>
   370→                    <p className="text-[10px] text-muted-foreground mt-1">تتغير تلقائياً مع الدولة — يمكنك اختيار عملة أخرى</p>
   371→                  </div>
   372→                  <div>
   373→                    <Label>اسم صاحب المتجر</Label>
   374→                    <Input value={data.ownerName || ""} onChange={(e) => setField("ownerName", e.target.value || null)} className="mt-1" />
   375→                  </div>
   376→                  <div>
