    71→interface ShopData {
    72→  slug: string; name: string; phone: string | null;
    73→  ownerName: string | null; ownerPhone: string | null;
    74→  whatsapp: string | null; email: string | null; address: string | null;
    75→  country: string | null; isActive: boolean; plan: string;
    76→  primaryColor: string | null; adminPin: string;
    77→  themeId: number | null;
    78→  logoIcon: string | null;
    79→  trialDays: number | null; trialStartsAt: string | null;
    80→  features: Record<string, boolean>;
    81→  ownerNotes: string | null;
    82→  paymentInfo: string | null;
    83→}
    84→
    85→interface AdminShopManagementProps {
    86→  open: boolean;
    87→  onClose: () => void;
    88→  shop: { slug: string; name: string } | null;
    89→  onSaved: () => void;
    90→}
    91→
    92→export function AdminShopManagement({ open, onClose, shop, onSaved }: AdminShopManagementProps) {
    93→  const [data, setData] = useState<ShopData | null>(null);
    94→  const [loading, setLoading] = useState(false);
    95→  const [saving, setSaving] = useState(false);
    96→  const [tab, setTab] = useState("basic");
    97→  const [hasChanges, setHasChanges] = useState(false);
    98→  const [originalData, setOriginalData] = useState<ShopData | null>(null);
    99→
   100→  useEffect(() => {
   101→    if (!open || !shop) return;
   102→    let cancelled = false;
   103→    (async () => {
   104→      setLoading(true);
   105→      setTab("basic");
   106→      try {
   107→        const res = await fetch(`/api/admin/shops/${shop.slug}`);
   108→        if (res.ok && !cancelled) {
   109→          const d = await res.json();
   110→          const s = d.shop;
   111→          const parsed: ShopData = {
   112→            slug: s.slug, name: s.name, phone: s.phone,
   113→            ownerName: s.ownerName, ownerPhone: s.ownerPhone,
   114→            whatsapp: s.whatsapp, email: s.email, address: s.address,
   115→            country: s.country, isActive: s.isActive,
   116→            plan: s.plan || "free", primaryColor: s.primaryColor,
   117→            adminPin: s.adminPin, themeId: s.themeId || 1,
   118→            logoIcon: s.logoIcon || "Printer",
   119→            trialDays: s.trialDays, trialStartsAt: s.trialStartsAt,
   120→            features: s.features ? (typeof s.features === "string" ? JSON.parse(s.features) : s.features) : {},
   121→            ownerNotes: s.ownerNotes, paymentInfo: s.paymentInfo,
   122→          };
   123→          setData(parsed);
   124→          setOriginalData(JSON.parse(JSON.stringify(parsed)));
   125→          setHasChanges(false);
   126→        } else if (!cancelled) {
   127→          toast.error("فشل تحميل بيانات المتجر");
   128→          onClose();
   129→        }
   130→      } catch { if (!cancelled) { toast.error("خطأ في الاتصال"); onClose(); } }
   131→      finally { if (!cancelled) setLoading(false); }
   132→    })();
   133→    return () => { cancelled = true; };
   134→  }, [open, shop, onClose]);
   135→
   136→  function setField<K extends keyof ShopData>(key: K, val: ShopData[K]) {
   137→    setData((prev) => prev ? { ...prev, [key]: val } : prev);
   138→    setHasChanges(true);
   139→  }
   140→
   141→  function toggleFeature(key: string) {
   142→    setData((prev) => {
   143→      if (!prev) return prev;
   144→      return { ...prev, features: { ...prev.features, [key]: !prev.features[key] } };
   145→    });
   146→    setHasChanges(true);
   147→  }
   148→
   149→  async function handleSave() {
   150→    if (!data) return;
   151→    setSaving(true);
   152→    try {
   153→      const payload: Record<string, unknown> = {
   154→        name: data.name, phone: data.phone, whatsapp: data.whatsapp,
   155→        email: data.email, address: data.address, country: data.country,
   156→        ownerName: data.ownerName, ownerPhone: data.ownerPhone,
   157→        adminPin: data.adminPin, isActive: data.isActive, plan: data.plan,
   158→        primaryColor: data.primaryColor, themeId: data.themeId,
   159→        logoIcon: data.logoIcon, trialDays: data.trialDays,
   160→        features: data.features, ownerNotes: data.ownerNotes,
   161→        paymentInfo: data.paymentInfo,
   162→      };
   163→      const res = await fetch(`/api/admin/shops/${data.slug}`, {
   164→        method: "PUT",
   165→        headers: { "Content-Type": "application/json" },
   166→        body: JSON.stringify(payload),
   167→      });
   168→      if (res.ok) {
   169→        toast.success("تم حفظ إعدادات المتجر");
   170→        setOriginalData(JSON.parse(JSON.stringify(data)));
   171→        setHasChanges(false);
   172→        onSaved();
   173→      } else {
   174→        toast.error("فشل حفظ الإعدادات");
   175→      }
   176→    } catch { toast.error("خطأ في الاتصال"); }
   177→    finally { setSaving(false); }
   178→  }
   179→
   180→  const tabs = [
   181→    { key: "basic", label: "الأساسيات", icon: Store },
   182→    { key: "plan", label: "الخطة والتجربة", icon: CreditCard },
   183→    { key: "features", label: "ميزات المتجر", icon: Sparkles },
   184→    { key: "merchant", label: "لوحة التحكم", icon: LayoutDashboard },
   185→    { key: "appearance", label: "المظهر", icon: Palette },
   186→    { key: "notes", label: "ملاحظات", icon: FileText },
   187→  ];
   188→
   189→  // Trial status
   190→  const trialStatus = (() => {
   191→    if (!data?.trialDays || !data?.trialStartsAt) return null;
   192→    const start = new Date(data.trialStartsAt).getTime();
   193→    const end = start + data.trialDays * 86400000;
   194→    const now = Date.now();
   195→    const remaining = Math.max(0, Math.ceil((end - now) / 86400000));
   196→    const active = now < end;
   197→    return { active, remaining, total: data.trialDays, start: data.trialStartsAt, end: new Date(end).toISOString() };
   198→  })();
   199→
   200→  return (
   201→    <Dialog open={open} onOpenChange={(v) => { if (!v && hasChanges) { /* could warn */ } onClose(); }}>
   202→      <DialogContent className="max-w-3xl max-h-[90vh] overflow-hidden flex flex-col p-0" dir="rtl">
   203→        <div className="flex items-center justify-between px-6 py-4 border-b border-border/50 shrink-0">
   204→          <div>
   205→            <DialogTitle className="text-lg font-bold">إدارة المتجر</DialogTitle>
   206→            <DialogDescription className="text-xs mt-0.5">تحكم شامل بإعدادات وميزات المتجر</DialogDescription>
   207→          </div>
   208→          {hasChanges && (
   209→            <Badge variant="outline" className="text-amber-600 border-amber-300 bg-amber-50 dark:bg-amber-500/10 dark:border-amber-500/30 dark:text-amber-400">
   210→              تعديلات غير محفوظة
   211→            </Badge>
   212→          )}
   213→        </div>
   214→
   215→        {/* Tabs */}
   216→        <div className="flex gap-1 px-4 pt-3 pb-0 overflow-x-auto no-scrollbar shrink-0">
   217→          {tabs.map((t) => (
   218→            <button
   219→              key={t.key}
   220→              onClick={() => setTab(t.key)}
   221→              className={`flex items-center gap-1.5 px-3 py-2 rounded-lg text-xs font-medium whitespace-nowrap transition-all ${
   222→                tab === t.key
   223→                  ? "bg-amber-500/10 text-amber-600 dark:text-amber-400"
   224→                  : "text-muted-foreground hover:text-foreground hover:bg-muted/50"
   225→              }`}
   226→            >
   227→              <t.icon className="h-3.5 w-3.5" />
   228→              <span className="hidden sm:inline">{t.label}</span>
   229→            </button>
   230→          ))}
   231→        </div>
   232→
   233→        {/* Content */}
   234→        <div className="flex-1 overflow-y-auto px-6 py-4">
   235→          {loading ? (
   236→            <div className="space-y-4">
   237→              {[1,2,3].map((i) => <div key={i} className="h-10 rounded-lg bg-muted animate-pulse" />)}
   238→            </div>
   239→          ) : !data ? null : (
   240→            <>
   241→              {/* ===== BASIC TAB ===== */}
   242→              {tab === "basic" && (
   243→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   244→                  <div className="sm:col-span-2">
   245→                    <Label>اسم المتجر</Label>
   246→                    <Input value={data.name} onChange={(e) => setField("name", e.target.value)} className="mt-1" />
   247→                  </div>
   248→                  <div>
   249→                    <Label>المعرف (slug)</Label>
   250→                    <Input value={data.slug} disabled className="mt-1 bg-muted/50" />
   251→                    <p className="text-[10px] text-muted-foreground mt-1">لا يمكن تغييره بعد الإنشاء</p>
   252→                  </div>
   253→                  <div>
   254→                    <Label>هاتف المتجر</Label>
   255→                    <Input value={data.phone || ""} onChange={(e) => setField("phone", e.target.value || null)} className="mt-1" dir="ltr" />
   256→                  </div>
   257→                  <div>
   258→                    <Label>واتساب</Label>
   259→                    <Input value={data.whatsapp || ""} onChange={(e) => setField("whatsapp", e.target.value || null)} className="mt-1" dir="ltr" />
   260→                  </div>
   261→                  <div>
   262→                    <Label>البريد الإلكتروني</Label>
   263→                    <Input type="email" value={data.email || ""} onChange={(e) => setField("email", e.target.value || null)} className="mt-1" dir="ltr" />
   264→                  </div>
   265→                  <div className="sm:col-span-2">
   266→                    <Label>العنوان</Label>
   267→                    <Input value={data.address || ""} onChange={(e) => setField("address", e.target.value || null)} className="mt-1" />
   268→                  </div>
   269→                  <div>
   270→                    <Label>الدولة</Label>
   271→                    <Select value={data.country || "DZ"} onValueChange={(v) => setField("country", v)}>
   272→                      <SelectTrigger className="mt-1"><SelectValue /></SelectTrigger>
   273→                      <SelectContent>
   274→                        {ARAB_COUNTRIES.map((c) => <SelectItem key={c.code} value={c.code}>{c.name} ({c.code})</SelectItem>)}
   275→                      </SelectContent>
   276→                    </Select>
   277→                  </div>
   278→                  <div>
   279→                    <Label>اسم صاحب المتجر</Label>
   280→                    <Input value={data.ownerName || ""} onChange={(e) => setField("ownerName", e.target.value || null)} className="mt-1" />
   281→                  </div>
   282→                  <div>
   283→                    <Label>هاتف صاحب المتجر</Label>
   284→                    <Input value={data.ownerPhone || ""} onChange={(e) => setField("ownerPhone", e.target.value || null)} className="mt-1" dir="ltr" />
   285→                  </div>
   286→                  <div>
   287→                    <Label>رمز PIN الإدارة</Label>
   288→                    <Input value={data.adminPin} onChange={(e) => setField("adminPin", e.target.value)} className="mt-1" dir="ltr" />
   289→                  </div>

   291→                    <div>
   292→                      <div className="font-medium text-sm">حالة المتجر</div>
   293→                      <div className="text-xs text-muted-foreground">تفعيل أو إيقاف المتجر</div>
   294→                    </div>
   295→                    <Switch checked={data.isActive} onCheckedChange={(v) => setField("isActive", v)} />
   296→                  </div>
   297→                </div>
   298→              )}
   299→
   300→              {/* ===== PLAN & TRIAL TAB ===== */}
   301→              {tab === "plan" && (
   302→                <div className="space-y-6">
   303→                  {/* Plan selection */}
   304→                  <div>
   305→                    <Label className="text-sm font-semibold">الخطة الحالية</Label>
   306→                    <div className="grid grid-cols-2 gap-3 mt-2">
   307→                      {[{ value: "free", label: "مجانية", desc: "ميزات أساسية محدودة", color: "border-neutral-300 dark:border-neutral-600" },
   308→                        { value: "paid", label: "مدفوعة", desc: "جميع الميزات متاحة", color: "border-amber-400" }
   309→                      ].map((p) => (
   310→                        <button
   311→                          key={p.value}
   312→                          type="button"
   313→                          onClick={() => setField("plan", p.value)}
   314→                          className={`p-4 rounded-xl border-2 text-right transition-all ${
   315→                            data.plan === p.value ? p.color + " bg-amber-50/50 dark:bg-amber-500/5" : "border-border/50 hover:border-border"
   316→                          }`}
   317→                        >
   318→                          <div className="font-bold text-sm">{p.label}</div>
   319→                          <div className="text-xs text-muted-foreground mt-1">{p.desc}</div>
   320→                          {data.plan === p.value && <CheckCircle2 className="h-4 w-4 text-amber-500 mt-2" />}
   321→                        </button>
   322→                      ))}
   323→                    </div>
   324→                  </div>
   325→
   326→                  <Separator />
   327→
   328→                  {/* Trial period */}
   329→                  <div>
   330→                    <Label className="text-sm font-semibold">فترة التجربة</Label>
   331→                    <p className="text-xs text-muted-foreground mt-1">تحديد مدة تجربة مجانية للتاجر. اتركه فارغاً لبلا حدود.</p>

   333→                      <div>
   334→                        <Label className="text-xs">مدة التجربة (أيام)</Label>
   335→                        <Input
   336→                          type="number" min="0" max="365"
   337→                          value={data.trialDays ?? ""}
   338→                          onChange={(e) => setField("trialDays", e.target.value ? Number(e.target.value) : null)}
   339→                          className="mt-1" dir="ltr" placeholder="مثلاً: 14"
   340→                        />
   341→                      </div>
   342→                      <div>
   343→                        <Label className="text-xs">تاريخ البداية</Label>
   344→                        <Input
   345→                          type="date"
   346→                          value={data.trialStartsAt ? new Date(data.trialStartsAt).toISOString().split("T")[0] : ""}
   347→                          onChange={(e) => setField("trialStartsAt", e.target.value || null)}
   348→                          className="mt-1" dir="ltr"
   349→                        />
   350→                      </div>
   351→                    </div>
   352→
   353→                    {/* Trial status card */}
   354→                    {trialStatus && (
   355→                      <div className={`mt-4 p-4 rounded-xl border ${

   358→                          : "bg-rose-50 dark:bg-rose-500/5 border-rose-200 dark:border-rose-500/20"
   359→                      }`}>
   360→                        <div className="flex items-center gap-2 mb-2">
   361→                          {trialStatus.active ? (
   362→                            <><Clock className="h-4 w-4 text-emerald-600" /><span className="font-semibold text-sm text-emerald-700 dark:text-emerald-400">التجربة سارية</span></>
   363→                          ) : (
   364→                            <><AlertTriangle className="h-4 w-4 text-rose-600" /><span className="font-semibold text-sm text-rose-700 dark:text-rose-400">انتهت فترة التجربة</span></>
   365→                          )}
   366→                        </div>
   367→                        <div className="text-xs text-muted-foreground space-y-1">

   371→                          ) : (
   372→                            <div>انتهت منذ {Math.abs(trialStatus.remaining)} يوم</div>
   373→                          )}
   374→                          <div>تاريخ البداية: {new Date(trialStatus.start).toLocaleDateString("ar-DZ")}</div>
   375→                        </div>
   376→                        {!trialStatus.active && (
   377→                          <Button
   378→                            size="sm"
   379→                            variant="outline"
   380→                            className="mt-3 gap-1 text-xs"
   381→                            onClick={() => {
   382→                              setField("trialStartsAt", new Date().toISOString());
   383→                            }}
   384→                          >
   385→                            <RefreshCw className="h-3 w-3" /> إعادة تفعيل التجربة
   386→                          </Button>
   387→                        )}
   388→                      </div>
   389→                    )}
   390→                  </div>
   391→                </div>
   392→              )}

   394→              {/* ===== FEATURES TAB ===== */}

   400→                  </div>
   401→                  <div className="grid grid-cols-1 sm:grid-cols-2 gap-2">
   402→                    {SHOP_FEATURES.map((f) => {
   403→                      const enabled = !!data.features[f.key];

   406→                          key={f.key}
   407→                          className={`flex items-center justify-between p-3 rounded-xl border transition-all cursor-pointer ${
