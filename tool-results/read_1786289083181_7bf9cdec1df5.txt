    99→    99→/** نسخة عميقة آمنة (للإعدادات القابلة للتسلسل JSON) */
   100→   100→function deepClone<T>(value: T): T {
   101→   101→  return JSON.parse(JSON.stringify(value)) as T;

   179→   179→  const [original, setOriginal] = useState<AppSettings | null>(null);
   180→   180→  const [loading, setLoading] = useState(true);
   181→   181→  const [saving, setSaving] = useState(false);

   187→   187→    setLoading(true);
   188→   188→    try {
   189→   189→      const res = await fetch("/api/settings");
   190→   190→      if (!res.ok) throw new Error("فشل تحميل الإعدادات");
   191→   191→      const data = (await res.json()) as AppSettings;
   192→   192→      const safe: AppSettings = {
   193→   193→        services: data.services ?? DEFAULT_SETTINGS.services,
   194→   194→        deliveryOptions: data.deliveryOptions ?? DEFAULT_SETTINGS.deliveryOptions,

   200→   200→      toast.error("تعذّر تحميل الإعدادات", {
   201→   201→        description: (e as Error).message,
   202→   202→      });

   204→   204→      setSettings(deepClone(DEFAULT_SETTINGS));
   205→   205→      setOriginal(deepClone(DEFAULT_SETTINGS));

   207→   207→      setLoading(false);
   208→   208→    }
   209→   209→  }, []);
