    85→// ===== Login Gate =====
    86→function InlineLoginGate({ onUnlock }: { onUnlock: () => void }) {
    87→  const [password, setPassword] = useState('');
    88→  const [error, setError] = useState(false);
    89→  const [loading, setLoading] = useState(false);
    90→
    91→  async function handleSubmit(e: React.FormEvent) {
    92→    e.preventDefault();
    93→    if (!password) return;
    94→    setLoading(true); setError(false);
    95→    try {
    96→      const res = await fetch('/api/super-admin/auth', {
    97→        method: 'POST',
    98→        headers: { 'Content-Type': 'application/json' },
    99→        body: JSON.stringify({ password }),
   100→      });
   101→      const data = await res.json();
   102→      if (data.success) { markAuthenticated(data.token, data.ts); onUnlock(); }
   103→      else setError(true);
   104→    } catch { setError(true); }
   105→    finally { setLoading(false); }
   106→  }
   107→
   108→  return (
   109→    <div className="min-h-screen flex items-center justify-center bg-background p-4" dir="rtl">
   110→      <Card className="max-w-sm w-full">
   111→        <CardContent className="p-6 space-y-4">
   112→          <div className="flex flex-col items-center gap-3">
   113→            <img src="/n.png" alt="طيف" className="w-16 h-16 rounded-2xl object-contain shadow-lg" />
   114→            <div className="text-center">
   115→              <h1 className="text-xl font-bold">لوحة تحكم المنصة</h1>
   116→              <p className="text-sm text-muted-foreground">أدخل كلمة المرور للدخول</p>
   117→            </div>
   118→          </div>
   119→          <form onSubmit={handleSubmit} className="space-y-3">
   120→            <div className="relative">
   121→              <Lock className="absolute right-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   122→              <Input type="password" placeholder="كلمة المرور" value={password} onChange={(e) => { setPassword(e.target.value); setError(false); }} className="pr-9" dir="ltr" />
   123→            </div>
   124→            {error && <p className="text-destructive text-xs text-center">كلمة المرور غير صحيحة</p>}
   125→            <Button type="submit" className="w-full bg-gradient-to-l from-amber-500 to-amber-600 text-white" disabled={loading}>
   126→              {loading ? 'جاري التحقق...' : 'دخول'}
   127→            </Button>
   128→          </form>
   129→          <div className="flex items-center justify-center gap-2 pt-2"><ThemeToggle /></div>
   130→        </CardContent>
   131→      </Card>
   132→    </div>
   133→  );
   134→}

   136→// ===== Create Shop Dialog =====
   137→function InlineCreateShop({ open, onClose, onCreated }: { open: boolean; onClose: () => void; onCreated: () => void }) {
   138→  const [name, setName] = useState('');
   139→  const [slug, setSlug] = useState('');
   140→  const [pin, setPin] = useState('');
   141→  const [ownerName, setOwnerName] = useState('');
   142→  const [ownerPhone, setOwnerPhone] = useState('');
   143→  const [country, setCountry] = useState('DZ');
   144→  const [submitting, setSubmitting] = useState(false);
   145→  const [created, setCreated] = useState<{ name: string; slug: string; pin: string } | null>(null);
   146→
   147→  function genSlug(t: string) { return t.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/^-|-$/g, ''); }
   148→  function handleName(v: string) { setName(v); if (!slug || slug === genSlug(name)) setSlug(genSlug(v)); }
   149→
   150→  async function handleSubmit(e: React.FormEvent) {
   151→    e.preventDefault();
   152→    if (!name || !slug || !pin) return;
   153→    setSubmitting(true);
   154→    try {
   155→      const res = await fetch('/api/shops', {
   156→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   157→        body: JSON.stringify({ name, slug, adminPin: pin, ownerName, ownerPhone, country }),
   158→      });
   159→      if (res.ok) { setCreated({ name, slug, pin }); onCreated(); }
   160→      else { const data = await res.json().catch(() => ({})); toast.error(data.error || 'فشل إنشاء المتجر'); }
   161→    } catch { toast.error('خطأ في الاتصال'); }
   162→    finally { setSubmitting(false); }
   163→  }
   164→
   165→  function handleClose() {
   166→    setName(''); setSlug(''); setPin(''); setOwnerName(''); setOwnerPhone('');
   167→    setCreated(null); onClose();
   168→  }
   169→
   170→  return (
   171→    <Dialog open={open} onOpenChange={(o) => !o && handleClose()}>
   172→      <DialogContent className="sm:max-w-md" dir="rtl">
   173→        <DialogTitle>{created ? 'تم إنشاء المتجر ✓' : 'إنشاء متجر جديد'}</DialogTitle>
   174→        {created ? (
   175→          <div className="space-y-4 py-4">
   176→            <div className="text-center space-y-2">
   177→              <p className="font-bold text-lg">{created.name}</p>
   178→              <div className="space-y-2 text-sm">
   179→                <div className="flex items-center justify-between p-2 rounded-lg bg-muted">
   180→                  <span className="text-muted-foreground">رابط المتجر:</span>
   181→                  <button onClick={() => robustCopy(`/s/${created.slug}`, 'تم نسخ الرابط', '')} className="font-mono text-xs text-primary hover:underline">/s/{created.slug}</button>
   182→                </div>
   183→                <div className="flex items-center justify-between p-2 rounded-lg bg-muted">
   184→                  <span className="text-muted-foreground">كلمة المرور:</span>
   185→                  <span className="font-mono text-xs font-bold">{created.pin}</span>
   186→                </div>
   187→              </div>
   188→            </div>
   189→            <Button onClick={handleClose} className="w-full">تم</Button>
   190→          </div>
   191→        ) : (
   192→          <form onSubmit={handleSubmit} className="space-y-4">
   193→            <div className="space-y-2"><Label>اسم المتجر *</Label><Input value={name} onChange={(e) => handleName(e.target.value)} placeholder="مثال: مطبعة النور" /></div>
   194→            <div className="space-y-2"><Label>المعرّف (slug) *</Label><Input value={slug} onChange={(e) => setSlug(e.target.value)} placeholder="matbaa-alnoor" dir="ltr" /></div>
   195→            <div className="space-y-2"><Label>كلمة مرور لوحة التحكم (PIN) *</Label><Input value={pin} onChange={(e) => setPin(e.target.value)} placeholder="1234" maxLength={10} dir="ltr" type="text" /></div>
   196→            <div className="grid grid-cols-2 gap-3">
   197→              <div className="space-y-2"><Label>اسم المالك</Label><Input value={ownerName} onChange={(e) => setOwnerName(e.target.value)} placeholder="اختياري" /></div>
   198→              <div className="space-y-2"><Label>هاتف المالك</Label><Input value={ownerPhone} onChange={(e) => setOwnerPhone(e.target.value)} placeholder="اختياري" dir="ltr" /></div>
   199→            </div>
   200→            <div className="space-y-2">
   201→              <Label>الدولة</Label>
   202→              <Select value={country} onValueChange={setCountry}>
   203→                <SelectTrigger><SelectValue /></SelectTrigger>
   204→                <SelectContent>{ARAB_COUNTRIES.map((c) => (<SelectItem key={c.code} value={c.code}>{c.flag} {c.name_ar}</SelectItem>))}</SelectContent>
   205→              </Select>
   206→            </div>
   207→            <div className="flex gap-2 pt-2">
   208→              <Button type="button" variant="outline" onClick={handleClose} className="flex-1">إلغاء</Button>
   209→              <Button type="submit" disabled={!name || !slug || !pin || submitting} className="flex-1 bg-gradient-to-l from-amber-500 to-amber-600 text-white">
   210→                {submitting ? 'جاري الإنشاء...' : 'إنشاء'}
   211→              </Button>
   212→            </div>
   213→          </form>
   214→        )}
   215→      </DialogContent>
   216→    </Dialog>
   217→  );
   218→}

   221→interface EditShopData {
   222→  slug: string; name: string; phone: string | null;

   229→function InlineEditShop({ open, onClose, shop, onSaved }: {
   230→  open: boolean; onClose: () => void;
   231→  shop: ShopItem | null; onSaved: () => void;
   232→}) {
   233→  const [data, setData] = useState<EditShopData | null>(null);
   234→  const [loading, setLoading] = useState(false);
   235→  const [saving, setSaving] = useState(false);
   236→
   237→  // جلب بيانات المتجر الكاملة عند الفتح
   238→  useEffect(() => {
   239→    if (!open || !shop) return;
   240→    let cancelled = false;
   241→    (async () => {
   242→      setLoading(true);
   243→      try {
   244→        const res = await fetch(`/api/admin/shops/${shop.slug}`);
   245→        if (res.ok && !cancelled) {
   246→          const d = await res.json();
   247→          const s = d.shop;
   248→          setData({
   249→            slug: s.slug, name: s.name, phone: s.phone,
   250→            ownerName: s.ownerName, ownerPhone: s.ownerPhone,
   251→            whatsapp: s.whatsapp, email: s.email, address: s.address,
   252→            country: s.country, isActive: s.isActive,
   253→            plan: s.plan || 'free', primaryColor: s.primaryColor,
   254→            adminPin: s.adminPin,
   255→          });
   256→        } else if (!cancelled) {
   257→          toast.error('فشل تحميل بيانات المتجر');
   258→          onClose();
   259→        }
   260→      } catch { if (!cancelled) { toast.error('خطأ في الاتصال'); onClose(); } }
   261→      finally { if (!cancelled) setLoading(false); }
   262→    })();
   263→    return () => { cancelled = true; };
   264→  }, [open, shop, onClose]);
   265→
   266→  async function handleSave(e: React.FormEvent) {
   267→    e.preventDefault();
   268→    if (!data) return;
   269→    setSaving(true);
   270→    try {
   271→      const res = await fetch(`/api/admin/shops/${data.slug}`, {
   272→        method: 'PUT',
   273→        headers: { 'Content-Type': 'application/json' },
   274→        body: JSON.stringify({
   275→          name: data.name, phone: data.phone, ownerName: data.ownerName,
   276→          ownerPhone: data.ownerPhone, whatsapp: data.whatsapp,
   277→          email: data.email, address: data.address, country: data.country,
   278→          isActive: data.isActive, plan: data.plan,
   279→          primaryColor: data.primaryColor, adminPin: data.adminPin,
   280→        }),
   281→      });
   282→      if (res.ok) { toast.success('تم تحديث المتجر بنجاح'); onSaved(); onClose(); }
   283→      else { const d = await res.json().catch(() => ({})); toast.error(d.error || 'فشل التحديث'); }
   284→    } catch { toast.error('خطأ في الاتصال'); }
   285→    finally { setSaving(false); }
   286→  }
   287→
   288→  function setField<K extends keyof EditShopData>(key: K, val: EditShopData[K]) {
   289→    setData((prev) => (prev ? { ...prev, [key]: val } : prev));
   290→  }
   291→
   292→  if (!open) return null;
   293→
   294→  return (
   295→    <Dialog open={open} onOpenChange={(o) => !o && onClose()}>
   296→      <DialogContent className="sm:max-w-lg max-h-[90vh] overflow-y-auto" dir="rtl">

   454→// ===== Mini Bar Chart (CSS-based, no Recharts) =====
   455→function MiniBarChart({ data, maxValue, color = 'bg-primary' }: { data: number[]; maxValue?: number; color?: string }) {
   456→  const max = maxValue || Math.max(...data, 1);
   457→  return (
   458→    <div className="flex items-end gap-1 h-12">
   459→      {data.map((v, i) => (
   460→        <div key={i} className="flex-1 flex flex-col items-center gap-0.5">
   461→          <div className={`${color} rounded-t-sm w-full transition-all duration-500`} style={{ height: `${Math.max((v / max) * 48, 2)}px` }} />
   462→        </div>
   463→      ))}
   464→    </div>
   465→  );
   466→}
