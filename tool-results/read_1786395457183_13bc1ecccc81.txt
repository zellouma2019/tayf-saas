    96→function isImageFile(fileType: string | null): boolean {
    97→  if (!fileType) return false;
    98→  const t = fileType.toUpperCase();
    99→  return ["JPG", "JPEG", "PNG", "WEBP", "GIF"].includes(t);
   100→}

   102→/// هل الملف PDF؟
   103→function isPdfFile(fileType: string | null): boolean {
   104→  if (!fileType) return false;
   105→  return fileType.toUpperCase() === "PDF";
   106→}
