   262→   262→
   263→   263→function AnimatedCounter({
   264→   264→  value,
   265→   265→  duration = 1.2,
   266→   266→  suffix = "",
   267→   267→}: {
   268→   268→  value: number;
   269→   269→  duration?: number;
   270→   270→  suffix?: string;
   271→   271→}) {
   272→   272→  const [display, setDisplay] = useState(0);
   273→   273→  const prevValue = useRef(0);
   274→   274→
   275→   275→  useEffect(() => {
   276→   276→    const start = prevValue.current;
   277→   277→    const end = value;
   278→   278→    const startTime = performance.now();
   279→   279→
   280→   280→    function animate(now: number) {
   281→   281→      const elapsed = (now - startTime) / (duration * 1000);
   282→   282→      const progress = Math.min(elapsed, 1);
   283→   283→      // easeOutExpo
   284→   284→      const eased = progress === 1 ? 1 : 1 - Math.pow(2, -10 * progress);
   285→   285→      setDisplay(Math.round(start + (end - start) * eased));
   286→   286→      if (progress < 1) requestAnimationFrame(animate);
   287→   287→    }
   288→   288→
   289→   289→    requestAnimationFrame(animate);
   290→   290→    prevValue.current = value;
   291→   291→  }, [value, duration]);
   292→   292→
   293→   293→  return <>{display.toLocaleString("en-US")}{suffix}</>;
   294→   294→}
   295→   295→
   296→   296→/* ═══════════════════════════════════════════════════════
   297→   297→   Sub-Components: CircularGauge
   298→   298→   ═══════════════════════════════════════════════════════ */
   299→   299→
   300→   300→function CircularGauge({
   301→   301→  value,
   302→   302→  max,
   303→   303→  size = 64,
   304→   304→  smSize,
   305→   305→  strokeWidth = 5,
   306→   306→  color,
   307→   307→  label,
   308→   308→  tooltipText,
   309→   309→}: {
   310→   310→  value: number;
   311→   311→  max: number;
   312→   312→  size?: number;
   313→   313→  smSize?: number;
   314→   314→  strokeWidth?: number;
   315→   315→  color: string;
   316→   316→  label: string;
   317→   317→  tooltipText?: string;
   318→   318→}) {
   319→   319→  const radius = (size - strokeWidth) / 2;
   320→   320→  const circumference = 2 * Math.PI * radius;
   321→   321→  const percentage = Math.min(value / max, 1);
   322→   322→  const offset = circumference - percentage * circumference;
   323→   323→  // Use CSS to make the gauge responsive: smaller on mobile, larger on desktop (smSize)
   324→   324→  const responsiveClass = smSize ? "w-14 h-14 sm:w-[72px] sm:h-[72px]" : "";
   325→   325→
   326→   326→  const gauge = (
   327→   327→    <div className="flex flex-col items-center gap-1">
   328→   328→      <div className={`relative ${responsiveClass}`} style={!smSize ? { width: size, height: size } : undefined}>
   329→   329→        <svg width="100%" height="100%" viewBox={`0 0 ${size} ${size}`} className="-rotate-90">
   330→   330→          <circle
   331→   331→            cx={size / 2}
   332→   332→            cy={size / 2}
   333→   333→            r={radius}
   334→   334→            fill="none"
   335→   335→            stroke="currentColor"
   336→   336→            strokeWidth={strokeWidth}
   337→   337→            className="text-muted/40"
   338→   338→          />
   339→   339→          <motion.circle
   340→   340→            cx={size / 2}
   341→   341→            cy={size / 2}
   342→   342→            r={radius}
   343→   343→            fill="none"
   344→   344→            stroke={color}
   345→   345→            strokeWidth={strokeWidth}
   346→   346→            strokeLinecap="round"
   347→   347→            strokeDasharray={circumference}
   348→   348→            initial={{ strokeDashoffset: circumference }}
   349→   349→            animate={{ strokeDashoffset: offset }}
   350→   350→            transition={{ duration: 1.4, ease: "easeOut", delay: 0.3 }}
   351→   351→          />
   352→   352→        </svg>
   353→   353→        <div className="absolute inset-0 flex items-center justify-center">
   354→   354→          <span className="text-xs sm:text-sm font-bold" style={{ color }}>
   355→   355→            <AnimatedCounter value={value} />
   356→   356→          </span>
   357→   357→        </div>
   358→   358→      </div>
   359→   359→      <span className="text-[10px] text-muted-foreground font-medium">{label}</span>
   360→   360→    </div>
   361→   361→  );
   362→   362→
   363→   363→  if (tooltipText) {
   364→   364→    return (
   365→   365→      <Tooltip>
   366→   366→        <TooltipTrigger asChild>
   367→   367→          <div className="relative cursor-help">{gauge}</div>
   368→   368→        </TooltipTrigger>
   369→   369→        <TooltipContent side="top" className="max-w-[220px] text-center">
   370→   370→          {tooltipText}
   371→   371→        </TooltipContent>
   372→   372→      </Tooltip>
   373→   373→    );
   374→   374→  }
   375→   375→
   376→   376→  return <div className="relative">{gauge}</div>;
   377→   377→}
