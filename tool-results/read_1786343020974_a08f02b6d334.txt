   131→function useCountdown(targetDate: string | undefined) {
   132→  const [secondsLeft, setSecondsLeft] = useState(() =>
   133→    targetDate ? Math.max(0, Math.round((new Date(targetDate).getTime() - Date.now()) / 1000)) : 0,
   134→  );
   135→
   136→  useEffect(() => {
   137→    if (!targetDate) return;
   138→    const target = new Date(targetDate).getTime();
   139→    const tick = () => {
   140→      const remaining = Math.max(0, Math.round((target - Date.now()) / 1000));
   141→      setSecondsLeft(remaining);
   142→      if (remaining > 0) requestAnimationFrame(() => setTimeout(tick, 500));
   143→    };
   144→    requestAnimationFrame(() => setTimeout(tick, 0));
   145→  }, [targetDate]);
   146→
   147→  const minutes = Math.floor(secondsLeft / 60);
   148→  const seconds = secondsLeft % 60;
   149→  const isExpired = secondsLeft <= 0;
   150→  const percentage = targetDate ? Math.max(0, (secondsLeft / 900) * 100) : 0;
   151→
   152→  return { secondsLeft, minutes, seconds, isExpired, percentage, formatted: `${minutes}:${seconds.toString().padStart(2, "0")}` };
   153→}
