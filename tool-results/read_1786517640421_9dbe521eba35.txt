    72→const SERVICE_EMOJI: Record<string, string> = {
    73→  document: "🖨️",
    74→  photo: "🖼️",
    75→  binding: "📚",
    76→  copy: "📄",
    77→  card: "🪪",
    78→  poster: "📜",
    79→};
    80→
    81→const IMAGE_EXTENSIONS = ["jpg", "jpeg", "png", "webp", "gif", "bmp"];
    82→
    83→function isImageFile(fileName: string | null | undefined): boolean {
    84→  if (!fileName) return false;
    85→  const ext = fileName.split(".").pop()?.toLowerCase() || "";
    86→  return IMAGE_EXTENSIONS.includes(ext);
    87→}
    88→
    89→function isPdfFile(fileName: string | null | undefined): boolean {
    90→  return !!fileName?.toLowerCase().endsWith(".pdf");
    91→}

    93→export function DirectPrintPreviewDialog({
    94→  order,
    95→  open,
    96→  onClose,
    97→  shopId,
    98→  shopName,
    99→  shopPhone,
   100→  shopAddress,
   101→  onPrintStart,
   102→  onPrintComplete,
   103→}: DirectPrintPreviewDialogProps) {
   104→  const [fileUrl, setFileUrl] = useState<string | null>(null);
   105→  const [fileLoading, setFileLoading] = useState(false);
   106→  const [verify, setVerify] = useState<VerifyResult | null>(null);
   107→  const [verifyLoading, setVerifyLoading] = useState(false);
   108→  const [confirming, setConfirming] = useState(false);
   109→  const [forcePrint, setForcePrint] = useState(false);
   110→  const [copies, setCopies] = useState(1);
   111→  const printedRef = useRef(false);
   112→
   113→  const isImg = isImageFile(order?.fileName);
   114→  const isPdf = isPdfFile(order?.fileName);
   115→
   116→  // ===== جلب الملف للمعاينة =====
   117→  const loadFile = useCallback(async () => {
   118→    if (!order) return;
   119→    setFileLoading(true);
   120→    try {
   121→      const res = await fetch(`/api/orders/${order.id}/file?shopId=${shopId}`);
   122→      if (!res.ok) throw new Error("فشل تحميل الملف");
   123→      const blob = await res.blob();
   124→      if (fileUrl) URL.revokeObjectURL(fileUrl);
   125→      setFileUrl(URL.createObjectURL(blob));
   126→    } catch (e) {
   127→      toast.error("تعذّر تحميل الملف للمعاينة", { description: (e as Error).message });
   128→    } finally {
   129→      setFileLoading(false);
   130→    }
   131→  }, [order, shopId, fileUrl]);
   132→
   133→  // ===== التحقق الذكي =====
   134→  const runVerify = useCallback(async () => {
   135→    if (!order) return;
   136→    setVerifyLoading(true);
   137→    setVerify(null);
   138→    try {
   139→      const res = await fetch(`/api/orders/${order.id}/verify-print?shopId=${shopId}`);
   140→      const data = (await res.json()) as VerifyResult;
   141→      if (!res.ok && !data.status) {
   142→        throw new Error(data.error || "فشل التحقق");
   143→      }
   144→      setVerify(data);
   145→    } catch (e) {
   146→      setVerify({
   147→        success: false,
   148→        canPrint: true,
   149→        confidence: 0,
   150→        status: "no_preview",
   151→        summary: "تعذّر التحقق التلقائي — راجع يدوياً",
   152→        alerts: [],
   153→        warnings: ["فشل تشغيل التحقق الذكي، يرجى المراجعة اليدوية"],
   154→        checks: [],
   155→        error: (e as Error).message,
   156→      });
   157→    } finally {
   158→      setVerifyLoading(false);
   159→    }
   160→  }, [order, shopId]);
   161→
   162→  useEffect(() => {
   163→    if (open && order) {
   164→      setVerify(null);
   165→      setForcePrint(false);
   166→      setCopies(order.copies || 1);
   167→      printedRef.current = false;
   168→      loadFile();
   169→      runVerify();
   170→    }
   171→    return () => {
   172→      if (fileUrl) URL.revokeObjectURL(fileUrl);
   173→    };
   174→  }, [open, order?.id]);
   175→
   176→  // ===== تأكيد الطباعة =====
   177→  async function handleConfirmPrint() {
   178→    if (!order || printedRef.current) return;
   179→    printedRef.current = true;
   180→    setConfirming(true);
   181→    try {
   182→      // 1) تحديث الحالة إلى printing
   183→      const printRes = await fetch(`/api/orders/${order.id}?shopId=${shopId}`, {
   184→        method: "PUT",
   185→        headers: { "Content-Type": "application/json" },
   186→        body: JSON.stringify({ status: "printing" }),
   187→      });
   188→      if (printRes.ok) {
   189→        onPrintStart();
   190→      }
   191→
   192→      toast.success("بدأت الطباعة", {
   193→        description: `${order.reference} — ${copies} نسخة`,
   194→      });
   195→
   196→      // 2) تشغيل الطباعة (نافذة الطباعة تعرض الملف + بيانات الطلب)
   197→      await new Promise((r) => setTimeout(r, 300));
   198→      window.print();
   199→
   200→      // 3) بعد إغلاق نافذة الطباعة → ready
   201→      const readyRes = await fetch(`/api/orders/${order.id}?shopId=${shopId}`, {
   202→        method: "PUT",
   203→        headers: { "Content-Type": "application/json" },
   204→        body: JSON.stringify({ status: "ready" }),
   205→      });
   206→      if (readyRes.ok) {
   207→        toast.success("تمت الطباعة", {
   208→          description: `${order.reference} — جاهز للاستلام`,
   209→        });
   210→        onPrintComplete();
   211→      }
   212→      onClose();
   213→    } catch (e) {
   214→      toast.error("خطأ في الطباعة", { description: (e as Error).message });
   215→      printedRef.current = false;
   216→    } finally {
   217→      setConfirming(false);
   218→    }
   219→  }
   220→
   221→  async function handleDownload() {
   222→    if (!order) return;
   223→    try {
   224→      const res = await fetch(`/api/orders/${order.id}/file?shopId=${shopId}`);
   225→      if (!res.ok) throw new Error("فشل التنزيل");
   226→      const blob = await res.blob();
   227→      const url = URL.createObjectURL(blob);
   228→      const a = document.createElement("a");
   229→      a.href = url;
   230→      a.download = order.fileName || `order-${order.reference}`;
   231→      document.body.appendChild(a);
   232→      a.click();
   233→      document.body.removeChild(a);
   234→      URL.revokeObjectURL(url);
   235→      toast.success("تم تنزيل الملف");
   236→    } catch (e) {
   237→      toast.error("تعذّر التنزيل", { description: (e as Error).message });
   238→    }
   239→  }
   240→
   241→  function handleOpenChange(isOpen: boolean) {
   242→    if (!isOpen) {
   243→      if (fileUrl) {
   244→        URL.revokeObjectURL(fileUrl);
   245→        setFileUrl(null);
   246→      }
   247→      onClose();
   248→    }
   249→  }
   250→
   251→  if (!order) return null;
   252→
   253→  const meta = STATUS_META[order.status];
   254→  const serviceEmoji = SERVICE_EMOJI[order.serviceType] || "🖨️";
   255→
   256→  // عدد النسخ الكلي للطباعة
   257→  const totalSheets = (order.pages || 1) * copies;
   258→
   259→  // الحالة العامة للتحقق
   260→  const verifyStatus = verify?.status;
   261→  const canPrintDirectly = verify?.canPrint === true || forcePrint;
   262→
   263→  const statusConfig: Record<string, { icon: typeof CheckCircle2; color: string; bg: string; label: string }> = {
   264→    match: { icon: CheckCircle2, color: "text-emerald-600", bg: "bg-emerald-50 border-emerald-200", label: "مطابق للمتطلبات" },
   265→    warning: { icon: AlertTriangle, color: "text-amber-600", bg: "bg-amber-50 border-amber-200", label: "تحذيرات — راجع قبل الطباعة" },
   266→    mismatch: { icon: XCircle, color: "text-rose-600", bg: "bg-rose-50 border-rose-200", label: "اختلاف — يمنع الطباعة" },
   267→    no_file: { icon: Info, color: "text-blue-600", bg: "bg-blue-50 border-blue-200", label: "لا يوجد ملف" },
   268→    no_preview: { icon: Info, color: "text-slate-600", bg: "bg-slate-50 border-slate-200", label: "مراجعة يدوية" },
   269→  };
   270→  const sCfg = verifyStatus ? statusConfig[verifyStatus] : null;
   271→
   272→  return (
   273→    <Dialog open={open} onOpenChange={handleOpenChange}>
   274→      <DialogContent
   275→        className="flex flex-col gap-0 p-0 rounded-none border-0 inset-0 top-0 left-0 translate-x-0 translate-y-0 h-[100dvh] w-full max-w-none sm:top-[50%] sm:left-[50%] sm:translate-x-[-50%] sm:translate-y-[-50%] sm:max-w-3xl sm:max-h-[92vh] sm:rounded-xl sm:border sm:h-auto overflow-hidden bg-background"
   276→        dir="rtl"
   277→        showCloseButton={false}
   278→        aria-describedby={undefined}
   279→      >
   280→        {/* ===== Header لاصق ===== */}
   281→        <div className="sticky top-0 z-30 flex items-center justify-between gap-2 px-3 sm:px-5 py-2.5 sm:py-3 border-b bg-background/95 backdrop-blur-sm">
   282→          <div className="min-w-0 flex-1 flex items-center gap-2">
   283→            <div className="w-9 h-9 sm:w-10 sm:h-10 rounded-xl bg-gradient-to-br from-amber-400 to-amber-600 flex items-center justify-center shrink-0 shadow-sm">
   284→              <Printer className="h-4 w-4 sm:h-5 sm:w-5 text-white" />
   285→            </div>
   286→            <div className="min-w-0">
   287→              <DialogTitle className="flex items-center gap-1.5 text-sm sm:text-base font-bold truncate">
   288→                <span>{serviceEmoji}</span>
   289→                <span className="font-mono">{order.reference}</span>
   290→              </DialogTitle>
   291→              <DialogDescription className="text-[11px] sm:text-xs truncate">
   292→                معاينة الطباعة المباشرة — {order.serviceName}
   293→              </DialogDescription>
   294→            </div>
   295→          </div>
   296→          <Button
   297→            variant="ghost"
   298→            size="icon"
   299→            className="h-9 w-9 shrink-0 rounded-lg hover:bg-muted"
   300→            onClick={onClose}
   301→          >
   302→            <X className="h-4 w-4" />
   303→          </Button>
   304→        </div>
   305→
   306→        {/* ===== المحتوى القابل للتمرير ===== */}
   307→        <div className="flex-1 overflow-y-auto custom-scroll px-3 sm:px-5 py-3 sm:py-4 space-y-3 sm:space-y-4">
   308→          {/* ===== قسم المعاينة ===== */}
   309→          <section className="rounded-xl border bg-card overflow-hidden">
   310→            <div className="flex items-center justify-between gap-2 px-3 sm:px-4 py-2.5 border-b bg-muted/30">
   311→              <h3 className="text-xs sm:text-sm font-bold flex items-center gap-2">
   312→                <Eye className="h-4 w-4 text-amber-500" />
   313→                معاينة الملف
   314→              </h3>
   315→              <div className="flex items-center gap-1.5">
   316→                <Button
   317→                  size="sm"
   318→                  variant="outline"
   319→                  className="h-7 text-[11px] px-2 rounded-lg"
   320→                  onClick={handleDownload}
   321→                >
   322→                  <Download className="h-3 w-3 ml-1" />
   323→                  تنزيل
   324→                </Button>
   325→              </div>
   326→            </div>
   327→            <div className="p-3 sm:p-4 bg-muted/20">
   328→              {fileLoading ? (
   329→                <div className="flex flex-col items-center justify-center py-10 text-muted-foreground">
   330→                  <Loader2 className="h-6 w-6 animate-spin text-amber-500 mb-2" />
   331→                  <span className="text-xs">جارٍ تحميل الملف...</span>
   332→                </div>
   333→              ) : fileUrl ? (
   334→                isImg ? (
   335→                  <div className="flex justify-center">
   336→                    <img
   337→                      src={fileUrl}
   338→                      alt={order.fileName || "معاينة"}
   339→                      className="max-w-full max-h-[40vh] sm:max-h-[45vh] object-contain rounded-lg border shadow-sm"
   340→                    />
   341→                  </div>
   342→                ) : isPdf ? (
   343→                  <iframe
   344→                    src={fileUrl}
   345→                    title="معاينة PDF"
   346→                    className="w-full h-[40vh] sm:h-[45vh] rounded-lg border bg-white"
   347→                  />
   348→                ) : (
   349→                  <div className="flex flex-col items-center justify-center py-10 text-center">
   350→                    <FileText className="h-10 w-10 text-muted-foreground/50 mb-2" />
   351→                    <p className="text-xs font-medium mb-1">{order.fileName}</p>
   352→                    <p className="text-[11px] text-muted-foreground">
   353→                      لا يمكن عرض معاينة لهذا النوع — استخدم التنزيل
   354→                    </p>
   355→                  </div>
   356→                )
   357→              ) : (
   358→                <div className="flex flex-col items-center justify-center py-10 text-muted-foreground">
   359→                  <FileText className="h-10 w-10 text-muted-foreground/50 mb-2" />
   360→                  <span className="text-xs">تعذّر تحميل المعاينة</span>
   361→                </div>
   362→              )}
   363→            </div>
   364→            {/* معلومات الملف */}
   365→            {order.fileName && (
   366→              <div className="px-3 sm:px-4 py-2 border-t bg-muted/20 flex items-center gap-2 flex-wrap text-[11px]">
   367→                <FileText className="h-3.5 w-3.5 text-amber-500 shrink-0" />
   368→                <span className="font-medium truncate flex-1 min-w-0">{order.fileName}</span>
   369→                <Badge variant="outline" className="text-[10px] h-5">
   370→                  {order.fileType || "—"}
   371→                </Badge>
   372→                {order.fileSize ? (
   373→                  <Badge variant="outline" className="text-[10px] h-5">
   374→                    {Math.round(order.fileSize / 1024)} ك.ب
   375→                  </Badge>
   376→                ) : null}
   377→              </div>
   378→            )}
   379→          </section>
   380→
   381→          {/* ===== التحقق الذكي بالـ AI ===== */}
   382→          <section className="rounded-xl border bg-card overflow-hidden">
   383→            <div className="flex items-center justify-between gap-2 px-3 sm:px-4 py-2.5 border-b bg-gradient-to-l from-violet-50/50 to-transparent">
   384→              <h3 className="text-xs sm:text-sm font-bold flex items-center gap-2">
   385→                <Sparkles className="h-4 w-4 text-violet-500" />
   386→                التحقق الذكي قبل الطباعة
   387→              </h3>
   388→              <Button
   389→                size="sm"
   390→                variant="ghost"
   391→                className="h-7 text-[11px] px-2 rounded-lg"
   392→                onClick={runVerify}
   393→                disabled={verifyLoading}
   394→              >
   395→                <RefreshCw className={cn("h-3 w-3 ml-1", verifyLoading && "animate-spin")} />
   396→                إعادة
   397→              </Button>
   398→            </div>
   399→            <div className="p-3 sm:p-4">
   400→              {verifyLoading ? (
   401→                <div className="flex flex-col items-center justify-center py-6">
   402→                  <Loader2 className="h-6 w-6 animate-spin text-violet-500 mb-2" />
   403→                  <p className="text-xs text-muted-foreground">جارٍ تحليل الملف ومقارنته بمتطلبات العميل...</p>
   404→                  <p className="text-[10px] text-muted-foreground/70 mt-1">قد يستغرق هذا بضع ثوانٍ</p>
   405→                </div>
   406→              ) : verify && sCfg ? (
   407→                <div className="space-y-3">
   408→                  {/* الحالة العامة */}
   409→                  <div className={cn("flex items-start gap-2.5 rounded-lg border p-3", sCfg.bg)}>

   413→                      {verify.summary && (
   414→                        <p className="text-[11px] text-foreground/80 mt-0.5 leading-relaxed">{verify.summary}</p>
   415→                      )}
   416→                      {verify.confidence > 0 && (
   417→                        <div className="flex items-center gap-1.5 mt-1.5">
   418→                          <div className="flex-1 h-1.5 bg-muted rounded-full overflow-hidden max-w-[120px]">
   419→                            <div
   420→                              className={cn("h-full rounded-full", verify.confidence >= 70 ? "bg-emerald-500" : verify.confidence >= 40 ? "bg-amber-500" : "bg-rose-500")}
   421→                              style={{ width: `${verify.confidence}%` }}
   422→                            />
   423→                          </div>
   424→                          <span className="text-[10px] text-muted-foreground tabular-nums">{verify.confidence}%</span>
   425→                        </div>
   426→                      )}
   427→                    </div>
   428→                  </div>
   429→
   430→                  {/* تنبيهات حرجة */}
   431→                  {verify.alerts.length > 0 && (
   432→                    <div className="space-y-1.5">
   433→                      {verify.alerts.map((a, i) => (
   434→                        <div key={i} className="flex items-start gap-2 rounded-lg border border-rose-200 bg-rose-50 p-2.5">
   435→                          <AlertTriangle className="h-4 w-4 text-rose-600 shrink-0 mt-0.5" />
   436→                          <p className="text-[11px] text-rose-700 leading-relaxed flex-1">{a}</p>
   437→                        </div>
   438→                      ))}
   439→                    </div>
   440→                  )}
   441→
   442→                  {/* تحذيرات */}
   443→                  {verify.warnings.length > 0 && (
   444→                    <div className="space-y-1.5">
   445→                      {verify.warnings.map((w, i) => (
   446→                        <div key={i} className="flex items-start gap-2 rounded-lg border border-amber-200 bg-amber-50 p-2.5">
   447→                          <Info className="h-4 w-4 text-amber-600 shrink-0 mt-0.5" />
   448→                          <p className="text-[11px] text-amber-700 leading-relaxed flex-1">{w}</p>
   449→                        </div>
   450→                      ))}
   451→                    </div>
   452→                  )}
   453→
   454→                  {/* الفحوصات التفصيلية */}
   455→                  {verify.checks.length > 0 && (
   456→                    <div className="rounded-lg border bg-muted/20 p-2.5 space-y-1.5">

   459→                        <div key={i} className="flex items-start gap-2 text-[11px]">

   474→                  {/* متطلبات العميل */}

   483→                </div>

   487→                </div>

   489→            </div>
   490→          </section>
   491→
   492→          {/* ===== ملخص الطباعة ===== */}

   507→                    variant="outline"
   508→                    className="h-8 w-8 p-0 rounded-lg"
   509→                    onClick={() => setCopies((c) => Math.max(1, c - 1))}
   510→                    disabled={copies <= 1}
   511→                  >
   512→                    −
   513→                  </Button>
   514→                  <Input2
   515→                    value={copies}
   516→                    onChange={(v) => setCopies(Math.max(1, Number(v) || 1))}

   518→                  <Button
   519→                    size="sm"
   520→                    variant="outline"
   521→                    className="h-8 w-8 p-0 rounded-lg"
   522→                    onClick={() => setCopies((c) => c + 1)}
   523→                  >
   524→                    +
   525→                  </Button>
   526→                </div>
   527→              </div>
   528→
   529→              {/* الخيارات */}
   530→              <div className="grid grid-cols-2 sm:grid-cols-3 gap-2">
   531→                {Object.entries(order.options)
   532→                  .filter(
   533→                    ([k, v]) =>
   534→                      v !== undefined &&
   535→                      v !== null &&
   536→                      v !== "" &&
   537→                      !HIDDEN_OPTION_KEYS.includes(k),
   538→                  )
   539→                  .map(([k, v]) => (
   540→                    <div key={k} className="rounded-lg bg-muted/30 border px-2.5 py-1.5">
   541→                      <div className="text-[10px] text-muted-foreground">{translateOptionKey(k)}</div>

   545→              </div>
   546→

   551→                  {order.pages} صفحة × {copies} نسخة = {totalSheets} ورقة
   552→                </span>
   553→              </div>
   554→
   555→              {/* المجموع */}
   556→              <div className="flex items-center justify-between gap-2 rounded-lg bg-gradient-to-l from-amber-50 to-transparent border border-amber-200/60 px-3 py-2">
   557→                <span className="text-xs text-muted-foreground">المجموع</span>

   560→            </div>

   562→
   563→          {/* ===== معلومات الطلب السريعة ===== */}

   567→            </div>
   568→            <div className="p-3 sm:p-4 grid grid-cols-2 gap-2 text-[11px]">

   577→        </div>

   579→        {/* ===== Footer لاصق ===== */}
   580→        <div className="sticky bottom-0 z-30 border-t bg-background/95 backdrop-blur-sm px-3 sm:px-5 py-2.5 sm:py-3 print-hide">
   581→          {/* تحذير فرض الطباعة */}
   582→          {verify && !verify.canPrint && !forcePrint && (

   588→              </div>
   589→              <Button
   590→                size="sm"
   591→                variant="ghost"
   592→                className="h-7 text-[10px] px-2 text-rose-700 hover:bg-rose-100 shrink-0"
   593→                onClick={() => setForcePrint(true)}
   594→              >
   595→                تجاوز
   596→              </Button>
   597→            </div>
   598→          )}
   599→          <div className="flex items-center gap-2">

   601→              variant="outline"
   602→              className="h-10 sm:h-11 px-3 rounded-xl text-xs sm:text-sm shrink-0"
   603→              onClick={onClose}
   604→            >
   605→              <X className="h-4 w-4 ml-1" />
   606→              إلغاء
   607→            </Button>
   608→            <Button
   609→              className={cn(
   610→                "flex-1 h-10 sm:h-11 rounded-xl text-xs sm:text-sm font-semibold gap-2 transition-all duration-200 active:scale-[0.98]",
   611→                canPrintDirectly
   612→                  ? "bg-gradient-to-l from-amber-500 to-amber-600 hover:from-amber-600 hover:to-amber-700 text-white shadow-md shadow-amber-200/60"
   613→                  : "bg-muted text-muted-foreground cursor-not-allowed",
   614→              )}
   615→              onClick={handleConfirmPrint}
   616→              disabled={!canPrintDirectly || confirming}
   617→            >
   618→              {confirming ? (
   619→                <>
   620→                  <RefreshCw className="h-4 w-4 animate-spin" />
   621→                  جارٍ الطباعة...
   622→                </>
   623→              ) : (
   624→                <>
   625→                  <Zap className="h-4 w-4" />
   626→                  {verify && !verify.canPrint && !forcePrint
   627→                    ? "راجع التنبيهات أولاً"
   628→                    : forcePrint
   629→                      ? "طباعة على مسؤوليتي"
   630→                      : `تأكيد وطباعة (${copies} نسخة)`}
   631→                </>
   632→              )}
   633→            </Button>
   634→          </div>
   635→        </div>
   636→      </DialogContent>
   637→
   638→      {/* ===== منطقة الطباعة (تظهر فقط عند window.print) ===== */}
   639→      <PrintArea
   640→        order={order}
   641→        shopName={shopName}
   642→        shopPhone={shopPhone}
   643→        shopAddress={shopAddress}
   644→        copies={copies}
   645→        fileUrl={fileUrl}
   646→        isImg={isImg}
   647→        isPdf={isPdf}
   648→      />
   649→    </Dialog>
   650→  );
   651→}
