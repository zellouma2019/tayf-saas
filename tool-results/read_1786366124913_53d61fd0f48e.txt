    70→interface ShopData {
    71→  slug: string; name: string; phone: string | null;
    72→  ownerName: string | null; ownerPhone: string | null;
    73→  whatsapp: string | null; email: string | null; address: string | null;
    74→  country: string | null; isActive: boolean; plan: string;
    75→  primaryColor: string | null; adminPin: string;
    76→  themeId: number | null;
    77→  logoIcon: string | null;
    78→  trialDays: number | null; trialStartsAt: string | null;
    79→  features: Record<string, boolean>;
    80→  ownerNotes: string | null;
    81→  paymentInfo: string | null;
    82→}
    83→
    84→interface AdminShopManagementProps {
    85→  open: boolean;
    86→  onClose: () => void;
    87→  shop: { slug: string; name: string } | null;
    88→  onSaved: () => void;
    89→}
    90→
    91→export function AdminShopManagement({ open, onClose, shop, onSaved }: AdminShopManagementProps) {
    92→  const [data, setData] = useState<ShopData | null>(null);
    93→  const [loading, setLoading] = useState(false);
    94→  const [saving, setSaving] = useState(false);
    95→  const [tab, setTab] = useState("basic");
    96→  const [hasChanges, setHasChanges] = useState(false);
    97→  const [originalData, setOriginalData] = useState<ShopData | null>(null);
    98→
    99→  useEffect(() => {
   100→    if (!open || !shop) return;
   101→    let cancelled = false;
   102→    (async () => {
   103→      setLoading(true);
   104→      setTab("basic");
   105→      try {
   106→        const res = await fetch(`/api/admin/shops/${shop.slug}`);
   107→        if (res.ok && !cancelled) {
   108→          const d = await res.json();
   109→          const s = d.shop;
   110→          const parsed: ShopData = {
   111→            slug: s.slug, name: s.name, phone: s.phone,
   112→            ownerName: s.ownerName, ownerPhone: s.ownerPhone,
   113→            whatsapp: s.whatsapp, email: s.email, address: s.address,
   114→            country: s.country, isActive: s.isActive,
   115→            plan: s.plan || "free", primaryColor: s.primaryColor,
   116→            adminPin: s.adminPin, themeId: s.themeId || 1,
   117→            logoIcon: s.logoIcon || "Printer",
   118→            trialDays: s.trialDays, trialStartsAt: s.trialStartsAt,
   119→            features: s.features ? (typeof s.features === "string" ? JSON.parse(s.features) : s.features) : {},
   120→            ownerNotes: s.ownerNotes, paymentInfo: s.paymentInfo,
   121→          };
   122→          setData(parsed);
   123→          setOriginalData(JSON.parse(JSON.stringify(parsed)));
   124→          setHasChanges(false);
   125→        } else if (!cancelled) {
   126→          toast.error("فشل تحميل بيانات المتجر");
   127→          onClose();
   128→        }
   129→      } catch { if (!cancelled) { toast.error("خطأ في الاتصال"); onClose(); } }
   130→      finally { if (!cancelled) setLoading(false); }
   131→    })();
   132→    return () => { cancelled = true; };
   133→  }, [open, shop, onClose]);
   134→
   135→  function setField<K extends keyof ShopData>(key: K, val: ShopData[K]) {
   136→    setData((prev) => prev ? { ...prev, [key]: val } : prev);
   137→    setHasChanges(true);
   138→  }
   139→
   140→  function toggleFeature(key: string) {
   141→    setData((prev) => {
   142→      if (!prev) return prev;
   143→      return { ...prev, features: { ...prev.features, [key]: !prev.features[key] } };
   144→    });
   145→    setHasChanges(true);
   146→  }
   147→
   148→  async function handleSave() {
   149→    if (!data) return;
   150→    setSaving(true);
   151→    try {
   152→      const payload: Record<string, unknown> = {
   153→        name: data.name, phone: data.phone, whatsapp: data.whatsapp,
   154→        email: data.email, address: data.address, country: data.country,
   155→        ownerName: data.ownerName, ownerPhone: data.ownerPhone,
   156→        adminPin: data.adminPin, isActive: data.isActive, plan: data.plan,
   157→        primaryColor: data.primaryColor, themeId: data.themeId,
   158→        logoIcon: data.logoIcon, trialDays: data.trialDays,
   159→        features: data.features, ownerNotes: data.ownerNotes,
   160→        paymentInfo: data.paymentInfo,
   161→      };
   162→      const res = await fetch(`/api/admin/shops/${data.slug}`, {
   163→        method: "PUT",
   164→        headers: { "Content-Type": "application/json" },
   165→        body: JSON.stringify(payload),
   166→      });
   167→      if (res.ok) {
   168→        toast.success("تم حفظ إعدادات المتجر");
   169→        setOriginalData(JSON.parse(JSON.stringify(data)));
   170→        setHasChanges(false);
   171→        onSaved();
   172→      } else {
   173→        toast.error("فشل حفظ الإعدادات");
   174→      }
   175→    } catch { toast.error("خطأ في الاتصال"); }
   176→    finally { setSaving(false); }
   177→  }
   178→
   179→  const tabs = [
   180→    { key: "basic", label: "الأساسيات", icon: Store },
   181→    { key: "plan", label: "الخطة والتجربة", icon: CreditCard },
   182→    { key: "features", label: "ميزات المتجر", icon: Sparkles },
   183→    { key: "merchant", label: "لوحة التحكم", icon: LayoutDashboard },
   184→    { key: "appearance", label: "المظهر", icon: Palette },
   185→    { key: "notes", label: "ملاحظات", icon: FileText },
   186→  ];
   187→
   188→  // Trial status
   189→  const trialStatus = (() => {
   190→    if (!data?.trialDays || !data?.trialStartsAt) return null;
   191→    const start = new Date(data.trialStartsAt).getTime();
   192→    const end = start + data.trialDays * 86400000;
   193→    const now = Date.now();
   194→    const remaining = Math.max(0, Math.ceil((end - now) / 86400000));
   195→    const active = now < end;
   196→    return { active, remaining, total: data.trialDays, start: data.trialStartsAt, end: new Date(end).toISOString() };
   197→  })();
   198→
   199→  return (
   200→    <Dialog open={open} onOpenChange={(v) => { if (!v && hasChanges) { /* could warn */ } onClose(); }}>
   201→      <DialogContent className="max-w-3xl max-h-[90vh] overflow-hidden flex flex-col p-0" dir="rtl">
   202→        <div className="flex items-center justify-between px-6 py-4 border-b border-border/50 shrink-0">
   203→          <div>
   204→            <DialogTitle className="text-lg font-bold">إدارة المتجر</DialogTitle>
   205→            <DialogDescription className="text-xs mt-0.5">تحكم شامل بإعدادات وميزات المتجر</DialogDescription>
   206→          </div>
   207→          {hasChanges && (
   208→            <Badge variant="outline" className="text-amber-600 border-amber-300 bg-amber-50 dark:bg-amber-500/10 dark:border-amber-500/30 dark:text-amber-400">
   209→              تعديلات غير محفوظة
   210→            </Badge>
   211→          )}
   212→        </div>
   213→
   214→        {/* Tabs */}
   215→        <div className="flex gap-1 px-4 pt-3 pb-0 overflow-x-auto no-scrollbar shrink-0">
   216→          {tabs.map((t) => (
   217→            <button
   218→              key={t.key}
   219→              onClick={() => setTab(t.key)}
   220→              className={`flex items-center gap-1.5 px-3 py-2 rounded-lg text-xs font-medium whitespace-nowrap transition-all ${
   221→                tab === t.key
   222→                  ? "bg-amber-500/10 text-amber-600 dark:text-amber-400"
   223→                  : "text-muted-foreground hover:text-foreground hover:bg-muted/50"
   224→              }`}
   225→            >
   226→              <t.icon className="h-3.5 w-3.5" />
   227→              <span className="hidden sm:inline">{t.label}</span>
   228→            </button>
   229→          ))}
   230→        </div>
   231→
   232→        {/* Content */}
   233→        <div className="flex-1 overflow-y-auto px-6 py-4">
   234→          {loading ? (
   235→            <div className="space-y-4">
   236→              {[1,2,3].map((i) => <div key={i} className="h-10 rounded-lg bg-muted animate-pulse" />)}
   237→            </div>
   238→          ) : !data ? null : (
   239→            <>
   240→              {/* ===== BASIC TAB ===== */}
   241→              {tab === "basic" && (
   242→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   243→                  <div className="sm:col-span-2">
   244→                    <Label>اسم المتجر</Label>
   245→                    <Input value={data.name} onChange={(e) => setField("name", e.target.value)} className="mt-1" />
   246→                  </div>
   247→                  <div>
   248→                    <Label>المعرف (slug)</Label>
   249→                    <Input value={data.slug} disabled className="mt-1 bg-muted/50" />
   250→                    <p className="text-[10px] text-muted-foreground mt-1">لا يمكن تغييره بعد الإنشاء</p>
   251→                  </div>
   252→                  <div>
   253→                    <Label>هاتف المتجر</Label>
   254→                    <Input value={data.phone || ""} onChange={(e) => setField("phone", e.target.value || null)} className="mt-1" dir="ltr" />
   255→                  </div>
   256→                  <div>
   257→                    <Label>واتساب</Label>
   258→                    <Input value={data.whatsapp || ""} onChange={(e) => setField("whatsapp", e.target.value || null)} className="mt-1" dir="ltr" />
   259→                  </div>
   260→                  <div>
   261→                    <Label>البريد الإلكتروني</Label>
   262→                    <Input type="email" value={data.email || ""} onChange={(e) => setField("email", e.target.value || null)} className="mt-1" dir="ltr" />
   263→                  </div>
   264→                  <div className="sm:col-span-2">
   265→                    <Label>العنوان</Label>
   266→                    <Input value={data.address || ""} onChange={(e) => setField("address", e.target.value || null)} className="mt-1" />
   267→                  </div>
   268→                  <div>
   269→                    <Label>الدولة</Label>
   270→                    <Select value={data.country || "DZ"} onValueChange={(v) => setField("country", v)}>
   271→                      <SelectTrigger className="mt-1"><SelectValue /></SelectTrigger>
   272→                      <SelectContent>
   273→                        {ARAB_COUNTRIES.map((c) => <SelectItem key={c.code} value={c.code}>{c.name} ({c.code})</SelectItem>)}
   274→                      </SelectContent>
   275→                    </Select>
   276→                  </div>
   277→                  <div>
   278→                    <Label>اسم صاحب المتجر</Label>
   279→                    <Input value={data.ownerName || ""} onChange={(e) => setField("ownerName", e.target.value || null)} className="mt-1" />
   280→                  </div>
   281→                  <div>
   282→                    <Label>هاتف صاحب المتجر</Label>
   283→                    <Input value={data.ownerPhone || ""} onChange={(e) => setField("ownerPhone", e.target.value || null)} className="mt-1" dir="ltr" />
   284→                  </div>
   285→                  <div>
   286→                    <Label>رمز PIN الإدارة</Label>
   287→                    <Input value={data.adminPin} onChange={(e) => setField("adminPin", e.target.value)} className="mt-1" dir="ltr" />
   288→                  </div>
   289→                  <div className="sm:col-span-2 flex items-center justify-between p-3 rounded-lg border border-border/50">
   290→                    <div>
   291→                      <div className="font-medium text-sm">حالة المتجر</div>
   292→                      <div className="text-xs text-muted-foreground">تفعيل أو إيقاف المتجر</div>
   293→                    </div>
   294→                    <Switch checked={data.isActive} onCheckedChange={(v) => setField("isActive", v)} />
   295→                  </div>
   296→                </div>
   297→              )}
   298→
   299→              {/* ===== PLAN & TRIAL TAB ===== */}
   300→              {tab === "plan" && (
   301→                <div className="space-y-6">
   302→                  {/* Plan selection */}
   303→                  <div>
   304→                    <Label className="text-sm font-semibold">الخطة الحالية</Label>
   305→                    <div className="grid grid-cols-2 gap-3 mt-2">
   306→                      {[{ value: "free", label: "مجانية", desc: "ميزات أساسية محدودة", color: "border-neutral-300 dark:border-neutral-600" },
   307→                        { value: "paid", label: "مدفوعة", desc: "جميع الميزات متاحة", color: "border-amber-400" }
   308→                      ].map((p) => (
   309→                        <button
   310→                          key={p.value}
   311→                          type="button"
   312→                          onClick={() => setField("plan", p.value)}
   313→                          className={`p-4 rounded-xl border-2 text-right transition-all ${
   314→                            data.plan === p.value ? p.color + " bg-amber-50/50 dark:bg-amber-500/5" : "border-border/50 hover:border-border"
   315→                          }`}
   316→                        >
   317→                          <div className="font-bold text-sm">{p.label}</div>
   318→                          <div className="text-xs text-muted-foreground mt-1">{p.desc}</div>
   319→                          {data.plan === p.value && <CheckCircle2 className="h-4 w-4 text-amber-500 mt-2" />}
   320→                        </button>
   321→                      ))}
   322→                    </div>
   323→                  </div>
   324→
   325→                  <Separator />
   326→
   327→                  {/* Trial period */}
   328→                  <div>
   329→                    <Label className="text-sm font-semibold">فترة التجربة</Label>
   330→                    <p className="text-xs text-muted-foreground mt-1">تحديد مدة تجربة مجانية للتاجر. اتركه فارغاً لبلا حدود.</p>
   331→                    <div className="grid grid-cols-2 gap-3 mt-3">
   332→                      <div>
   333→                        <Label className="text-xs">مدة التجربة (أيام)</Label>
   334→                        <Input
   335→                          type="number" min="0" max="365"
   336→                          value={data.trialDays ?? ""}
   337→                          onChange={(e) => setField("trialDays", e.target.value ? Number(e.target.value) : null)}
   338→                          className="mt-1" dir="ltr" placeholder="مثلاً: 14"
   339→                        />
   340→                      </div>
   341→                      <div>
   342→                        <Label className="text-xs">تاريخ البداية</Label>
   343→                        <Input
   344→                          type="date"
   345→                          value={data.trialStartsAt ? new Date(data.trialStartsAt).toISOString().split("T")[0] : ""}
   346→                          onChange={(e) => setField("trialStartsAt", e.target.value || null)}
   347→                          className="mt-1" dir="ltr"
   348→                        />
   349→                      </div>
   350→                    </div>
   351→
   352→                    {/* Trial status card */}
   353→                    {trialStatus && (
   354→                      <div className={`mt-4 p-4 rounded-xl border ${
   355→                        trialStatus.active
   356→                          ? "bg-emerald-50 dark:bg-emerald-500/5 border-emerald-200 dark:border-emerald-500/20"
   357→                          : "bg-rose-50 dark:bg-rose-500/5 border-rose-200 dark:border-rose-500/20"
   358→                      }`}>
   359→                        <div className="flex items-center gap-2 mb-2">
   360→                          {trialStatus.active ? (
   361→                            <><Clock className="h-4 w-4 text-emerald-600" /><span className="font-semibold text-sm text-emerald-700 dark:text-emerald-400">التجربة سارية</span></>
   362→                          ) : (
   363→                            <><AlertTriangle className="h-4 w-4 text-rose-600" /><span className="font-semibold text-sm text-rose-700 dark:text-rose-400">انتهت فترة التجربة</span></>
   364→                          )}
   365→                        </div>
   366→                        <div className="text-xs text-muted-foreground space-y-1">
   367→                          <div>المدة الكلية: {trialStatus.total} يوم</div>
   368→                          {trialStatus.active ? (
   369→                            <div>المتبقي: <span className="font-bold text-foreground">{trialStatus.remaining} يوم</span></div>
   370→                          ) : (
   371→                            <div>انتهت منذ {Math.abs(trialStatus.remaining)} يوم</div>
   372→                          )}
   373→                          <div>تاريخ البداية: {new Date(trialStatus.start).toLocaleDateString("ar-DZ")}</div>
   374→                        </div>
   375→                        {!trialStatus.active && (
   376→                          <Button
   377→                            size="sm"
   378→                            variant="outline"
   379→                            className="mt-3 gap-1 text-xs"
   380→                            onClick={() => {
   381→                              setField("trialStartsAt", new Date().toISOString());
   382→                            }}
   383→                          >
   384→                            <RefreshCw className="h-3 w-3" /> إعادة تفعيل التجربة
   385→                          </Button>
   386→                        )}
   387→                      </div>
   388→                    )}
   389→                  </div>
   390→                </div>
   391→              )}
   392→
   393→              {/* ===== FEATURES TAB ===== */}
   394→              {tab === "features" && (
   395→                <div className="space-y-3">
   396→                  <div className="flex items-center gap-2 mb-4">
   397→                    <Info className="h-4 w-4 text-muted-foreground" />
   398→                    <p className="text-xs text-muted-foreground">تفعيل أو تعطيل ميزات المتجر التي تظهر للزبائن</p>
   399→                  </div>
   400→                  <div className="grid grid-cols-1 sm:grid-cols-2 gap-2">
   401→                    {SHOP_FEATURES.map((f) => {
   402→                      const enabled = !!data.features[f.key];
   403→                      return (
   404→                        <div
   405→                          key={f.key}
   406→                          className={`flex items-center justify-between p-3 rounded-xl border transition-all cursor-pointer ${
   407→                            enabled
   408→                              ? "border-amber-300/50 bg-amber-50/50 dark:border-amber-500/20 dark:bg-amber-500/5"
   409→                              : "border-border/50 hover:border-border"
