    86→// ===== Login Gate =====
    87→function InlineLoginGate({ onUnlock }: { onUnlock: () => void }) {
    88→  const [password, setPassword] = useState('');
    89→  const [error, setError] = useState(false);
    90→  const [loading, setLoading] = useState(false);
    91→
    92→  async function handleSubmit(e: React.FormEvent) {
    93→    e.preventDefault();
    94→    if (!password) return;
    95→    setLoading(true); setError(false);
    96→    try {
    97→      const res = await fetch('/api/super-admin/auth', {
    98→        method: 'POST',
    99→        headers: { 'Content-Type': 'application/json' },
   100→        body: JSON.stringify({ password }),
   101→      });
   102→      const data = await res.json();
   103→      if (data.success) { markAuthenticated(data.token, data.ts); onUnlock(); }
   104→      else setError(true);
   105→    } catch { setError(true); }
   106→    finally { setLoading(false); }
   107→  }
   108→
   109→  return (
   110→    <div className="min-h-screen flex items-center justify-center bg-background p-4" dir="rtl">
   111→      <Card className="max-w-sm w-full">
   112→        <CardContent className="p-6 space-y-4">
   113→          <div className="flex flex-col items-center gap-3">
   114→            <img src="/n.png" alt="طيف" className="w-16 h-16 rounded-2xl object-contain shadow-lg" />
   115→            <div className="text-center">
   116→              <h1 className="text-xl font-bold">لوحة تحكم المنصة</h1>
   117→              <p className="text-sm text-muted-foreground">أدخل كلمة المرور للدخول</p>
   118→            </div>
   119→          </div>
   120→          <form onSubmit={handleSubmit} className="space-y-3">
   121→            <div className="relative">
   122→              <Lock className="absolute right-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   123→              <Input type="password" placeholder="كلمة المرور" value={password} onChange={(e) => { setPassword(e.target.value); setError(false); }} className="pr-9" dir="ltr" />
   124→            </div>
   125→            {error && <p className="text-destructive text-xs text-center">كلمة المرور غير صحيحة</p>}
   126→            <Button type="submit" className="w-full bg-gradient-to-l from-amber-500 to-amber-600 text-white" disabled={loading}>
   127→              {loading ? 'جاري التحقق...' : 'دخول'}
   128→            </Button>
   129→          </form>
   130→          <div className="flex items-center justify-center gap-2 pt-2"><ThemeToggle /></div>
   131→        </CardContent>
   132→      </Card>
   133→    </div>
   134→  );
   135→}

   137→// ===== Create Shop Dialog =====
   138→function InlineCreateShop({ open, onClose, onCreated }: { open: boolean; onClose: () => void; onCreated: () => void }) {
   139→  const [name, setName] = useState('');
   140→  const [slug, setSlug] = useState('');
   141→  const [pin, setPin] = useState('');
   142→  const [ownerName, setOwnerName] = useState('');
   143→  const [ownerPhone, setOwnerPhone] = useState('');
   144→  const [country, setCountry] = useState('DZ');
   145→  const [submitting, setSubmitting] = useState(false);
   146→  const [created, setCreated] = useState<{ name: string; slug: string; pin: string } | null>(null);
   147→
   148→  function genSlug(t: string) { return t.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/^-|-$/g, ''); }
   149→  function handleName(v: string) { setName(v); if (!slug || slug === genSlug(name)) setSlug(genSlug(v)); }
   150→
   151→  async function handleSubmit(e: React.FormEvent) {
   152→    e.preventDefault();
   153→    if (!name || !slug || !pin) return;
   154→    setSubmitting(true);
   155→    try {
   156→      const res = await fetch('/api/shops', {
   157→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   158→        body: JSON.stringify({ name, slug, adminPin: pin, ownerName, ownerPhone, country }),
   159→      });
   160→      if (res.ok) { setCreated({ name, slug, pin }); onCreated(); }
   161→      else { const data = await res.json().catch(() => ({})); toast.error(data.error || 'فشل إنشاء المتجر'); }
   162→    } catch { toast.error('خطأ في الاتصال'); }
   163→    finally { setSubmitting(false); }
   164→  }
   165→
   166→  function handleClose() {
   167→    setName(''); setSlug(''); setPin(''); setOwnerName(''); setOwnerPhone('');
   168→    setCreated(null); onClose();
   169→  }
   170→
   171→  return (
   172→    <Dialog open={open} onOpenChange={(o) => !o && handleClose()}>
   173→      <DialogContent className="sm:max-w-md" dir="rtl">
   174→        <DialogTitle>{created ? 'تم إنشاء المتجر ✓' : 'إنشاء متجر جديد'}</DialogTitle>
   175→        {created ? (
   176→          <div className="space-y-4 py-4">
   177→            <div className="text-center space-y-2">
   178→              <p className="font-bold text-lg">{created.name}</p>
   179→              <div className="space-y-2 text-sm">
   180→                <div className="flex items-center justify-between p-2 rounded-lg bg-muted">
   181→                  <span className="text-muted-foreground">رابط المتجر:</span>
   182→                  <button onClick={() => robustCopy(`/s/${created.slug}`, 'تم نسخ الرابط', '')} className="font-mono text-xs text-primary hover:underline">/s/{created.slug}</button>
   183→                </div>
   184→                <div className="flex items-center justify-between p-2 rounded-lg bg-muted">
   185→                  <span className="text-muted-foreground">كلمة المرور:</span>
   186→                  <span className="font-mono text-xs font-bold">{created.pin}</span>
   187→                </div>
   188→              </div>
   189→            </div>
   190→            <Button onClick={handleClose} className="w-full">تم</Button>
   191→          </div>
   192→        ) : (
   193→          <form onSubmit={handleSubmit} className="space-y-4">
   194→            <div className="space-y-2"><Label>اسم المتجر *</Label><Input value={name} onChange={(e) => handleName(e.target.value)} placeholder="مثال: مطبعة النور" /></div>
   195→            <div className="space-y-2"><Label>المعرّف (slug) *</Label><Input value={slug} onChange={(e) => setSlug(e.target.value)} placeholder="matbaa-alnoor" dir="ltr" /></div>
   196→            <div className="space-y-2"><Label>كلمة مرور لوحة التحكم (PIN) *</Label><Input value={pin} onChange={(e) => setPin(e.target.value)} placeholder="1234" maxLength={10} dir="ltr" type="text" /></div>
   197→            <div className="grid grid-cols-2 gap-3">
   198→              <div className="space-y-2"><Label>اسم المالك</Label><Input value={ownerName} onChange={(e) => setOwnerName(e.target.value)} placeholder="اختياري" /></div>
   199→              <div className="space-y-2"><Label>هاتف المالك</Label><Input value={ownerPhone} onChange={(e) => setOwnerPhone(e.target.value)} placeholder="اختياري" dir="ltr" /></div>
   200→            </div>
   201→            <div className="space-y-2">
   202→              <Label>الدولة</Label>
   203→              <Select value={country} onValueChange={setCountry}>
   204→                <SelectTrigger><SelectValue /></SelectTrigger>
   205→                <SelectContent>{ARAB_COUNTRIES.map((c) => (<SelectItem key={c.code} value={c.code}>{c.flag} {c.name_ar}</SelectItem>))}</SelectContent>
   206→              </Select>
   207→            </div>
   208→            <div className="flex gap-2 pt-2">
   209→              <Button type="button" variant="outline" onClick={handleClose} className="flex-1">إلغاء</Button>
   210→              <Button type="submit" disabled={!name || !slug || !pin || submitting} className="flex-1 bg-gradient-to-l from-amber-500 to-amber-600 text-white">
   211→                {submitting ? 'جاري الإنشاء...' : 'إنشاء'}
   212→              </Button>
   213→            </div>
   214→          </form>
   215→        )}
   216→      </DialogContent>
   217→    </Dialog>
   218→  );
   219→}

   230→function InlineEditShop({ open, onClose, shop, onSaved }: {
   231→  open: boolean; onClose: () => void;
   232→  shop: ShopItem | null; onSaved: () => void;
   233→}) {
   234→  const [data, setData] = useState<EditShopData | null>(null);
   235→  const [loading, setLoading] = useState(false);
   236→  const [saving, setSaving] = useState(false);
   237→
   238→  // جلب بيانات المتجر الكاملة عند الفتح
   239→  useEffect(() => {
   240→    if (!open || !shop) return;
   241→    let cancelled = false;
   242→    (async () => {
   243→      setLoading(true);
   244→      try {
   245→        const res = await fetch(`/api/admin/shops/${shop.slug}`);
   246→        if (res.ok && !cancelled) {
   247→          const d = await res.json();
   248→          const s = d.shop;
   249→          setData({
   250→            slug: s.slug, name: s.name, phone: s.phone,
   251→            ownerName: s.ownerName, ownerPhone: s.ownerPhone,
   252→            whatsapp: s.whatsapp, email: s.email, address: s.address,
   253→            country: s.country, isActive: s.isActive,
   254→            plan: s.plan || 'free', primaryColor: s.primaryColor,
   255→            adminPin: s.adminPin,
   256→          });
   257→        } else if (!cancelled) {
   258→          toast.error('فشل تحميل بيانات المتجر');
   259→          onClose();
   260→        }
   261→      } catch { if (!cancelled) { toast.error('خطأ في الاتصال'); onClose(); } }
   262→      finally { if (!cancelled) setLoading(false); }
   263→    })();
   264→    return () => { cancelled = true; };
   265→  }, [open, shop, onClose]);
   266→
   267→  async function handleSave(e: React.FormEvent) {
   268→    e.preventDefault();
   269→    if (!data) return;
   270→    setSaving(true);
   271→    try {
   272→      const res = await fetch(`/api/admin/shops/${data.slug}`, {
   273→        method: 'PUT',
   274→        headers: { 'Content-Type': 'application/json' },
   275→        body: JSON.stringify({
   276→          name: data.name, phone: data.phone, ownerName: data.ownerName,
   277→          ownerPhone: data.ownerPhone, whatsapp: data.whatsapp,
   278→          email: data.email, address: data.address, country: data.country,
   279→          isActive: data.isActive, plan: data.plan,
   280→          primaryColor: data.primaryColor, adminPin: data.adminPin,
   281→        }),
   282→      });
   283→      if (res.ok) { toast.success('تم تحديث المتجر بنجاح'); onSaved(); onClose(); }
   284→      else { const d = await res.json().catch(() => ({})); toast.error(d.error || 'فشل التحديث'); }
   285→    } catch { toast.error('خطأ في الاتصال'); }
   286→    finally { setSaving(false); }
   287→  }
   288→

   292→
   293→  if (!open) return null;
   294→
   295→  return (

   455→// ===== Mini Bar Chart (CSS-based, no Recharts) =====
   456→function MiniBarChart({ data, maxValue, color = 'bg-primary' }: { data: number[]; maxValue?: number; color?: string }) {
   457→  const max = maxValue || Math.max(...data, 1);
   458→  return (
   459→    <div className="flex items-end gap-1 h-12">
   460→      {data.map((v, i) => (
   461→        <div key={i} className="flex-1 flex flex-col items-center gap-0.5">
   462→          <div className={`${color} rounded-t-sm w-full transition-all duration-500`} style={{ height: `${Math.max((v / max) * 48, 2)}px` }} />
   463→        </div>
   464→      ))}
   465→    </div>
   466→  );
   467→}
