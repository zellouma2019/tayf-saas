    89→  themeId: number | null;
    90→  logoIcon: string | null;

    94→  paymentInfo: string | null;
    95→}

    97→interface AdminShopManagementProps {
    98→  open: boolean;
    99→  onClose: () => void;
   100→  shop: { slug: string; name: string } | null;
   101→  onSaved: () => void;
   102→}
   103→
   104→export function AdminShopManagement({ open, onClose, shop, onSaved }: AdminShopManagementProps) {
   105→  const [data, setData] = useState<ShopData | null>(null);
   106→  const [loading, setLoading] = useState(false);
   107→  const [saving, setSaving] = useState(false);
   108→  const [tab, setTab] = useState("basic");
   109→  const [hasChanges, setHasChanges] = useState(false);
   110→  const [originalData, setOriginalData] = useState<ShopData | null>(null);
   111→
   112→  useEffect(() => {
   113→    if (!open || !shop) return;
   114→    let cancelled = false;
   115→    (async () => {
   116→      setLoading(true);
   117→      setTab("basic");
   118→      try {
   119→        const res = await fetch(`/api/admin/shops/${shop.slug}`);
   120→        if (res.ok && !cancelled) {
   121→          const d = await res.json();
   122→          const s = d.shop;
   123→          const parsed: ShopData = {
   124→            slug: s.slug, name: s.name, phone: s.phone,
   125→            ownerName: s.ownerName, ownerPhone: s.ownerPhone,
   126→            whatsapp: s.whatsapp, email: s.email, address: s.address,
   127→            country: s.country, language: s.language || "ar",
   128→            customCurrency: s.customCurrency || null,
   129→            isActive: s.isActive,
   130→            plan: s.plan || "free", primaryColor: s.primaryColor,
   131→            adminPin: s.adminPin, themeId: s.themeId || 1,
   132→            logoIcon: s.logoIcon || "Printer",
   133→            trialDays: s.trialDays, trialStartsAt: s.trialStartsAt,
   134→            features: s.features ? (typeof s.features === "string" ? JSON.parse(s.features) : s.features) : {},
   135→            ownerNotes: s.ownerNotes, paymentInfo: s.paymentInfo,
   136→          };
   137→          setData(parsed);
   138→          setOriginalData(JSON.parse(JSON.stringify(parsed)));
   139→          setHasChanges(false);
   140→        } else if (!cancelled) {
   141→          toast.error("فشل تحميل بيانات المتجر");
   142→          onClose();
   143→        }
   144→      } catch { if (!cancelled) { toast.error("خطأ في الاتصال"); onClose(); } }
   145→      finally { if (!cancelled) setLoading(false); }
   146→    })();
   147→    return () => { cancelled = true; };
   148→  }, [open, shop, onClose]);
   149→
   150→  function setField<K extends keyof ShopData>(key: K, val: ShopData[K]) {
   151→    setData((prev) => prev ? { ...prev, [key]: val } : prev);
   152→    setHasChanges(true);
   153→  }
   154→
   155→  function toggleFeature(key: string) {
   156→    setData((prev) => {
   157→      if (!prev) return prev;
   158→      return { ...prev, features: { ...prev.features, [key]: !prev.features[key] } };
   159→    });
   160→    setHasChanges(true);
   161→  }
   162→
   163→  async function handleSave() {
   164→    if (!data) return;
   165→    setSaving(true);
   166→    try {
   167→      const payload: Record<string, unknown> = {
   168→        name: data.name, phone: data.phone, whatsapp: data.whatsapp,
   169→        email: data.email, address: data.address, country: data.country,
   170→        language: data.language, customCurrency: data.customCurrency,
   171→        ownerName: data.ownerName, ownerPhone: data.ownerPhone,
   172→        adminPin: data.adminPin, isActive: data.isActive, plan: data.plan,
   173→        primaryColor: data.primaryColor, themeId: data.themeId,
   174→        logoIcon: data.logoIcon, trialDays: data.trialDays,
   175→        features: data.features, ownerNotes: data.ownerNotes,
   176→        paymentInfo: data.paymentInfo,
   177→      };
   178→      const res = await fetch(`/api/admin/shops/${data.slug}`, {
   179→        method: "PUT",
   180→        headers: { "Content-Type": "application/json" },
   181→        body: JSON.stringify(payload),
   182→      });
   183→      if (res.ok) {
   184→        toast.success("تم حفظ إعدادات المتجر");
   185→        setOriginalData(JSON.parse(JSON.stringify(data)));
   186→        setHasChanges(false);
   187→        onSaved();
   188→      } else {
   189→        toast.error("فشل حفظ الإعدادات");
   190→      }
   191→    } catch { toast.error("خطأ في الاتصال"); }
   192→    finally { setSaving(false); }
   193→  }
   194→
   195→  const tabs = [

   219→          <div>
   220→            <DialogTitle className="text-lg font-bold">إدارة المتجر</DialogTitle>
   221→            <DialogDescription className="text-xs mt-0.5">تحكم شامل بإعدادات وميزات المتجر</DialogDescription>
