   487→    // التحقق من الصيغة
   488→    const ACCEPTED = [".pdf", ".docx", ".jpg", ".jpeg", ".png", ".webp"];
   489→    const ext = (f.name.split(".").pop() || "").toLowerCase();
   490→    if (!ACCEPTED.includes(`.${ext}`)) {
   491→      setAnalysisPhase("error");
   492→      setUploadError(`صيغة الملف ".${ext}" غير مدعومة. الصيغ المدعومة: ${ACCEPTED.join(", ")}`);
   493→      return;
   494→    }
   495→    // التحقق من الحجم
   496→    if (f.size > 50 * 1024 * 1024) {
   497→      setAnalysisPhase("error");
   498→      setUploadError(`حجم الملف ${(f.size / (1024 * 1024)).toFixed(1)} ميغابايت يتجاوز الحد الأقصى (50 ميغابايت)`);
   499→      return;
   500→    }
   501→    if (f.size === 0) {
   502→      setAnalysisPhase("error");
   503→      setUploadError("الملف فارغ — يرجى اختيار ملف آخر");
   504→      return;
   505→    }
   506→
   507→    setFileName(f.name);
   508→    setFileType(ext.toUpperCase());
   509→    setFileSize(f.size);
   510→    setFileDataUrl("");
   511→    setUploadError("");
   512→
   513→    // ─── المرحلة 1: رفع الملف ───
   514→    setAnalysisPhase("uploading");
   515→    setUploadStatus("uploading");
   516→    setUploadProgress(0);
   517→    setAnalysisTimings({ upload: null, local: null, ai: null, total: null });
   518→    const tUploadStart = performance.now();
   519→
   520→    try {
   521→      const CHUNK_THRESHOLD = 4.5 * 1024 * 1024; // 4.5 ميغابايت — أقل من حد البوابة 5MB
   522→      const CHUNK_SIZE = 4.5 * 1024 * 1024;
   523→      const MAX_CONCURRENT = 3; // رفع متوازي — 3 أجزاء في نفس الوقت
   524→
   525→      let storedFileName: string;
   526→
   527→      if (f.size <= CHUNK_THRESHOLD) {
   528→        // ملف صغير — رفع مباشر
   529→        const formData = new FormData();
   530→        formData.append("file", f);
   531→
   532→        storedFileName = await new Promise<string>((resolve, reject) => {
   533→          const xhr = new XMLHttpRequest();
   534→          xhr.open("POST", "/api/orders/upload");
   535→
   536→          xhr.upload.onprogress = (e) => {
   537→            if (e.lengthComputable) {
   538→              setUploadProgress(Math.round((e.loaded / e.total) * 100));
   539→            }
   540→          };
   541→
   542→          xhr.onload = () => {
   543→            if (xhr.status === 200) {
   544→              const data = JSON.parse(xhr.responseText);
   545→              resolve(data.storedFileName);
   546→            } else {
   547→              let msg = `فشل رفع الملف — رمز الخطأ: ${xhr.status}`;
   548→              try { const d = JSON.parse(xhr.responseText); if (d.error) msg = d.error; } catch {}
   549→              reject(new Error(msg));
   550→            }
   551→          };
   552→
   553→          xhr.onerror = () => reject(new Error("خطأ في الاتصال بالخادم. تحقق من اتصالك بالإنترنت."));
   554→          xhr.send(formData);
   555→        });
   556→      } else {
   557→        // ملف كبير — رفع مجزأ متوازي (لتجاوز حد البوابة)
   558→        const fileId = crypto.randomUUID();
   559→        const totalChunks = Math.ceil(f.size / CHUNK_SIZE);
   560→        let completedChunks = 0;
   561→        let storedFileNameFromChunk = "";
   562→
   563→        const uploadChunk = async (chunkIndex: number) => {
   564→          const start = chunkIndex * CHUNK_SIZE;
   565→          const end = Math.min(start + CHUNK_SIZE, f.size);
   566→          const blob = f.slice(start, end);
   567→
   568→          const formData = new FormData();
   569→          formData.append("chunk", blob, `chunk_${chunkIndex}`);
   570→          formData.append("fileId", fileId);
   571→          formData.append("chunkIndex", String(chunkIndex));
   572→          formData.append("totalChunks", String(totalChunks));
   573→          formData.append("fileName", f.name);
   574→          formData.append("fileSize", String(f.size));
   575→          formData.append("fileExt", ext);
   576→
   577→          return new Promise<{ complete: boolean; storedFileName?: string; error?: string }>((resolve, reject) => {
   578→            const xhr = new XMLHttpRequest();
   579→            xhr.open("POST", "/api/orders/upload-chunk");
   580→
   581→            xhr.onload = () => {
   582→              if (xhr.status === 200) {
   583→                resolve(JSON.parse(xhr.responseText));
   584→              } else {
   585→                let msg = `فشل رفع الجزء ${chunkIndex + 1}/${totalChunks} — رمز الخطأ: ${xhr.status}`;
   586→                try { const d = JSON.parse(xhr.responseText); if (d.error) msg = d.error; } catch {}
   587→                reject(new Error(msg));
   588→              }
   589→            };
   590→
   591→            xhr.onerror = () => reject(new Error(`خطأ في الاتصال أثناء رفع الجزء ${chunkIndex + 1}/${totalChunks}`));
   592→            xhr.send(formData);
   593→          });
   594→        };
   595→
   596→        // رفع متوازي مع حد أقصى
   597→        const chunkPromises: Promise<void>[] = [];
   598→        for (let i = 0; i < totalChunks; i++) {
   599→          const p = uploadChunk(i).then((result) => {
   600→            completedChunks++;
   601→            if (result.complete && result.storedFileName) {
   602→              storedFileNameFromChunk = result.storedFileName;
   603→            }
   604→            const pct = Math.round((completedChunks / totalChunks) * 100);
   605→            setUploadProgress(Math.min(pct, 99));
   606→          });
   607→          chunkPromises.push(p);
   608→          // انتظر إذا وصلنا للحد الأقصى من التوازي
   609→          if (chunkPromises.length >= MAX_CONCURRENT) {
   610→            await Promise.race(chunkPromises);
   611→            // إزالة الأجزاء المكتملة
   612→            const settled = await Promise.allSettled(chunkPromises.map((pr) => pr.then(() => true, () => false)));
   613→            const completed = settled.filter((s) => s.status === "fulfilled" && s.value).length;
   614→            // نحتفظ فقط بالأجزاء التي لم تكتمل بعد
   615→            const remaining: Promise<void>[] = [];
   616→            for (let j = 0; j < chunkPromises.length; j++) {
   617→              if (j >= completed) remaining.push(chunkPromises[j]);
   618→            }
   619→            chunkPromises.length = 0;
   620→            chunkPromises.push(...remaining);
   621→          }
   622→        }
   623→        // انتظر جميع الأجزاء المتبقية
   624→        await Promise.all(chunkPromises);
   625→
   626→        if (!storedFileNameFromChunk) {
   627→          throw new Error("فشل في تجميع الملف. حاول مرة أخرى.");
   628→        }
   629→        storedFileName = storedFileNameFromChunk;
   630→      }
   631→
   632→      setFileDataUrl(storedFileName);
   633→      setUploadProgress(100);
   634→      setUploadStatus("done");
   635→      setAnalysisTimings((prev) => ({ ...prev, upload: Math.round(performance.now() - tUploadStart) }));
   636→    } catch (uploadErr) {
   637→      setAnalysisPhase("error");
   638→      setUploadError((uploadErr as Error).message || "فشل رفع الملف");
   639→      setUploadStatus("error");
   640→      return;
   641→    }
   642→
   643→    // ─── المرحلة 2: التحليل المحلي ───
   644→    setAnalyzing(true);
   645→    setAnalysis(null);
   646→    setAnalysisPhase("local-analysis");
   647→    const tLocalStart = performance.now();
   648→
   649→    try {
   650→      const basicResult = await analyzeFileReal(f);
   651→      setAnalysisTimings((prev) => ({ ...prev, local: Math.round(performance.now() - tLocalStart) }));
   652→
   653→      // عرض النتيجة الأساسية فوراً
   654→      setAnalysis(basicResult);
   655→      setTotalPages(basicResult.pageCount);
   656→      setPages(basicResult.pageCount);
   657→      setPrintRange("all");
   658→      setPageRange("");
   659→      setServiceType(basicResult.detectedService);
   660→
   661→      // تطبيق التوصيات الأساسية
   662→      const spec = dynamicSpecs[basicResult.detectedService] || STATIC_SERVICE_SPECS[basicResult.detectedService];
   663→      const defaults: Record<string, string> = {};
   664→      if (spec) {
   665→        spec.sections.forEach((section) => {
   666→          // تطبيق توصيات التحليل الأساسي (color, paperSize, paperType, binding, photoSize)
   667→          if (section.optionKey === "color" && basicResult.suggestedColor) {
   668→            const valid = section.options.find((o) => o.id === basicResult.suggestedColor);
   669→            if (valid) { defaults.color = basicResult.suggestedColor; return; }
   670→          } else if (section.optionKey === "paperSize" && basicResult.suggestedPaperSize) {
   671→            const valid = section.options.find((o) => o.id === basicResult.suggestedPaperSize);
   672→            if (valid) { defaults.paperSize = basicResult.suggestedPaperSize; return; }
   673→          } else if (section.optionKey === "paperType" && basicResult.suggestedPaperType) {
   674→            const valid = section.options.find((o) => o.id === basicResult.suggestedPaperType);
   675→            if (valid) { defaults.paperType = basicResult.suggestedPaperType; return; }
   676→          } else if (section.optionKey === "binding" && basicResult.suggestedBinding) {
   677→            const valid = section.options.find((o) => o.id === basicResult.suggestedBinding);
   678→            if (valid) { defaults.binding = basicResult.suggestedBinding; return; }
   679→          } else if (section.optionKey === "photoSize" && basicResult.suggestedPhotoSize) {
   680→            // خدمة "photo" تستخدم photoSize لا paperSize — كانت مفقودة سابقاً
   681→            const valid = section.options.find((o) => o.id === basicResult.suggestedPhotoSize);
   682→            if (valid) { defaults.photoSize = basicResult.suggestedPhotoSize; return; }
   683→          }
   684→          // القيمة الافتراضية: أول خيار
   685→          if (section.options.length > 0) {
   686→            defaults[section.optionKey] = section.options[0].id;
   687→          }
   688→        });
   689→      }
   690→      setSpecOptions(defaults);
   691→
   692→      toast.success("اكتمل التحليل الأساسي", {
   693→        description: `${basicResult.detectedServiceName} · ${basicResult.pageCount} صفحة`,
   694→      });
   695→
   696→      // ─── المرحلة 3: التحليل الذكي بالـ VLM ───
   697→      setAnalysisPhase("ai-analysis");
   698→      const tAiStart = performance.now();
   699→
   700→      analyzeFileWithAI(f, basicResult).then(({ vlmAnalysis, enhancedAnalysis }) => {
   701→        setAnalysisTimings((prev) => ({ ...prev, ai: Math.round(performance.now() - tAiStart) }));
   702→        if (vlmAnalysis) {
   703→          setAnalysis(enhancedAnalysis);
   704→          setServiceType(enhancedAnalysis.detectedService as ServiceType);
   705→          setTotalPages(enhancedAnalysis.pageCount);
   706→          setPages(enhancedAnalysis.pageCount);
   707→
   708→          const updatedSpec = dynamicSpecs[enhancedAnalysis.detectedService] || STATIC_SERVICE_SPECS[enhancedAnalysis.detectedService as ServiceType];
   709→          const updatedDefaults: Record<string, string> = {};
   710→          const aiOptions = enhancedAnalysis.suggestedOptions || {};
   711→          if (updatedSpec) {
   712→            updatedSpec.sections.forEach((section) => {
   713→              // 1. تحقق أولاً من suggestedOptions المباشر (من VLM)
   714→              const aiValue = aiOptions[section.optionKey];
   715→              if (aiValue) {
   716→                // تحقق أن القيمة موجودة فعلاً في خيارات هذا القسم
   717→                const validOption = section.options.find((o) => o.id === aiValue);
   718→                if (validOption) {
   719→                  updatedDefaults[section.optionKey] = aiValue;
   720→                  return; // تم تطبيق اقتراح VLM
   721→                }
   722→              }
   723→              // 2. التوافق مع الحقول القديمة (suggestedColor, suggestedPaperSize, إلخ)
   724→              if (section.optionKey === "color" && enhancedAnalysis.suggestedColor) {
   725→                const valid = section.options.find((o) => o.id === enhancedAnalysis.suggestedColor);
   726→                if (valid) { updatedDefaults.color = enhancedAnalysis.suggestedColor; return; }
   727→              } else if (section.optionKey === "paperSize" && enhancedAnalysis.suggestedPaperSize) {
   728→                const valid = section.options.find((o) => o.id === enhancedAnalysis.suggestedPaperSize);
   729→                if (valid) { updatedDefaults.paperSize = enhancedAnalysis.suggestedPaperSize; return; }
   730→              } else if (section.optionKey === "paperType" && enhancedAnalysis.suggestedPaperType) {
   731→                const valid = section.options.find((o) => o.id === enhancedAnalysis.suggestedPaperType);
   732→                if (valid) { updatedDefaults.paperType = enhancedAnalysis.suggestedPaperType; return; }
   733→              } else if (section.optionKey === "binding" && enhancedAnalysis.suggestedBinding) {
   734→                const valid = section.options.find((o) => o.id === enhancedAnalysis.suggestedBinding);

   742→                updatedDefaults[section.optionKey] = section.options[0].id;
   743→              }
   744→            });
   745→          }
   746→          setSpecOptions(updatedDefaults);
   747→
   748→          toast.success("🤖 تم التحليل الذكي", {
   749→            description: `${vlmAnalysis.documentType} · ${vlmAnalysis.qualityAssessment} · دقة ${vlmAnalysis.confidence}%`,
   750→          });
   751→        }
   752→        setAnalysisPhase("done");
   753→        setAnalysisTimings((prev) => ({ ...prev, total: Math.round(performance.now() - tTotalStart) }));
   754→      }).catch(() => {
   755→        // VLM فشل — التحليل الأساسي كافٍ
   756→        setAnalysisPhase("done");
   757→        setAnalysisTimings((prev) => ({ ...prev, total: Math.round(performance.now() - tTotalStart) }));
   758→      });
   759→    } catch (err) {
   760→      setAnalysisPhase("error");
   761→      setUploadError((err as Error).message || "تعذّر تحليل الملف. تأكد أن الملف غير تالف.");
   762→      toast.error("تعذّر تحليل الملف", { description: (err as Error).message });
   763→    } finally {
   764→      setAnalyzing(false);
   765→    }
