   485→  async function processFile(f: File) {
   486→    const tTotalStart = performance.now();
   487→    // التحقق من الصيغة
   488→    const ACCEPTED = [".pdf", ".docx", ".jpg", ".jpeg", ".png", ".webp"];
   489→    const ext = (f.name.split(".").pop() || "").toLowerCase();
   490→    if (!ACCEPTED.includes(`.${ext}`)) {
   491→      setAnalysisPhase("error");
   492→      setUploadError(`صيغة الملف ".${ext}" غير مدعومة. الصيغ المدعومة: ${ACCEPTED.join(", ")}`);
   493→      return;
   494→    }
   495→    // التحقق من الحجم
   496→    if (f.size > 50 * 1024 * 1024) {
   497→      setAnalysisPhase("error");
   498→      setUploadError(`حجم الملف ${(f.size / (1024 * 1024)).toFixed(1)} ميغابايت يتجاوز الحد الأقصى (50 ميغابايت)`);
   499→      return;
   500→    }
   501→    if (f.size === 0) {
   502→      setAnalysisPhase("error");
   503→      setUploadError("الملف فارغ — يرجى اختيار ملف آخر");
   504→      return;
   505→    }
   506→
   507→    setFileName(f.name);
   508→    setFileType(ext.toUpperCase());
   509→    setFileSize(f.size);
   510→    setFileDataUrl("");
   511→    setUploadError("");
   512→
   513→    // ─── المرحلة 1: رفع الملف ───
   514→    setAnalysisPhase("uploading");
   515→    setUploadStatus("uploading");
   516→    setUploadProgress(0);
   517→    setAnalysisTimings({ upload: null, local: null, ai: null, total: null });
   518→    const tUploadStart = performance.now();
   519→
   520→    try {
   521→      const CHUNK_THRESHOLD = 900 * 1024; // 900 كيلوبايت
   522→      const CHUNK_SIZE = 900 * 1024;
   523→
   524→      let storedFileName: string;
   525→
   526→      if (f.size <= CHUNK_THRESHOLD) {
   527→        // ملف صغير — رفع مباشر
   528→        const formData = new FormData();
   529→        formData.append("file", f);
   530→
   531→        storedFileName = await new Promise<string>((resolve, reject) => {
   532→          const xhr = new XMLHttpRequest();
   533→          xhr.open("POST", "/api/orders/upload");
   534→
   535→          xhr.upload.onprogress = (e) => {
   536→            if (e.lengthComputable) {
   537→              setUploadProgress(Math.round((e.loaded / e.total) * 100));
   538→            }
   539→          };
   540→
   541→          xhr.onload = () => {
   542→            if (xhr.status === 200) {
   543→              const data = JSON.parse(xhr.responseText);
   544→              resolve(data.storedFileName);
   545→            } else {
   546→              let msg = `فشل رفع الملف — رمز الخطأ: ${xhr.status}`;
   547→              try { const d = JSON.parse(xhr.responseText); if (d.error) msg = d.error; } catch {}
   548→              reject(new Error(msg));
   549→            }
   550→          };
   551→
   552→          xhr.onerror = () => reject(new Error("خطأ في الاتصال بالخادم. تحقق من اتصالك بالإنترنت."));
   553→          xhr.send(formData);
   554→        });
   555→      } else {
   556→        // ملف كبير — رفع مجزأ (لتجاوز حد 1 ميغا في البوابة)
   557→        const fileId = crypto.randomUUID();
   558→        const totalChunks = Math.ceil(f.size / CHUNK_SIZE);
   559→        let overallLoaded = 0;
   560→
   561→        for (let i = 0; i < totalChunks; i++) {
   562→          const start = i * CHUNK_SIZE;
   563→          const end = Math.min(start + CHUNK_SIZE, f.size);
   564→          const blob = f.slice(start, end);
   565→
   566→          const formData = new FormData();
   567→          formData.append("chunk", blob, `chunk_${i}`);
   568→          formData.append("fileId", fileId);
   569→          formData.append("chunkIndex", String(i));
   570→          formData.append("totalChunks", String(totalChunks));
   571→          formData.append("fileName", f.name);
   572→          formData.append("fileSize", String(f.size));
   573→          formData.append("fileExt", ext);
   574→
   575→          const result = await new Promise<{ complete: boolean; storedFileName?: string; error?: string }>((resolve, reject) => {
   576→            const xhr = new XMLHttpRequest();
   577→            xhr.open("POST", "/api/orders/upload-chunk");
   578→
   579→            xhr.upload.onprogress = (e) => {
   580→              if (e.lengthComputable) {
   581→                const chunkPct = e.loaded / e.total;
   582→                const pct = Math.round(((overallLoaded + start + e.loaded) / f.size) * 100);
   583→                setUploadProgress(Math.min(pct, 99));
   584→              }
   585→            };
   586→
   587→            xhr.onload = () => {
   588→              if (xhr.status === 200) {
   589→                resolve(JSON.parse(xhr.responseText));
   590→              } else {
   591→                let msg = `فشل رفع الجزء ${i + 1}/${totalChunks} — رمز الخطأ: ${xhr.status}`;
   592→                try { const d = JSON.parse(xhr.responseText); if (d.error) msg = d.error; } catch {}
   593→                reject(new Error(msg));
   594→              }
   595→            };
   596→
   597→            xhr.onerror = () => reject(new Error(`خطأ في الاتصال أثناء رفع الجزء ${i + 1}/${totalChunks}`));
   598→            xhr.send(formData);
   599→          });
   600→
   601→          overallLoaded = end;
   602→
   603→          if (result.complete && result.storedFileName) {
   604→            storedFileName = result.storedFileName;
   605→          }
   606→        }
   607→        // إذا لم نحصل على storedFileName من آخر جزء
   608→        if (!storedFileName) {
   609→          throw new Error("فشل في تجميع الملف. حاول مرة أخرى.");
   610→        }
   611→      }
   612→
   613→      setFileDataUrl(storedFileName);
   614→      setUploadProgress(100);
   615→      setUploadStatus("done");
   616→      setAnalysisTimings((prev) => ({ ...prev, upload: Math.round(performance.now() - tUploadStart) }));
   617→    } catch (uploadErr) {
   618→      setAnalysisPhase("error");
   619→      setUploadError((uploadErr as Error).message || "فشل رفع الملف");
   620→      setUploadStatus("error");
   621→      return;
   622→    }
   623→
   624→    // ─── المرحلة 2: التحليل المحلي ───
   625→    setAnalyzing(true);
   626→    setAnalysis(null);
   627→    setAnalysisPhase("local-analysis");
   628→    const tLocalStart = performance.now();
   629→
   630→    try {
   631→      const basicResult = await analyzeFileReal(f);
   632→      setAnalysisTimings((prev) => ({ ...prev, local: Math.round(performance.now() - tLocalStart) }));
   633→
   634→      // عرض النتيجة الأساسية فوراً
   635→      setAnalysis(basicResult);
   636→      setTotalPages(basicResult.pageCount);
   637→      setPages(basicResult.pageCount);
   638→      setPrintRange("all");
   639→      setPageRange("");
   640→      setServiceType(basicResult.detectedService);
   641→
   642→      // تطبيق التوصيات الأساسية
   643→      const spec = dynamicSpecs[basicResult.detectedService] || STATIC_SERVICE_SPECS[basicResult.detectedService];
   644→      const defaults: Record<string, string> = {};
   645→      if (spec) {
   646→        spec.sections.forEach((section) => {
   647→          // تطبيق توصيات التحليل الأساسي (color, paperSize, paperType, binding, photoSize)
   648→          if (section.optionKey === "color" && basicResult.suggestedColor) {

   650→            if (valid) { defaults.color = basicResult.suggestedColor; return; }
   651→          } else if (section.optionKey === "paperSize" && basicResult.suggestedPaperSize) {
