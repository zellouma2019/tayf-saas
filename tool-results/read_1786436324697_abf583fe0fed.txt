    87→    87→function InlineLoginGate({ onUnlock }: { onUnlock: () => void }) {
    88→    88→  const [password, setPassword] = useState('');
    89→    89→  const [error, setError] = useState(false);
    90→    90→  const [loading, setLoading] = useState(false);
    91→    91→
    92→    92→  async function handleSubmit(e: React.FormEvent) {
    93→    93→    e.preventDefault();
    94→    94→    if (!password) return;
    95→    95→    setLoading(true); setError(false);
    96→    96→    try {
    97→    97→      const res = await fetch('/api/super-admin/auth', {
    98→    98→        method: 'POST',
    99→    99→        headers: { 'Content-Type': 'application/json' },
   100→   100→        body: JSON.stringify({ password }),
   101→   101→      });
   102→   102→      const data = await res.json();
   103→   103→      if (data.success) { markAuthenticated(data.token, data.ts); onUnlock(); }
   104→   104→      else setError(true);
   105→   105→    } catch { setError(true); }
   106→   106→    finally { setLoading(false); }
   107→   107→  }
   108→   108→
   109→   109→  return (
   110→   110→    <div className="min-h-screen flex items-center justify-center bg-background p-4" dir="rtl">

   115→   115→            <div className="text-center">
   116→   116→              <h1 className="text-2xl font-extrabold tracking-tight">لوحة تحكم المنصة</h1>

   123→   123→              <Input type="password" placeholder="كلمة المرور" value={password} onChange={(e) => { setPassword(e.target.value); setError(false); }} className="pr-9" dir="ltr" />
