   121→  const [auditLoading, setAuditLoading] = useState(false);

   151→      /* silent */
   152→    } finally {
   153→      setAuditLoading(false);

   160→      setAuditLogs([]);
   161→      setShowDeleteConfirm(false);

   181→      setEditTags(Array.isArray(order.tags) ? order.tags : JSON.parse(order.tags ? String(order.tags) : "[]"));

   189→  function handleOpenChange(isOpen: boolean) {
   190→    if (!isOpen) onClose();
   191→  }

   193→  async function handleSave() {
   194→    if (!order) return;
   195→    setSaving(true);
   196→    try {
   197→      const payload: Record<string, unknown> = { action: "edit" };
   198→
   199→      const customerUpdates: Record<string, string> = {};
   200→      if (editName !== (order.customer?.name || "")) customerUpdates.name = editName;
   201→      if (editPhone !== (order.customer?.phone || "")) customerUpdates.phone = editPhone;
   202→      if (editWhatsApp !== (order.customer?.whatsapp || "")) customerUpdates.whatsapp = editWhatsApp;
   203→      if (editEmail !== (order.customer?.email || "")) customerUpdates.email = editEmail;
   204→      if (editAddress !== (order.customer?.address || "")) customerUpdates.address = editAddress;
   205→      if (Object.keys(customerUpdates).length > 0) payload.customer = customerUpdates;
   206→
   207→      if (editAdminNotes !== (order.adminNotes || "")) payload.adminNotes = editAdminNotes;
   208→
   209→      const currentTags = Array.isArray(order.tags) ? order.tags : [];
   210→      if (JSON.stringify(editTags) !== JSON.stringify(currentTags)) {
   211→        payload.tags = JSON.stringify(editTags);
   212→      }
   213→
   214→      if (editCost !== (order.cost || 0)) payload.cost = editCost;
   215→      if (editCopies !== order.copies) payload.copies = editCopies;
   216→      if (editPages !== order.pages) payload.pages = editPages;
   217→
   218→      if (Object.keys(payload).length <= 1) {
   219→        toast.info("لا توجد تغييرات لحفظها");
   220→        setSaving(false);
   221→        return;
   222→      }
   223→
   224→      const res = await fetch(`/api/orders/${order.id}?shopId=${shopId}`, {
   225→        method: "PUT",
   226→        headers: { "Content-Type": "application/json" },
   227→        body: JSON.stringify(payload),
   228→      });
   229→      if (!res.ok) {
   230→        const err = await res.json();
   231→        throw new Error(err.error || "فشل التحديث");
   232→      }
   233→
   234→      toast.success("تم تحديث الطلب بنجاح");
   235→      onUpdated();
   236→      onClose();
   237→    } catch (e) {
   238→      toast.error("خطأ في التحديث", { description: (e as Error).message });
   239→    } finally {
   240→      setSaving(false);
   241→    }
   242→  }
   243→
   244→  async function handlePrintingAction(action: "start" | "complete") {
   245→    if (!order) return;
   246→    setPrintingAction(action);
   247→    try {
   248→      const status = action === "start" ? "printing" : "ready";
   249→      const res = await fetch(`/api/orders/${order.id}?shopId=${shopId}`, {
   250→        method: "PUT",
   251→        headers: { "Content-Type": "application/json" },
   252→        body: JSON.stringify({ status }),
   253→      });
   254→      if (!res.ok) {
   255→        const err = await res.json();
   256→        throw new Error(err.error || "فشل التحديث");
   257→      }
   258→
   259→      const actionLabel = action === "start" ? "بدأ الطباعة" : "انتهى الطباعة";
   260→      toast.success(actionLabel, {
   261→        description: `${order.reference} → ${STATUS_META[status].label}`,
   262→      });
   263→      onStatusChange(order, status);
   264→      onUpdated();
   265→    } catch (e) {
   266→      toast.error("خطأ", { description: (e as Error).message });
   267→    } finally {
   268→      setPrintingAction(null);
   269→    }
   270→  }

   272→  // ===== طباعة مباشرة — تفتح نافذة المعاينة والتحقق الذكي =====
   273→  async function handleDirectPrint() {

   278→  async function handleDelete() {
   279→    if (!order) return;

   281→    try {
   282→      const res = await fetch(`/api/orders/${order.id}?shopId=${shopId}`, {
   283→        method: "DELETE",
   284→      });

   286→        const err = await res.json();
   287→        throw new Error(err.error || "فشل الحذف");
   288→      }
   289→
   290→      toast.success("تم حذف الطلب", {
   291→        description: `${order.reference} تم حذفه نهائياً`,
   292→      });
   293→      onUpdated();
   294→      onClose();
   295→    } catch (e) {
   296→      toast.error("خطأ في الحذف", { description: (e as Error).message });
   297→    } finally {
   298→      setDeleting(false);
   299→      setShowDeleteConfirm(false);
   300→    }
   301→  }
   302→
   303→  function handleStatusChange(status: string) {

   308→  function togglePresetTag(tag: string) {
   309→    setEditTags((prev) =>
   310→      prev.includes(tag) ? prev.filter((t) => t !== tag) : [...prev, tag],
   311→    );
   312→  }

   314→  function addCustomTag() {
   315→    const trimmed = customTagInput.trim();
   316→    if (!trimmed) return;
   317→    if (editTags.includes(trimmed)) {
   318→      toast.info("الوسم موجود بالفعل");
   319→      return;
   320→    }
   321→    setEditTags((prev) => [...prev, trimmed]);
   322→    setCustomTagInput("");
   323→  }
   324→
   325→  function removeTag(tag: string) {
   326→    setEditTags((prev) => prev.filter((t) => t !== tag));
   327→  }

   329→  // تنزيل الملف — يستخدم دائماً الـ endpoint الذي يجلب fileData من DB
   330→  // (fileData غير مُضمَّن في قائمة الطلبات لتجنب 504)
   331→  async function downloadFile() {
   332→    if (!order || downloading) return;
   333→    setDownloading(true);
   334→    try {
   335→      const res = await fetch(`/api/orders/${order.id}/file?shopId=${shopId}`);
   336→      if (!res.ok) {
   337→        const err = await res.json().catch(() => ({}));
   338→        throw new Error(err.error || "فشل تنزيل الملف");
   339→      }
   340→      const blob = await res.blob();
   341→      const url = URL.createObjectURL(blob);
   342→      const a = document.createElement("a");
   343→      a.href = url;
   344→      a.download = order.fileName || `order-${order.reference}`;
   345→      document.body.appendChild(a);
   346→      a.click();
   347→      document.body.removeChild(a);
   348→      URL.revokeObjectURL(url);
   349→      toast.success("تم تنزيل الملف");
   350→    } catch (e) {
   351→      toast.error("تعذّر تنزيل الملف", { description: (e as Error).message });
   352→    } finally {
   353→      setDownloading(false);
   354→    }
   355→  }

   452→                  >
   453→                    <CheckCircle2 className="h-3.5 w-3.5 ml-1" />
