   104→   104→export function AdminShopManagement({ open, onClose, shop, onSaved }: AdminShopManagementProps) {
   105→   105→  const [data, setData] = useState<ShopData | null>(null);
   106→   106→  const [loading, setLoading] = useState(false);

   110→   110→  const [originalData, setOriginalData] = useState<ShopData | null>(null);
   111→   111→
   112→   112→  useEffect(() => {
   113→   113→    if (!open || !shop) return;
   114→   114→    let cancelled = false;
   115→   115→    (async () => {
   116→   116→      setLoading(true);
   117→   117→      setTab("basic");
   118→   118→      try {
   119→   119→        const res = await fetch(`/api/admin/shops/${shop.slug}`);
   120→   120→        if (res.ok && !cancelled) {
   121→   121→          const d = await res.json();
   122→   122→          const s = d.shop;
   123→   123→          const parsed: ShopData = {
   124→   124→            slug: s.slug, name: s.name, phone: s.phone,
   125→   125→            ownerName: s.ownerName, ownerPhone: s.ownerPhone,
   126→   126→            whatsapp: s.whatsapp, email: s.email, address: s.address,
   127→   127→            country: s.country, language: s.language || "ar",
   128→   128→            customCurrency: s.customCurrency || null,

   131→   131→            adminPin: s.adminPin, themeId: s.themeId || 1,
   132→   132→            logoIcon: s.logoIcon || "Printer",
   133→   133→            trialDays: s.trialDays, trialStartsAt: s.trialStartsAt,
   134→   134→            features: s.features ? (typeof s.features === "string" ? JSON.parse(s.features) : s.features) : {},
   135→   135→            ownerNotes: s.ownerNotes, paymentInfo: s.paymentInfo,
   136→   136→          };
   137→   137→          setData(parsed);
   138→   138→          setOriginalData(JSON.parse(JSON.stringify(parsed)));
   139→   139→          setHasChanges(false);
   140→   140→        } else if (!cancelled) {
   141→   141→          toast.error("فشل تحميل بيانات المتجر");
   142→   142→          onClose();
   143→   143→        }
   144→   144→      } catch { if (!cancelled) { toast.error("خطأ في الاتصال"); onClose(); } }
   145→   145→      finally { if (!cancelled) setLoading(false); }
   146→   146→    })();
   147→   147→    return () => { cancelled = true; };
   148→   148→  }, [open, shop, onClose]);

   150→   150→  function setField<K extends keyof ShopData>(key: K, val: ShopData[K]) {
   151→   151→    setData((prev) => {
   152→   152→      if (!prev) return prev;
   153→   153→      const next = { ...prev, [key]: val };
   154→   154→      // عند تغيير الدولة، حدّث العملة تلقائياً إن لم يكن هناك عملة مخصصة
   155→   155→      if (key === "country" && !prev.customCurrency) {
   156→   156→        const country = ARAB_COUNTRIES.find((c) => c.code === val);
   157→   157→        if (country) {
   158→   158→          next.customCurrency = country.currencyCode;
   159→   159→        }
   160→   160→      }
   161→   161→      // عند تغيير الخطة إلى مدفوعة، فعّل كل الميزات
   162→   162→      if (key === "plan" && val === "paid") {
   163→   163→        const allFeatures: Record<string, boolean> = {};
   164→   164→        SHOP_FEATURES.forEach((f) => { allFeatures[f.key] = true; });
   165→   165→        MERCHANT_ADMIN_TABS.forEach((t) => { allFeatures[t.key] = true; });
   166→   166→        MERCHANT_PERMISSIONS.forEach((p) => { allFeatures[p.key] = true; });
   167→   167→        next.features = allFeatures;
   168→   168→      }
   169→   169→      // عند تغيير الخطة إلى مجانية، أوقف الميزات المدفوعة فقط
   170→   170→      if (key === "plan" && val === "free") {
   171→   171→        const paidOnly = ["aiAssistant", "loyaltyProgram", "priceCalculator", "multiBranch", "couponSystem", "customerSupport"];
   172→   172→        const updated = { ...prev.features };
   173→   173→        paidOnly.forEach((k) => { updated[k] = false; });
   174→   174→        next.features = updated;
   175→   175→      }
   176→   176→      return next;
   177→   177→    });
   178→   178→    setHasChanges(true);
   179→   179→  }
   180→   180→
   181→   181→  function toggleFeature(key: string) {
   182→   182→    setData((prev) => {
   183→   183→      if (!prev) return prev;
   184→   184→      return { ...prev, features: { ...prev.features, [key]: !prev.features[key] } };
   185→   185→    });
   186→   186→    setHasChanges(true);
   187→   187→  }

   201→   201→        features: data.features, ownerNotes: data.ownerNotes,
   202→   202→        paymentInfo: data.paymentInfo,
   203→   203→      };

   205→   205→        method: "PUT",
   206→   206→        headers: { "Content-Type": "application/json" },
   207→   207→        body: JSON.stringify(payload),
   208→   208→      });

   210→   210→        toast.success("تم حفظ إعدادات المتجر");
   211→   211→        setOriginalData(JSON.parse(JSON.stringify(data)));
   212→   212→        setHasChanges(false);
   213→   213→        onSaved();

   215→   215→        toast.error("فشل حفظ الإعدادات");
   216→   216→      }

   280→   280→          ) : !data ? null : (

   282→   282→              {/* ===== BASIC TAB ===== */}
