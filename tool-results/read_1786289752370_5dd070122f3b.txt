   143→// ===== تلميح مخصص بالعربية =====
   144→function ArabicTooltip({
   145→  active,
   146→  payload,
   147→  label,
   148→  valueLabel,
   149→  countLabel,
   150→}: {
   151→  active?: boolean;
   152→  payload?: Array<{ value: number; name: string; color: string; dataKey: string }>;
   153→  label?: string;
   154→  valueLabel?: string;
   155→  countLabel?: string;
   156→}) {
   157→  if (!active || !payload?.length) return null;
   158→  return (
   159→    <div className="bg-white dark:bg-neutral-800 border border-border rounded-lg shadow-lg px-3 py-2 text-sm min-w-[140px]" dir="rtl">
   160→      <div className="font-bold text-neutral-800 dark:text-neutral-200 mb-1">{label}</div>
   161→      {payload.map((entry, i) => (
   162→        <div key={i} className="flex items-center justify-between gap-3">
   163→          <span className="flex items-center gap-1.5 text-xs text-muted-foreground">
   164→            <span className="w-2.5 h-2.5 rounded-sm inline-block" style={{ backgroundColor: entry.color }} />
   165→            {entry.dataKey === "revenue" ? valueLabel || "الإيرادات" : countLabel || "الطلبات"}
   166→          </span>
   167→          <span className="font-bold tabular-nums text-neutral-900 dark:text-neutral-100">
   168→            {entry.dataKey === "revenue"
   169→              ? entry.value.toLocaleString("ar-SA-u-nu-latn")
   170→              : entry.value + " طلب"}
   171→          </span>
   172→        </div>
   173→      ))}
   174→    </div>
   175→  );
   176→}
   177→
   178→function PieTooltip({
   179→  active,
   180→  payload,
   181→}: {
   182→  active?: boolean;
   183→  payload?: Array<{ name: string; value: number; payload: { fill: string; percent?: number } }>;
   184→}) {
   185→  if (!active || !payload?.length) return null;
   186→  const d = payload[0];
   187→  return (
   188→    <div className="bg-white dark:bg-neutral-800 border border-border rounded-lg shadow-lg px-3 py-2 text-sm" dir="rtl">
   189→      <div className="flex items-center gap-1.5">
   190→        <span className="w-2.5 h-2.5 rounded-sm" style={{ backgroundColor: d.payload.fill }} />
   191→        <span className="font-medium text-neutral-800 dark:text-neutral-200">{d.name}</span>
   192→      </div>
   193→      <div className="font-bold tabular-nums text-neutral-900 dark:text-neutral-100 mt-0.5">
   194→        {d.value} طلب {d.payload.percent ? `(${(d.payload.percent * 100).toFixed(0)}%)` : ""}
   195→      </div>
   196→    </div>
   197→  );
   198→}

   726→function WeeklyHeatmap({ weeks }: { weeks: WeekBucket[] }) {
   727→  const maxCount = Math.max(
   728→    ...weeks.flatMap((w) => w.days.map((d) => d.count)),
   729→    1,
   730→  );
   731→
   732→  function colorForCount(count: number) {
   733→    if (count === 0) return "bg-muted/60";
   734→    const intensity = count / maxCount;
   735→    if (intensity > 0.75) return "bg-amber-500 text-white";
   736→    if (intensity > 0.5) return "bg-amber-400 text-neutral-900 dark:text-neutral-900";
   737→    if (intensity > 0.25) return "bg-amber-300 text-neutral-900 dark:text-neutral-900";
   738→    return "bg-amber-200 text-neutral-900 dark:text-neutral-900";
   739→  }
   740→
   741→  return (
   742→    <div className="space-y-2">
   743→      <div className="flex items-center gap-1" dir="rtl">
   744→        <div className="w-14 md:w-20 text-xs text-muted-foreground shrink-0">
   745→          اليوم
   746→        </div>
   747→        {weeks.map((w, i) => (
   748→          <div
   749→            key={i}
   750→            title={w.rangeLabel}
   751→            className="flex-1 text-center text-xs font-bold text-neutral-700 tabular-nums cursor-help"
   752→          >
   753→            {w.label}
   754→          </div>
   755→        ))}
   756→      </div>
   757→      {WEEKDAY_NAMES_AR.map((dayName, dayIdx) => (
   758→        <div key={dayIdx} className="flex items-center gap-1" dir="rtl">
   759→          <div className="w-14 md:w-20 text-xs text-muted-foreground shrink-0 truncate">
   760→            {dayName}
   761→          </div>
   762→          {weeks.map((w, weekIdx) => {
   763→            const day = w.days.find((d) => d.date.getDay() === dayIdx);
   764→            const count = day?.count || 0;
   765→            const dateLabel = day
   766→              ? day.date.toLocaleDateString("ar-SA-u-nu-latn", {
   767→                  day: "2-digit",
   768→                  month: "2-digit",
   769→                })
   770→              : "";
   771→            return (
   772→              <div
   773→                key={weekIdx}
   774→                className={cn(
   775→                  "flex-1 aspect-square rounded-md flex items-center justify-center text-xs font-bold tabular-nums transition-colors",
   776→                  colorForCount(count),
   777→                )}
   778→                title={`${dayName} ${dateLabel}: ${count} طلب`}
   779→              >
   780→                {count > 0 ? count : ""}
   781→              </div>
   782→            );
   783→          })}
   784→        </div>
   785→      ))}
   786→      <div
   787→        className="flex items-center gap-1 pt-2 border-t border-border/50"
   788→        dir="rtl"
   789→      >
   790→        <div className="w-14 md:w-20 text-xs text-muted-foreground shrink-0">
   791→          الإجمالي
   792→        </div>
   793→        {weeks.map((w, i) => (
   794→          <div
   795→            key={i}
   796→            className="flex-1 text-center text-xs font-bold text-neutral-900 dark:text-neutral-100 tabular-nums py-1 rounded bg-muted/40"
   797→          >
   798→            {w.total}
   799→          </div>
   800→        ))}
   801→      </div>
   802→      <div className="flex items-center justify-center gap-3 pt-2 text-xs text-muted-foreground flex-wrap">
   803→        <span>أقل</span>
   804→        <span className="w-4 h-4 rounded-sm bg-muted/60" />
   805→        <span className="w-4 h-4 rounded-sm bg-amber-200" />
   806→        <span className="w-4 h-4 rounded-sm bg-amber-300" />
   807→        <span className="w-4 h-4 rounded-sm bg-amber-400" />
   808→        <span className="w-4 h-4 rounded-sm bg-amber-500" />
   809→        <span>أكثر</span>

   825→// ===== حساب كل المؤشرات =====
   826→function computeAnalytics(orders: PrintOrderLite[]): AnalyticsResult {
   827→  const now = new Date();
   828→  const thisMonth = now.getMonth();
   829→  const thisYear = now.getFullYear();
   830→
   831→  // طلبات الشهر الحالي
   832→  const monthlyOrders = orders.filter((o) => {
   833→    const d = new Date(o.createdAt);
   834→    return d.getMonth() === thisMonth && d.getFullYear() === thisYear;
   835→  });
   836→  const monthlyRevenue = monthlyOrders.reduce((s, o) => s + o.total, 0);
   837→  const monthlyCount = monthlyOrders.length;
   838→
   839→  // طلبات الشهر الماضي
   840→  const lastMonthDate = new Date(thisYear, thisMonth - 1, 1);
   841→  const lastMonthOrders = orders.filter((o) => {
   842→    const d = new Date(o.createdAt);
   843→    return (
   844→      d.getMonth() === lastMonthDate.getMonth() &&
   845→      d.getFullYear() === lastMonthDate.getFullYear()
   846→    );
   847→  });
   848→  const lastMonthRevenue = lastMonthOrders.reduce((s, o) => s + o.total, 0);
   849→  const lastMonthCount = lastMonthOrders.length;
   850→
   851→  const monthlyRevenueTrend =
   852→    lastMonthRevenue > 0
   853→      ? ((monthlyRevenue - lastMonthRevenue) / lastMonthRevenue) * 100
   854→      : monthlyRevenue > 0
   855→        ? 100
   856→        : 0;
   857→  const monthlyCountTrend =
   858→    lastMonthCount > 0
   859→      ? ((monthlyCount - lastMonthCount) / lastMonthCount) * 100
   860→      : monthlyCount > 0
   861→        ? 100
   862→        : 0;
   863→
   864→  const avgOrderValue = monthlyCount > 0 ? monthlyRevenue / monthlyCount : 0;
   865→
   866→  const nonCancelled = orders.filter((o) => o.status !== "cancelled");
   867→  const delivered = orders.filter((o) => o.status === "delivered");
   868→  const deliveryRate =
   869→    nonCancelled.length > 0
   870→      ? (delivered.length / nonCancelled.length) * 100
   871→      : 0;
   872→
   873→  // آخر 6 أشهر
   874→  const last6Months: MonthBucket[] = [];
   875→  for (let i = 5; i >= 0; i--) {
   876→    const d = new Date(thisYear, thisMonth - i, 1);
   877→    const m = d.getMonth();
   878→    const y = d.getFullYear();
   879→    const monthOrders = orders.filter((o) => {
   880→      const od = new Date(o.createdAt);
   881→      return od.getMonth() === m && od.getFullYear() === y;
   882→    });
   883→    last6Months.push({
   884→      label: MONTH_NAMES_AR[m],
   885→      revenue: monthOrders.reduce((s, o) => s + o.total, 0),
   886→      count: monthOrders.length,
   887→    });
   888→  }
   889→
   890→  // منحنى يومي (آخر 14 يوم)
   891→  const dailyTrend: DayBucket[] = [];
   892→  for (let i = 13; i >= 0; i--) {
   893→    const dayDate = new Date(now);
   894→    dayDate.setDate(now.getDate() - i);
   895→    dayDate.setHours(0, 0, 0, 0);
   896→    const nextDay = new Date(dayDate);
   897→    nextDay.setDate(dayDate.getDate() + 1);
   898→    const dayOrders = orders.filter((o) => {
   899→      const od = new Date(o.createdAt);
   900→      return od >= dayDate && od < nextDay;
   901→    });
   902→    dailyTrend.push({
   903→      label: dayDate.toLocaleDateString("ar-SA-u-nu-latn", {
   904→        day: "2-digit",
   905→        month: "2-digit",
   906→      }),
   907→      revenue: dayOrders.reduce((s, o) => s + o.total, 0),
   908→      count: dayOrders.length,
   909→    });
   910→  }
   911→
   912→  // أفضل العملاء
   913→  const customerMap: Record<string, TopCustomer> = {};
   914→  orders.forEach((o) => {
   915→    const key = o.customer.phone || o.customer.name;
   916→    if (!customerMap[key]) {
   917→      customerMap[key] = {
   918→        name: o.customer.name,
   919→        phone: o.customer.phone,
   920→        count: 0,
   921→        total: 0,
   922→      };
   923→    }
   924→    customerMap[key].count += 1;
   925→    customerMap[key].total += o.total;
   926→  });
   927→  const topCustomers = Object.values(customerMap)
   928→    .sort((a, b) => b.count - a.count || b.total - a.total)
   929→    .slice(0, 3);
   930→
   931→  // خريطة النشاط الأسبوعي
   932→  const today = new Date();
   933→  today.setHours(0, 0, 0, 0);
   934→  const dayOfWeek = today.getDay();
   935→  const currentWeekStartOffset = (dayOfWeek + 1) % 7;
   936→  const currentWeekStart = new Date(today);
   937→  currentWeekStart.setDate(today.getDate() - currentWeekStartOffset);
   938→
   939→  const weeklyHeatmap: WeekBucket[] = [];
   940→  for (let w = 3; w >= 0; w--) {
   941→    const weekStart = new Date(currentWeekStart);
   942→    weekStart.setDate(currentWeekStart.getDate() - w * 7);
   943→    const weekEnd = new Date(weekStart);
   944→    weekEnd.setDate(weekStart.getDate() + 6);
   945→    const days: { date: Date; count: number }[] = [];
   946→    let total = 0;
   947→    for (let d = 0; d < 7; d++) {
   948→      const dayDate = new Date(weekStart);
   949→      dayDate.setDate(weekStart.getDate() + d);
   950→      const dayCount = orders.filter((o) => {
   951→        const od = new Date(o.createdAt);
   952→        return od.toDateString() === dayDate.toDateString();
   953→      }).length;
   954→      days.push({ date: dayDate, count: dayCount });
   955→      total += dayCount;
   956→    }
   957→    const weekNum = 4 - w;
   958→    const rangeLabel = `${weekStart.toLocaleDateString("ar-SA-u-nu-latn", {
   959→      day: "2-digit",
   960→      month: "2-digit",
   961→    })} - ${weekEnd.toLocaleDateString("ar-SA-u-nu-latn", {
   962→      day: "2-digit",
   963→      month: "2-digit",
   964→    })}`;
   965→    weeklyHeatmap.push({ label: `أ${weekNum}`, rangeLabel, days, total });
   966→  }
   967→
   968→  return {
   969→    monthlyRevenue,
   970→    monthlyCount,
   971→    monthlyRevenueTrend,
   972→    monthlyCountTrend,
   973→    avgOrderValue,
   974→    deliveryRate,
   975→    last6Months,
   976→    topCustomers,
   977→    weeklyHeatmap,
   978→    dailyTrend,
   979→  };
   980→}
