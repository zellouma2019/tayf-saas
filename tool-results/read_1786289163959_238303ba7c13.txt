     1→"use client";
     2→
     3→import { useState, useCallback } from "react";
     4→import { useQuery, useQueryClient } from "@tanstack/react-query";
     5→import {
     6→  Card,
     7→  CardContent,
     8→} from "@/components/ui/card";
     9→import { Button } from "@/components/ui/button";
    10→import { Input } from "@/components/ui/input";
    11→import { Label } from "@/components/ui/label";
    12→import {
    13→  Select,
    14→  SelectContent,
    15→  SelectItem,
    16→  SelectTrigger,
    17→  SelectValue,
    18→} from "@/components/ui/select";
    19→import {
    20→  Table,
    21→  TableBody,
    22→  TableCell,
    23→  TableHead,
    24→  TableHeader,
    25→  TableRow,
    26→} from "@/components/ui/table";
    27→import {
    28→  Receipt,
    29→  Plus,
    30→  Trash2,
    31→  Loader2,
    32→  TrendingDown,
    33→  CalendarDays,
    34→  ChevronLeft,
    35→  ChevronRight,
    36→  Pencil,
    37→  Check,
    38→  X,
    39→} from "lucide-react";
    40→import { toast } from "sonner";
    41→import { useAppStore } from "@/lib/store";
    42→import { formatDA, formatDateAr } from "@/lib/print-config";
    43→
    44→const CATEGORIES = [
    45→  { value: "paper", label: "ورق", emoji: "📄" },
    46→  { value: "ink", label: "حبر", emoji: "🖤" },
    47→  { value: "maintenance", label: "صيانة", emoji: "🔧" },
    48→  { value: "rent", label: "إيجار", emoji: "🏠" },
    49→  { value: "other", label: "أخرى", emoji: "📦" },
    50→];
    51→
    52→const CATEGORY_MAP = Object.fromEntries(CATEGORIES.map((c) => [c.value, c]));
    53→
    54→interface Expense {
    55→  id: string;
    56→  category: string;
    57→  amount: number;
    58→  description: string | null;
    59→  date: string;
    60→  createdAt: string;
    61→  updatedAt: string;
    62→}
    63→
    64→interface PaginationInfo {
    65→  page: number;
    66→  limit: number;
    67→  total: number;
    68→  totalPages: number;
    69→}
    70→
    71→const HEADERS = { "Content-Type": "application/json" };
    72→
    73→function useAdminHeaders() {
    74→  const adminCode = useAppStore((s) => s.adminCode);
    75→  return { ...HEADERS, "x-admin-code": adminCode };
    76→}
    77→
    78→function todayStr() {
    79→  return new Date().toISOString().slice(0, 10);
    80→}
    81→
    82→function monthStartStr() {
    83→  const d = new Date();
    84→  d.setDate(1);
    85→  d.setHours(0, 0, 0, 0);
    86→  return d.toISOString();
    87→}
    88→
    89→export function AdminExpenses() {
    90→  const [categoryFilter, setCategoryFilter] = useState<string | null>(null);
    91→  const [page, setPage] = useState(1);
    92→  const [deletingId, setDeletingId] = useState<string | null>(null);
    93→  const [submitting, setSubmitting] = useState(false);
    94→  const [editingId, setEditingId] = useState<string | null>(null);
    95→  const [editCategory, setEditCategory] = useState("");
    96→  const [editAmount, setEditAmount] = useState("");
    97→  const [editDescription, setEditDescription] = useState("");
    98→  const [editDate, setEditDate] = useState("");
    99→  const [savingId, setSavingId] = useState<string | null>(null);
   100→
   101→  // Form state
   102→  const [formCategory, setFormCategory] = useState("");
   103→  const [formAmount, setFormAmount] = useState("");
   104→  const [formDescription, setFormDescription] = useState("");
   105→  const [formDate, setFormDate] = useState(todayStr());
   106→
   107→  const queryClient = useQueryClient();
   108→  const headers = useAdminHeaders();
   109→
   110→  const limit = 50;
   111→
   112→  const fetchExpenses = useCallback(async (): Promise<{
   113→    expenses: Expense[];
   114→    totalAmount: number;
   115→    pagination: PaginationInfo;
   116→  }> => {
   117→    const params = new URLSearchParams({
   118→      page: String(page),
   119→      limit: String(limit),
   120→    });
   121→    if (categoryFilter) params.set("category", categoryFilter);
   122→    const res = await fetch(`/api/expenses?${params}`, { headers });
   123→    if (!res.ok) throw new Error("فشل تحميل المصاريف");
   124→    return res.json();
   125→  }, [page, categoryFilter, headers]);
   126→
   127→  const fetchMonthTotal = useCallback(async (): Promise<number> => {
   128→    const from = monthStartStr();
   129→    const res = await fetch(`/api/expenses?from=${from}&limit=1`, { headers });
   130→    if (!res.ok) return 0;
   131→    const data = await res.json();
   132→    return data.totalAmount || 0;
   133→  }, [headers]);
   134→
   135→  const { data, isLoading } = useQuery({
   136→    queryKey: ["expenses", page, categoryFilter],
   137→    queryFn: fetchExpenses,
   138→    staleTime: 15_000,
   139→  });
   140→
   141→  const { data: monthTotal } = useQuery({
   142→    queryKey: ["expenses-month-total"],
   143→    queryFn: fetchMonthTotal,
   144→    staleTime: 30_000,
   145→  });
   146→
   147→  const expenses = data?.expenses ?? [];
   148→  const pagination = data?.pagination ?? { page: 1, limit, total: 0, totalPages: 0 };
   149→  const allTimeTotal = data?.totalAmount ?? 0;
   150→
   151→  const resetForm = () => {
   152→    setFormCategory("");
   153→    setFormAmount("");
   154→    setFormDescription("");
   155→    setFormDate(todayStr());
   156→  };
   157→
   158→  const handleSubmit = async (e: React.FormEvent) => {
   159→    e.preventDefault();
   160→    if (!formCategory || !formAmount) {
   161→      toast.error("الفئة والمبلغ مطلوبان");
   162→      return;
   163→    }
   164→    setSubmitting(true);
   165→    try {
   166→      const res = await fetch("/api/expenses", {
   167→        method: "POST",
   168→        headers,
   169→        body: JSON.stringify({
   170→          category: formCategory,
   171→          amount: Number(formAmount),
   172→          description: formDescription || undefined,
   173→          date: formDate || undefined,
   174→        }),
   175→      });
   176→      if (res.ok) {
   177→        toast.success("تمت إضافة المصروف بنجاح");
   178→        resetForm();
   179→        queryClient.invalidateQueries({ queryKey: ["expenses"] });
   180→        queryClient.invalidateQueries({ queryKey: ["expenses-month-total"] });
   181→      } else {
   182→        const r = await res.json();
   183→        toast.error(r.error || "فشل الإضافة");
   184→      }
   185→    } catch {
   186→      toast.error("خطأ في إضافة المصروف");
   187→    } finally {
   188→      setSubmitting(false);
   189→    }
   190→  };
   191→
   192→  const startEdit = (exp: Expense) => {
   193→    setEditingId(exp.id);
   194→    setEditCategory(exp.category);
   195→    setEditAmount(String(exp.amount));
   196→    setEditDescription(exp.description || "");
   197→    setEditDate(exp.date);
   198→  };
   199→
   200→  const cancelEdit = () => {
   201→    setEditingId(null);
   202→  };
   203→
   204→  const handleSaveEdit = async (id: string) => {
   205→    if (!editCategory || !editAmount) {
   206→      toast.error("الفئة والمبلغ مطلوبان");
   207→      return;
   208→    }
   209→    setSavingId(id);
   210→    try {
   211→      const res = await fetch(`/api/expenses/${id}`, {
   212→        method: "PUT",
   213→        headers,
   214→        body: JSON.stringify({
   215→          category: editCategory,
   216→          amount: Number(editAmount),
   217→          description: editDescription || undefined,
   218→          date: editDate || undefined,
   219→        }),
   220→      });
   221→      if (res.ok) {
   222→        toast.success("تم تحديث المصروف بنجاح");
   223→        setEditingId(null);
   224→        queryClient.invalidateQueries({ queryKey: ["expenses"] });
   225→        queryClient.invalidateQueries({ queryKey: ["expenses-month-total"] });
   226→      } else {
   227→        const r = await res.json();
   228→        toast.error(r.error || "فشل التحديث");
   229→      }
   230→    } catch {
   231→      toast.error("خطأ في تحديث المصروف");
   232→    } finally {
   233→      setSavingId(null);
   234→    }
   235→  };
   236→
   237→  const handleDelete = async (id: string) => {
   238→    setDeletingId(id);
   239→    try {
   240→      const res = await fetch(`/api/expenses/${id}`, {
   241→        method: "DELETE",
   242→        headers,
   243→      });
   244→      if (res.ok) {
   245→        toast.success("تم حذف المصروف");
   246→        queryClient.invalidateQueries({ queryKey: ["expenses"] });
   247→        queryClient.invalidateQueries({ queryKey: ["expenses-month-total"] });
   248→      } else {
   249→        const r = await res.json();
   250→        toast.error(r.error || "فشل الحذف");
   251→      }
   252→    } catch {
   253→      toast.error("خطأ في حذف المصروف");
   254→    } finally {
   255→      setDeletingId(null);
   256→    }
   257→  };
   258→
   259→  const handleFilterChange = (value: string) => {
   260→    setCategoryFilter(value === "__all__" ? null : value);
   261→    setPage(1);
   262→  };
   263→
   264→  return (
   265→    <div className="space-y-4">
   266→      {/* ===== العنوان ===== */}
   267→      <h2 className="text-lg font-bold flex items-center gap-2">
   268→        <Receipt className="h-5 w-5 text-amber-600" />
   269→        إدارة المصاريف
   270→      </h2>
   271→
   272→      {/* ===== نموذج الإضافة ===== */}
   273→      <Card className="p-4">
   274→        <form onSubmit={handleSubmit} className="space-y-3">
   275→          <p className="text-sm font-semibold flex items-center gap-1.5 mb-1">
   276→            <Plus className="h-4 w-4 text-emerald-600" />
   277→            إضافة مصروف جديد
   278→          </p>
   279→          <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-3">
   280→            <div className="space-y-1.5">
   281→              <Label className="text-xs">الفئة</Label>
   282→              <Select value={formCategory} onValueChange={setFormCategory}>
   283→                <SelectTrigger className="h-9">
   284→                  <SelectValue placeholder="اختر الفئة..." />
   285→                </SelectTrigger>
   286→                <SelectContent>
   287→                  {CATEGORIES.map((cat) => (
   288→                    <SelectItem key={cat.value} value={cat.value}>
   289→                      {cat.emoji} {cat.label}
   290→                    </SelectItem>
   291→                  ))}
   292→                </SelectContent>
   293→              </Select>
   294→            </div>
   295→            <div className="space-y-1.5">
   296→              <Label className="text-xs">المبلغ (دج)</Label>
   297→              <Input
   298→                type="number"
   299→                min="0"
   300→                placeholder="0"
   301→                value={formAmount}
   302→                onChange={(e) => setFormAmount(e.target.value)}
   303→                className="h-9"
   304→                dir="ltr"
   305→              />
   306→            </div>
   307→            <div className="space-y-1.5">
   308→              <Label className="text-xs">الوصف</Label>
   309→              <Input
   310→                placeholder="وصف اختياري..."
   311→                value={formDescription}
   312→                onChange={(e) => setFormDescription(e.target.value)}
   313→                className="h-9"
   314→              />
   315→            </div>
   316→            <div className="space-y-1.5">
   317→              <Label className="text-xs">التاريخ</Label>
   318→              <Input
   319→                type="date"
   320→                value={formDate}
   321→                onChange={(e) => setFormDate(e.target.value)}
   322→                className="h-9"
   323→                dir="ltr"
   324→              />
   325→            </div>
   326→          </div>
   327→          <Button type="submit" disabled={submitting || !formCategory || !formAmount} size="sm" className="gap-1.5">
   328→            {submitting ? (
   329→              <Loader2 className="h-4 w-4 animate-spin" />
   330→            ) : (
   331→              <Plus className="h-4 w-4" />
   332→            )}
   333→            إضافة
   334→          </Button>
   335→        </form>
   336→      </Card>
   337→
   338→      {/* ===== ملخص المصاريف ===== */}
   339→      <div className="grid grid-cols-2 gap-3">
   340→        <Card className="p-4">
   341→          <div className="flex items-center gap-2 mb-1">
   342→            <CalendarDays className="h-4 w-4 text-amber-600" />
   343→            <span className="text-xs text-muted-foreground">مصاريف الشهر</span>
   344→          </div>
   345→          <p className="text-xl font-bold">{formatDA(monthTotal ?? 0)}</p>
   346→        </Card>
   347→        <Card className="p-4">
   348→          <div className="flex items-center gap-2 mb-1">
   349→            <TrendingDown className="h-4 w-4 text-rose-600" />
   350→            <span className="text-xs text-muted-foreground">إجمالي المصاريف</span>
   351→          </div>
   352→          <p className="text-xl font-bold">{formatDA(allTimeTotal)}</p>
   353→        </Card>
   354→      </div>
   355→
   356→      {/* ===== فلتر الفئات ===== */}
   357→      <div className="flex flex-wrap gap-2">
   358→        <Button
   359→          variant={!categoryFilter ? "default" : "outline"}
   360→          size="sm"
   361→          className="h-8 text-xs"
   362→          onClick={() => handleFilterChange("__all__")}
   363→        >
   364→          الكل
   365→        </Button>
   366→        {CATEGORIES.map((cat) => (
   367→          <Button
   368→            key={cat.value}
   369→            variant={categoryFilter === cat.value ? "default" : "outline"}
   370→            size="sm"
   371→            className="h-8 text-xs gap-1"
   372→            onClick={() => handleFilterChange(cat.value)}
   373→          >
   374→            <span>{cat.emoji}</span>
   375→            <span>{cat.label}</span>
   376→          </Button>
   377→        ))}
   378→      </div>
   379→
   380→      {/* ===== التحميل ===== */}
   381→      {isLoading && (
   382→        <div className="py-16 text-center text-muted-foreground text-sm">
   383→          <Loader2 className="h-5 w-5 animate-spin mx-auto mb-2" />
   384→          جارٍ تحميل المصاريف...
   385→        </div>
   386→      )}
   387→
   388→      {/* ===== قائمة فارغة ===== */}
   389→      {!isLoading && expenses.length === 0 && (
   390→        <Card className="p-8 text-center">
   391→          <Receipt className="h-10 w-10 mx-auto mb-3 text-muted-foreground/40" />
   392→          <p className="text-muted-foreground text-sm">
   393→            {categoryFilter ? "لا توجد مصاريف في هذه الفئة" : "لا توجد مصاريف مسجّلة بعد"}
   394→          </p>
   395→        </Card>
   396→      )}
   397→
   398→      {/* ===== جدول سطح المكتب ===== */}
   399→      {!isLoading && expenses.length > 0 && (
   400→        <>
   401→          <Card className="hidden md:block overflow-hidden">
   402→            <Table>
   403→              <TableHeader>
   404→                <TableRow className="bg-muted/50">
   405→                  <TableHead>الفئة</TableHead>
   406→                  <TableHead className="text-center">المبلغ</TableHead>
   407→                  <TableHead>الوصف</TableHead>
   408→                  <TableHead>التاريخ</TableHead>
   409→                  <TableHead className="text-center">إجراءات</TableHead>
   410→                </TableRow>
   411→              </TableHeader>
   412→              <TableBody>
   413→                {expenses.map((exp) => {
   414→                  const cat = CATEGORY_MAP[exp.category];
   415→                  const isEditing = editingId === exp.id;
   416→                  return (
   417→                    <TableRow key={exp.id} className="group">
   418→                      <TableCell>
   419→                        {isEditing ? (
   420→                          <Select value={editCategory} onValueChange={setEditCategory}>
   421→                            <SelectTrigger className="h-8 w-32 text-xs">
   422→                              <SelectValue />
   423→                            </SelectTrigger>
   424→                            <SelectContent>
   425→                              {CATEGORIES.map((c) => (
   426→                                <SelectItem key={c.value} value={c.value}>
   427→                                  {c.emoji} {c.label}
   428→                                </SelectItem>
   429→                              ))}
   430→                            </SelectContent>
   431→                          </Select>
   432→                        ) : (
   433→                          <span className="flex items-center gap-1.5 text-sm font-medium">
   434→                            <span>{cat?.emoji || "📦"}</span>
   435→                            <span>{cat?.label || exp.category}</span>
   436→                          </span>
   437→                        )}
   438→                      </TableCell>
   439→                      <TableCell className="text-center">
   440→                        {isEditing ? (
   441→                          <Input
   442→                            type="number"
   443→                            min="0"
   444→                            value={editAmount}
   445→                            onChange={(e) => setEditAmount(e.target.value)}
   446→                            className="h-8 w-28 text-center text-xs mx-auto"
   447→                            dir="ltr"
   448→                          />
   449→                        ) : (
   450→                          <span className="font-semibold text-rose-600">
   451→                            {formatDA(exp.amount)}
   452→                          </span>
   453→                        )}
   454→                      </TableCell>
   455→                      <TableCell>
   456→                        {isEditing ? (
   457→                          <Input
   458→                            value={editDescription}
   459→                            onChange={(e) => setEditDescription(e.target.value)}
   460→                            className="h-8 text-xs"
   461→                            placeholder="وصف..."
   462→                          />
   463→                        ) : (
   464→                          <span className="text-sm text-muted-foreground max-w-[200px] truncate">
   465→                            {exp.description || "—"}
   466→                          </span>
   467→                        )}
   468→                      </TableCell>
   469→                      <TableCell>
   470→                        {isEditing ? (
   471→                          <Input
   472→                            type="date"
   473→                            value={editDate}
   474→                            onChange={(e) => setEditDate(e.target.value)}
   475→                            className="h-8 text-xs w-36"
   476→                            dir="ltr"
   477→                          />
   478→                        ) : (
   479→                          <span className="text-sm text-muted-foreground">
   480→                            {formatDateAr(exp.date)}
   481→                          </span>
   482→                        )}
   483→                      </TableCell>
   484→                      <TableCell className="text-center">
   485→                        <div className="flex items-center justify-center opacity-0 group-hover:opacity-100 transition-opacity">
   486→                          {isEditing ? (
   487→                            <>
   488→                              <Button
   489→                                variant="ghost"
   490→                                size="icon"
   491→                                className="h-8 w-8 text-emerald-600 hover:text-emerald-700 hover:bg-emerald-50"
   492→                                onClick={() => handleSaveEdit(exp.id)}
   493→                                disabled={savingId === exp.id}
   494→                              >
   495→                                {savingId === exp.id ? (
   496→                                  <Loader2 className="h-3.5 w-3.5 animate-spin" />
   497→                                ) : (
   498→                                  <Check className="h-3.5 w-3.5" />
   499→                                )}
   500→                              </Button>
   501→                              <Button
   502→                                variant="ghost"
   503→                                size="icon"
   504→                                className="h-8 w-8 text-muted-foreground hover:bg-muted"
   505→                                onClick={cancelEdit}
   506→                                disabled={savingId === exp.id}
   507→                              >
   508→                                <X className="h-3.5 w-3.5" />
   509→                              </Button>
   510→                            </>
   511→                          ) : (
   512→                            <>
   513→                              <Button
   514→                                variant="ghost"
   515→                                size="icon"
   516→                                className="h-8 w-8 text-amber-600 hover:text-amber-700 hover:bg-amber-50"
   517→                                onClick={() => startEdit(exp)}
   518→                              >
   519→                                <Pencil className="h-3.5 w-3.5" />
   520→                              </Button>
   521→                              <Button
   522→                                variant="ghost"
   523→                                size="icon"
   524→                                className="h-8 w-8 text-rose-500 hover:text-rose-600 hover:bg-rose-50"
   525→                                onClick={() => handleDelete(exp.id)}
   526→                                disabled={deletingId === exp.id}
   527→                              >
   528→                                {deletingId === exp.id ? (
   529→                                  <Loader2 className="h-3.5 w-3.5 animate-spin" />
   530→                                ) : (
   531→                                  <Trash2 className="h-3.5 w-3.5" />
   532→                                )}
   533→                              </Button>
   534→                            </>
   535→                          )}
   536→                        </div>
   537→                      </TableCell>
   538→                    </TableRow>
   539→                  );
   540→                })}
   541→              </TableBody>
   542→            </Table>
   543→          </Card>
   544→
   545→          {/* ===== بطاقات الجوال ===== */}
   546→          <div className="md:hidden space-y-3">
   547→            {expenses.map((exp) => {
   548→              const cat = CATEGORY_MAP[exp.category];
   549→              const isEditing = editingId === exp.id;
   550→              return (
   551→                <Card key={exp.id} className="p-4">
   552→                  {isEditing ? (
   553→                    <div className="space-y-3">
   554→                      <div className="space-y-2">
   555→                        <Label className="text-xs">الفئة</Label>
   556→                        <Select value={editCategory} onValueChange={setEditCategory}>
   557→                          <SelectTrigger className="h-9">
   558→                            <SelectValue />
   559→                          </SelectTrigger>
   560→                          <SelectContent>
   561→                            {CATEGORIES.map((c) => (
   562→                              <SelectItem key={c.value} value={c.value}>
   563→                                {c.emoji} {c.label}
   564→                              </SelectItem>
   565→                            ))}
   566→                          </SelectContent>
   567→                        </Select>
   568→                      </div>
   569→                      <div className="grid grid-cols-2 gap-2">
   570→                        <div className="space-y-1">
   571→                          <Label className="text-xs">المبلغ (دج)</Label>
   572→                          <Input
   573→                            type="number"
   574→                            min="0"
   575→                            value={editAmount}
   576→                            onChange={(e) => setEditAmount(e.target.value)}
   577→                            className="h-9"
   578→                            dir="ltr"
