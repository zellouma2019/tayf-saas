   117→   117→export function AdminShopManagement({ open, onClose, shop, onSaved }: AdminShopManagementProps) {
   118→   118→  const [data, setData] = useState<ShopData | null>(null);
   119→   119→  const [loading, setLoading] = useState(false);
   120→   120→  const [saving, setSaving] = useState(false);
   121→   121→  const [tab, setTab] = useState("basic");
   122→   122→  const [hasChanges, setHasChanges] = useState(false);
   123→   123→  const [originalData, setOriginalData] = useState<ShopData | null>(null);
   124→   124→
   125→   125→  useEffect(() => {
   126→   126→    if (!open || !shop) return;
   127→   127→    let cancelled = false;
   128→   128→    (async () => {
   129→   129→      setLoading(true);
   130→   130→      setTab("basic");
   131→   131→      try {
   132→   132→        const res = await fetch(`/api/admin/shops/${shop.slug}`);
   133→   133→        if (res.ok && !cancelled) {
   134→   134→          const d = await res.json();
   135→   135→          const s = d.shop;
   136→   136→          const parsed: ShopData = {
   137→   137→            slug: s.slug, name: s.name, phone: s.phone,
   138→   138→            ownerName: s.ownerName, ownerPhone: s.ownerPhone,
   139→   139→            whatsapp: s.whatsapp, email: s.email, address: s.address,
   140→   140→            country: s.country, language: s.language || "ar",
   141→   141→            customCurrency: s.customCurrency || null,
   142→   142→            isActive: s.isActive,
   143→   143→            plan: s.plan || "free", primaryColor: s.primaryColor,
   144→   144→            adminPin: s.adminPin, themeId: s.themeId || 1,
   145→   145→            logoIcon: s.logoIcon || "Printer",
   146→   146→            trialDays: s.trialDays, trialStartsAt: s.trialStartsAt,
   147→   147→            features: s.features ? (typeof s.features === "string" ? JSON.parse(s.features) : s.features) : {},
   148→   148→            ownerNotes: s.ownerNotes, paymentInfo: s.paymentInfo,
   149→   149→          };
   150→   150→          setData(parsed);
   151→   151→          setOriginalData(JSON.parse(JSON.stringify(parsed)));
   152→   152→          setHasChanges(false);
   153→   153→        } else if (!cancelled) {
   154→   154→          toast.error("فشل تحميل بيانات المتجر");
   155→   155→          onClose();
   156→   156→        }
   157→   157→      } catch { if (!cancelled) { toast.error("خطأ في الاتصال"); onClose(); } }
   158→   158→      finally { if (!cancelled) setLoading(false); }
   159→   159→    })();
   160→   160→    return () => { cancelled = true; };
   161→   161→  }, [open, shop, onClose]);
   162→   162→
   163→   163→  function setField<K extends keyof ShopData>(key: K, val: ShopData[K]) {
   164→   164→    setData((prev) => {
   165→   165→      if (!prev) return prev;
   166→   166→      const next = { ...prev, [key]: val };
   167→   167→      // عند تغيير الدولة، حدّث العملة تلقائياً إن لم يكن هناك عملة مخصصة
   168→   168→      if (key === "country" && !prev.customCurrency) {
   169→   169→        const country = ARAB_COUNTRIES.find((c) => c.code === val);
   170→   170→        if (country) {
   171→   171→          next.customCurrency = country.currencyCode;
   172→   172→        }
   173→   173→      }
   174→   174→      // عند تغيير الخطة إلى مدفوعة، فعّل كل الميزات والتبويبات والصلاحيات
   175→   175→      if (key === "plan" && val === "paid") {
   176→   176→        const allFeatures: Record<string, boolean> = {};
   177→   177→        SHOP_FEATURES.forEach((f) => { allFeatures[f.key] = true; });
   178→   178→        MERCHANT_ADMIN_TABS.forEach((t) => { allFeatures[t.key] = true; });
   179→   179→        MERCHANT_PERMISSIONS.forEach((p) => { allFeatures[p.key] = true; });
   180→   180→        next.features = allFeatures;
   181→   181→      }
   182→   182→      // عند تغيير الخطة إلى مجانية، أوقف الميزات المدفوعة فقط
   183→   183→      if (key === "plan" && val === "free") {
   184→   184→        const updated = { ...prev.features };
   185→   185→        SHOP_FEATURES.forEach((f) => { if (!f.isFree) updated[f.key] = false; });
   186→   186→        MERCHANT_ADMIN_TABS.forEach((t) => { updated[t.key] = true; });
   187→   187→        MERCHANT_PERMISSIONS.forEach((p) => { updated[p.key] = true; });
   188→   188→        next.features = updated;
   189→   189→      }
   190→   190→      return next;
   191→   191→    });
   192→   192→    setHasChanges(true);
   193→   193→  }
   194→   194→
   195→   195→  function toggleFeature(key: string) {
   196→   196→    setData((prev) => {
   197→   197→      if (!prev) return prev;
   198→   198→      return { ...prev, features: { ...prev.features, [key]: !prev.features[key] } };
   199→   199→    });
   200→   200→    setHasChanges(true);
   201→   201→  }

   203→   203→  async function handleSave() {
   204→   204→    if (!data) return;
   205→   205→    setSaving(true);
   206→   206→    try {
   207→   207→      const payload: Record<string, unknown> = {
   208→   208→        name: data.name, phone: data.phone, whatsapp: data.whatsapp,
   209→   209→        email: data.email, address: data.address, country: data.country,
   210→   210→        language: data.language, customCurrency: data.customCurrency,
   211→   211→        ownerName: data.ownerName, ownerPhone: data.ownerPhone,
   212→   212→        adminPin: data.adminPin, isActive: data.isActive, plan: data.plan,
   213→   213→        primaryColor: data.primaryColor, themeId: data.themeId,
   214→   214→        logoIcon: data.logoIcon, trialDays: data.trialDays,
   215→   215→        features: data.features, ownerNotes: data.ownerNotes,
   216→   216→        paymentInfo: data.paymentInfo,
   217→   217→      };
   218→   218→      const res = await fetch(`/api/admin/shops/${data.slug}`, {
   219→   219→        method: "PUT",
   220→   220→        headers: { "Content-Type": "application/json" },
   221→   221→        body: JSON.stringify(payload),
   222→   222→      });
   223→   223→      if (res.ok) {
   224→   224→        toast.success("تم حفظ إعدادات المتجر");
   225→   225→        setOriginalData(JSON.parse(JSON.stringify(data)));
   226→   226→        setHasChanges(false);
   227→   227→        onSaved();
   228→   228→      } else {
   229→   229→        toast.error("فشل حفظ الإعدادات");
   230→   230→      }
   231→   231→    } catch { toast.error("خطأ في الاتصال"); }
   232→   232→    finally { setSaving(false); }
   233→   233→  }
   234→   234→
   235→   235→  const tabs = [

   241→   241→    { key: "notes", label: "ملاحظات", icon: FileText },
   242→   242→  ];
   243→   243→
   244→   244→  // Trial status
   245→   245→  const trialStatus = (() => {
   246→   246→    if (!data?.trialDays || !data?.trialStartsAt) return null;
   247→   247→    const start = new Date(data.trialStartsAt).getTime();
   248→   248→    const end = start + data.trialDays * 86400000;

   250→   250→    const remaining = Math.max(0, Math.ceil((end - now) / 86400000));
   251→   251→    const active = now < end;
   252→   252→    return { active, remaining, total: data.trialDays, start: data.trialStartsAt, end: new Date(end).toISOString() };
   253→   253→  })();
   254→   254→
   255→   255→  return (
   256→   256→    <Dialog open={open} onOpenChange={(v) => { if (!v && hasChanges) { /* could warn */ } onClose(); }}>
   257→   257→      <DialogContent className="max-w-3xl max-h-[90vh] overflow-hidden flex flex-col p-0" dir="rtl">
   258→   258→        <div className="flex items-center justify-between px-6 py-4 border-b border-border/50 shrink-0">
   259→   259→          <div>
   260→   260→            <DialogTitle className="text-lg font-bold">إدارة المتجر</DialogTitle>
   261→   261→            <DialogDescription className="text-xs mt-0.5">تحكم شامل بإعدادات وميزات المتجر</DialogDescription>
   262→   262→          </div>
   263→   263→          {hasChanges && (
   264→   264→            <Badge variant="outline" className="text-amber-600 border-amber-300 bg-amber-50 dark:bg-amber-500/10 dark:border-amber-500/30 dark:text-amber-400">
   265→   265→              تعديلات غير محفوظة
   266→   266→            </Badge>
   267→   267→          )}
   268→   268→        </div>
   269→   269→
   270→   270→        {/* Tabs */}
   271→   271→        <div className="flex gap-1 px-4 pt-3 pb-0 overflow-x-auto no-scrollbar shrink-0">
   272→   272→          {tabs.map((t) => (
   273→   273→            <button
   274→   274→              key={t.key}
   275→   275→              onClick={() => setTab(t.key)}
   276→   276→              className={`flex items-center gap-1.5 px-3 py-2 rounded-lg text-xs font-medium whitespace-nowrap transition-all ${
   277→   277→                tab === t.key
   278→   278→                  ? "bg-amber-500/10 text-amber-600 dark:text-amber-400"
   279→   279→                  : "text-muted-foreground hover:text-foreground hover:bg-muted/50"
   280→   280→              }`}
   281→   281→            >
   282→   282→              <t.icon className="h-3.5 w-3.5" />
   283→   283→              <span className="hidden sm:inline">{t.label}</span>
   284→   284→            </button>
   285→   285→          ))}
   286→   286→        </div>
   287→   287→
   288→   288→        {/* Content */}
   289→   289→        <div className="flex-1 overflow-y-auto px-6 py-4">
   290→   290→          {loading ? (
   291→   291→            <div className="space-y-4">
   292→   292→              {[1,2,3].map((i) => <div key={i} className="h-10 rounded-lg bg-muted animate-pulse" />)}
   293→   293→            </div>
   294→   294→          ) : !data ? null : (
   295→   295→            <>
   296→   296→              {/* ===== BASIC TAB ===== */}
   297→   297→              {tab === "basic" && (
   298→   298→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   299→   299→                  <div className="sm:col-span-2">
   300→   300→                    <Label>اسم المتجر</Label>
   301→   301→                    <Input value={data.name} onChange={(e) => setField("name", e.target.value)} className="mt-1" />
