    95→interface ShopData {
    96→  slug: string; name: string; phone: string | null;
    97→  ownerName: string | null; ownerPhone: string | null;
    98→  whatsapp: string | null; email: string | null; address: string | null;
    99→  country: string | null; language: string | null; customCurrency: string | null;
   100→  isActive: boolean; plan: string;
   101→  primaryColor: string | null; adminPin: string;
   102→  themeId: number | null;
   103→  logoIcon: string | null;
   104→  trialDays: number | null; trialStartsAt: string | null;
   105→  features: Record<string, boolean>;
   106→  ownerNotes: string | null;
   107→  paymentInfo: string | null;
   108→}
   109→
   110→interface AdminShopManagementProps {
   111→  open: boolean;
   112→  onClose: () => void;
   113→  shop: { slug: string; name: string } | null;
   114→  onSaved: () => void;
   115→}
   116→
   117→export function AdminShopManagement({ open, onClose, shop, onSaved }: AdminShopManagementProps) {
   118→  const [data, setData] = useState<ShopData | null>(null);
   119→  const [loading, setLoading] = useState(false);
   120→  const [saving, setSaving] = useState(false);
   121→  const [tab, setTab] = useState("basic");
   122→  const [hasChanges, setHasChanges] = useState(false);
   123→  const [originalData, setOriginalData] = useState<ShopData | null>(null);
   124→
   125→  useEffect(() => {
   126→    if (!open || !shop) return;
   127→    let cancelled = false;
   128→    (async () => {
   129→      setLoading(true);
   130→      setTab("basic");
   131→      try {
   132→        const res = await fetch(`/api/admin/shops/${shop.slug}`);
   133→        if (res.ok && !cancelled) {
   134→          const d = await res.json();
   135→          const s = d.shop;
   136→          const parsed: ShopData = {
   137→            slug: s.slug, name: s.name, phone: s.phone,
   138→            ownerName: s.ownerName, ownerPhone: s.ownerPhone,
   139→            whatsapp: s.whatsapp, email: s.email, address: s.address,
   140→            country: s.country, language: s.language || "ar",
   141→            customCurrency: s.customCurrency || null,
   142→            isActive: s.isActive,
   143→            plan: s.plan || "free", primaryColor: s.primaryColor,
   144→            adminPin: s.adminPin, themeId: s.themeId || 1,
   145→            logoIcon: s.logoIcon || "Printer",
   146→            trialDays: s.trialDays, trialStartsAt: s.trialStartsAt,
   147→            features: s.features ? (typeof s.features === "string" ? JSON.parse(s.features) : s.features) : {},
   148→            ownerNotes: s.ownerNotes, paymentInfo: s.paymentInfo,
   149→          };
   150→          setData(parsed);
   151→          setOriginalData(JSON.parse(JSON.stringify(parsed)));
   152→          setHasChanges(false);
   153→        } else if (!cancelled) {
   154→          toast.error("فشل تحميل بيانات المتجر");
   155→          onClose();
   156→        }
   157→      } catch { if (!cancelled) { toast.error("خطأ في الاتصال"); onClose(); } }
   158→      finally { if (!cancelled) setLoading(false); }
   159→    })();
   160→    return () => { cancelled = true; };
   161→  }, [open, shop, onClose]);
   162→
   163→  function setField<K extends keyof ShopData>(key: K, val: ShopData[K]) {
   164→    setData((prev) => {
   165→      if (!prev) return prev;
   166→      const next = { ...prev, [key]: val };
   167→      // عند تغيير الدولة، حدّث العملة تلقائياً إن لم يكن هناك عملة مخصصة
   168→      if (key === "country" && !prev.customCurrency) {
   169→        const country = ARAB_COUNTRIES.find((c) => c.code === val);
   170→        if (country) {
   171→          next.customCurrency = country.currencyCode;
   172→        }
   173→      }
   174→      // عند تغيير الخطة إلى مدفوعة، فعّل كل الميزات والتبويبات والصلاحيات
   175→      if (key === "plan" && val === "paid") {
   176→        const allFeatures: Record<string, boolean> = {};
   177→        SHOP_FEATURES.forEach((f) => { allFeatures[f.key] = true; });
   178→        MERCHANT_ADMIN_TABS.forEach((t) => { allFeatures[t.key] = true; });
   179→        MERCHANT_PERMISSIONS.forEach((p) => { allFeatures[p.key] = true; });
   180→        next.features = allFeatures;
   181→      }
   182→      // عند تغيير الخطة إلى مجانية، أوقف الميزات المدفوعة فقط
   183→      if (key === "plan" && val === "free") {
   184→        const updated = { ...prev.features };
   185→        SHOP_FEATURES.forEach((f) => { if (!f.isFree) updated[f.key] = false; });
   186→        MERCHANT_ADMIN_TABS.forEach((t) => { updated[t.key] = true; });
   187→        MERCHANT_PERMISSIONS.forEach((p) => { updated[p.key] = true; });
   188→        next.features = updated;
   189→      }
   190→      return next;
   191→    });
   192→    setHasChanges(true);
   193→  }
   194→
   195→  function toggleFeature(key: string) {
   196→    setData((prev) => {
   197→      if (!prev) return prev;
   198→      return { ...prev, features: { ...prev.features, [key]: !prev.features[key] } };
   199→    });
   200→    setHasChanges(true);
   201→  }
   202→
   203→  async function handleSave() {
   204→    if (!data) return;
   205→    setSaving(true);
   206→    try {
   207→      const payload: Record<string, unknown> = {
   208→        name: data.name, phone: data.phone, whatsapp: data.whatsapp,
   209→        email: data.email, address: data.address, country: data.country,
   210→        language: data.language, customCurrency: data.customCurrency,
   211→        ownerName: data.ownerName, ownerPhone: data.ownerPhone,
   212→        adminPin: data.adminPin, isActive: data.isActive, plan: data.plan,
   213→        primaryColor: data.primaryColor, themeId: data.themeId,
   214→        logoIcon: data.logoIcon, trialDays: data.trialDays,
   215→        features: data.features, ownerNotes: data.ownerNotes,
   216→        paymentInfo: data.paymentInfo,
   217→      };
   218→      const res = await fetch(`/api/admin/shops/${data.slug}`, {
   219→        method: "PUT",
   220→        headers: { "Content-Type": "application/json" },
   221→        body: JSON.stringify(payload),
   222→      });
   223→      if (res.ok) {
   224→        toast.success("تم حفظ إعدادات المتجر");
   225→        setOriginalData(JSON.parse(JSON.stringify(data)));
   226→        setHasChanges(false);
   227→        onSaved();
   228→      } else {
   229→        toast.error("فشل حفظ الإعدادات");
   230→      }
   231→    } catch { toast.error("خطأ في الاتصال"); }
   232→    finally { setSaving(false); }
   233→  }
   234→
   235→  const tabs = [
   236→    { key: "basic", label: "الأساسيات", icon: Store },
   237→    { key: "plan", label: "الخطة والتجربة", icon: CreditCard },
   238→    { key: "features", label: "ميزات المتجر", icon: Sparkles },
   239→    { key: "merchant", label: "لوحة التحكم", icon: LayoutDashboard },
   240→    { key: "appearance", label: "المظهر", icon: Palette },
   241→    { key: "notes", label: "ملاحظات", icon: FileText },
   242→  ];
   243→
   244→  // Trial status
   245→  const trialStatus = (() => {
   246→    if (!data?.trialDays || !data?.trialStartsAt) return null;
   247→    const start = new Date(data.trialStartsAt).getTime();
   248→    const end = start + data.trialDays * 86400000;
   249→    const now = Date.now();
   250→    const remaining = Math.max(0, Math.ceil((end - now) / 86400000));
   251→    const active = now < end;
   252→    return { active, remaining, total: data.trialDays, start: data.trialStartsAt, end: new Date(end).toISOString() };
   253→  })();
   254→
   255→  return (
   256→    <Dialog open={open} onOpenChange={(v) => { if (!v && hasChanges) { /* could warn */ } onClose(); }}>
   257→      <DialogContent className="max-w-3xl max-h-[90vh] overflow-hidden flex flex-col p-0" dir="rtl">
   258→        <div className="flex items-center justify-between px-6 py-4 border-b border-border/50 shrink-0">
   259→          <div>
   260→            <DialogTitle className="text-lg font-bold">إدارة المتجر</DialogTitle>
   261→            <DialogDescription className="text-xs mt-0.5">تحكم شامل بإعدادات وميزات المتجر</DialogDescription>
   262→          </div>
   263→          {hasChanges && (
   264→            <Badge variant="outline" className="text-amber-600 border-amber-300 bg-amber-50 dark:bg-amber-500/10 dark:border-amber-500/30 dark:text-amber-400">
   265→              تعديلات غير محفوظة
   266→            </Badge>
   267→          )}
   268→        </div>
   269→
   270→        {/* Tabs */}
   271→        <div className="flex gap-1 px-4 pt-3 pb-0 overflow-x-auto no-scrollbar shrink-0">
   272→          {tabs.map((t) => (
   273→            <button
   274→              key={t.key}
   275→              onClick={() => setTab(t.key)}
   276→              className={`flex items-center gap-1.5 px-3 py-2 rounded-lg text-xs font-medium whitespace-nowrap transition-all ${
   277→                tab === t.key
   278→                  ? "bg-amber-500/10 text-amber-600 dark:text-amber-400"
   279→                  : "text-muted-foreground hover:text-foreground hover:bg-muted/50"
   280→              }`}
   281→            >
   282→              <t.icon className="h-3.5 w-3.5" />
   283→              <span className="hidden sm:inline">{t.label}</span>
   284→            </button>
   285→          ))}
   286→        </div>
   287→
   288→        {/* Content */}
   289→        <div className="flex-1 overflow-y-auto px-6 py-4">
   290→          {loading ? (
   291→            <div className="space-y-4">
   292→              {[1,2,3].map((i) => <div key={i} className="h-10 rounded-lg bg-muted animate-pulse" />)}
   293→            </div>
   294→          ) : !data ? null : (
   295→            <>
   296→              {/* ===== BASIC TAB ===== */}
   297→              {tab === "basic" && (
   298→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   299→                  <div className="sm:col-span-2">
   300→                    <Label>اسم المتجر</Label>
   301→                    <Input value={data.name} onChange={(e) => setField("name", e.target.value)} className="mt-1" />
   302→                  </div>
   303→                  <div>
   304→                    <Label>المعرف (slug)</Label>
   305→                    <Input value={data.slug} disabled className="mt-1 bg-muted/50" />
   306→                    <p className="text-[10px] text-muted-foreground mt-1">لا يمكن تغييره بعد الإنشاء</p>
   307→                  </div>
   308→                  <div>
   309→                    <Label>هاتف المتجر</Label>
   310→                    <Input value={data.phone || ""} onChange={(e) => setField("phone", e.target.value || null)} className="mt-1" dir="ltr" />
   311→                  </div>
   312→                  <div>
   313→                    <Label>واتساب</Label>
   314→                    <Input value={data.whatsapp || ""} onChange={(e) => setField("whatsapp", e.target.value || null)} className="mt-1" dir="ltr" />
   315→                  </div>
   316→                  <div>
   317→                    <Label>البريد الإلكتروني</Label>
   318→                    <Input type="email" value={data.email || ""} onChange={(e) => setField("email", e.target.value || null)} className="mt-1" dir="ltr" />
   319→                  </div>
   320→                  <div className="sm:col-span-2">
   321→                    <Label>العنوان</Label>
   322→                    <Input value={data.address || ""} onChange={(e) => setField("address", e.target.value || null)} className="mt-1" />
   323→                  </div>
   324→                  <div>
   325→                    <Label>الدولة</Label>
   326→                    <Select value={data.country || "DZ"} onValueChange={(v) => setField("country", v)}>
   327→                      <SelectTrigger className="mt-1"><SelectValue /></SelectTrigger>
   328→                      <SelectContent>
   329→                        {ARAB_COUNTRIES.map((c) => (
   330→                          <SelectItem key={c.code} value={c.code}>
   331→                            <span className="flex items-center gap-2">
   332→                              <span>{c.flag}</span>
   333→                              <span>{c.nameAr}</span>
   334→                              <span className="text-muted-foreground text-xs">({c.code})</span>
   335→                            </span>
   336→                          </SelectItem>
   337→                        ))}
   338→                      </SelectContent>
   339→                    </Select>
   340→                  </div>
   341→                  <div>
   342→                    <Label>اللغة</Label>
   343→                    <Select value={data.language || "ar"} onValueChange={(v) => setField("language", v)}>
   344→                      <SelectTrigger className="mt-1"><SelectValue /></SelectTrigger>
   345→                      <SelectContent>

   354→                  <div>

   356→                    <Select
   357→                      value={data.customCurrency || ARAB_COUNTRIES.find(c => c.code === data.country)?.currencyCode || "DZD"}
