   142→// ===== تلميح مخصص بالعربية =====
   143→function ArabicTooltip({
   144→  active,
   145→  payload,
   146→  label,
   147→  valueLabel,
   148→  countLabel,
   149→}: {
   150→  active?: boolean;
   151→  payload?: Array<{ value: number; name: string; color: string; dataKey: string }>;
   152→  label?: string;
   153→  valueLabel?: string;
   154→  countLabel?: string;
   155→}) {
   156→  if (!active || !payload?.length) return null;
   157→  return (
   158→    <div className="bg-white border border-border rounded-lg shadow-lg px-3 py-2 text-sm min-w-[140px]" dir="rtl">
   159→      <div className="font-bold text-neutral-800 mb-1">{label}</div>
   160→      {payload.map((entry, i) => (
   161→        <div key={i} className="flex items-center justify-between gap-3">
   162→          <span className="flex items-center gap-1.5 text-xs text-muted-foreground">
   163→            <span className="w-2.5 h-2.5 rounded-sm inline-block" style={{ backgroundColor: entry.color }} />
   164→            {entry.dataKey === "revenue" ? valueLabel || "الإيرادات" : countLabel || "الطلبات"}
   165→          </span>
   166→          <span className="font-bold tabular-nums text-neutral-900">
   167→            {entry.dataKey === "revenue"
   168→              ? entry.value.toLocaleString("ar-DZ") + " دج"
   169→              : entry.value + " طلب"}
   170→          </span>
   171→        </div>
   172→      ))}
   173→    </div>
   174→  );
   175→}
   176→
   177→function PieTooltip({
   178→  active,
   179→  payload,
   180→}: {
   181→  active?: boolean;
   182→  payload?: Array<{ name: string; value: number; payload: { fill: string; percent?: number } }>;
   183→}) {
   184→  if (!active || !payload?.length) return null;
   185→  const d = payload[0];
   186→  return (
   187→    <div className="bg-white border border-border rounded-lg shadow-lg px-3 py-2 text-sm" dir="rtl">
   188→      <div className="flex items-center gap-1.5">
   189→        <span className="w-2.5 h-2.5 rounded-sm" style={{ backgroundColor: d.payload.fill }} />
   190→        <span className="font-medium text-neutral-800">{d.name}</span>
   191→      </div>
   192→      <div className="font-bold tabular-nums text-neutral-900 mt-0.5">
   193→        {d.value} طلب {d.payload.percent ? `(${(d.payload.percent * 100).toFixed(0)}%)` : ""}
   194→      </div>
   195→    </div>
   196→  );
   197→}

   724→// ===== خريطة النشاط الأسبوعي (CSS Grid) =====
   725→function WeeklyHeatmap({ weeks }: { weeks: WeekBucket[] }) {
   726→  const maxCount = Math.max(
   727→    ...weeks.flatMap((w) => w.days.map((d) => d.count)),
   728→    1,
   729→  );
   730→
   731→  function colorForCount(count: number) {
   732→    if (count === 0) return "bg-muted/60";
   733→    const intensity = count / maxCount;
   734→    if (intensity > 0.75) return "bg-amber-500 text-white";
   735→    if (intensity > 0.5) return "bg-amber-400 text-neutral-900";
   736→    if (intensity > 0.25) return "bg-amber-300 text-neutral-900";
   737→    return "bg-amber-200 text-neutral-900";
   738→  }
   739→
   740→  return (
   741→    <div className="space-y-2">
   742→      <div className="flex items-center gap-1" dir="rtl">
   743→        <div className="w-14 md:w-20 text-xs text-muted-foreground shrink-0">
   744→          اليوم
   745→        </div>
   746→        {weeks.map((w, i) => (
   747→          <div
   748→            key={i}
   749→            title={w.rangeLabel}
   750→            className="flex-1 text-center text-xs font-bold text-neutral-700 tabular-nums cursor-help"
   751→          >
   752→            {w.label}
   753→          </div>
   754→        ))}
   755→      </div>
   756→      {WEEKDAY_NAMES_AR.map((dayName, dayIdx) => (
   757→        <div key={dayIdx} className="flex items-center gap-1" dir="rtl">
   758→          <div className="w-14 md:w-20 text-xs text-muted-foreground shrink-0 truncate">
   759→            {dayName}
   760→          </div>
   761→          {weeks.map((w, weekIdx) => {
   762→            const day = w.days.find((d) => d.date.getDay() === dayIdx);
   763→            const count = day?.count || 0;
   764→            const dateLabel = day
   765→              ? day.date.toLocaleDateString("ar-DZ-u-nu-latn", {
   766→                  day: "2-digit",
   767→                  month: "2-digit",
   768→                })
   769→              : "";
   770→            return (
   771→              <div
   772→                key={weekIdx}
   773→                className={cn(
   774→                  "flex-1 aspect-square rounded-md flex items-center justify-center text-xs font-bold tabular-nums transition-colors",
   775→                  colorForCount(count),
   776→                )}
   777→                title={`${dayName} ${dateLabel}: ${count} طلب`}
   778→              >
   779→                {count > 0 ? count : ""}
   780→              </div>
   781→            );
   782→          })}
   783→        </div>
   784→      ))}
   785→      <div
   786→        className="flex items-center gap-1 pt-2 border-t border-border/50"
   787→        dir="rtl"
   788→      >
   789→        <div className="w-14 md:w-20 text-xs text-muted-foreground shrink-0">
   790→          الإجمالي
   791→        </div>
   792→        {weeks.map((w, i) => (
   793→          <div
   794→            key={i}
   795→            className="flex-1 text-center text-xs font-bold text-neutral-900 tabular-nums py-1 rounded bg-muted/40"
   796→          >
   797→            {w.total}
   798→          </div>
   799→        ))}
   800→      </div>
   801→      <div className="flex items-center justify-center gap-3 pt-2 text-xs text-muted-foreground flex-wrap">
   802→        <span>أقل</span>
   803→        <span className="w-4 h-4 rounded-sm bg-muted/60" />
   804→        <span className="w-4 h-4 rounded-sm bg-amber-200" />
   805→        <span className="w-4 h-4 rounded-sm bg-amber-300" />
   806→        <span className="w-4 h-4 rounded-sm bg-amber-400" />
   807→        <span className="w-4 h-4 rounded-sm bg-amber-500" />
   808→        <span>أكثر</span>
   809→      </div>
   810→    </div>

   824→// ===== حساب كل المؤشرات =====
   825→function computeAnalytics(orders: PrintOrderLite[]): AnalyticsResult {
   826→  const now = new Date();
   827→  const thisMonth = now.getMonth();
   828→  const thisYear = now.getFullYear();
   829→
   830→  // طلبات الشهر الحالي
   831→  const monthlyOrders = orders.filter((o) => {
   832→    const d = new Date(o.createdAt);
   833→    return d.getMonth() === thisMonth && d.getFullYear() === thisYear;
   834→  });
   835→  const monthlyRevenue = monthlyOrders.reduce((s, o) => s + o.total, 0);
   836→  const monthlyCount = monthlyOrders.length;
   837→
   838→  // طلبات الشهر الماضي
   839→  const lastMonthDate = new Date(thisYear, thisMonth - 1, 1);
   840→  const lastMonthOrders = orders.filter((o) => {
   841→    const d = new Date(o.createdAt);
   842→    return (
   843→      d.getMonth() === lastMonthDate.getMonth() &&
   844→      d.getFullYear() === lastMonthDate.getFullYear()
   845→    );
   846→  });
   847→  const lastMonthRevenue = lastMonthOrders.reduce((s, o) => s + o.total, 0);
   848→  const lastMonthCount = lastMonthOrders.length;
   849→
   850→  const monthlyRevenueTrend =
   851→    lastMonthRevenue > 0
   852→      ? ((monthlyRevenue - lastMonthRevenue) / lastMonthRevenue) * 100
   853→      : monthlyRevenue > 0
   854→        ? 100
   855→        : 0;
   856→  const monthlyCountTrend =
   857→    lastMonthCount > 0
   858→      ? ((monthlyCount - lastMonthCount) / lastMonthCount) * 100
   859→      : monthlyCount > 0
   860→        ? 100
   861→        : 0;
   862→
   863→  const avgOrderValue = monthlyCount > 0 ? monthlyRevenue / monthlyCount : 0;
   864→
   865→  const nonCancelled = orders.filter((o) => o.status !== "cancelled");
   866→  const delivered = orders.filter((o) => o.status === "delivered");
   867→  const deliveryRate =
   868→    nonCancelled.length > 0
   869→      ? (delivered.length / nonCancelled.length) * 100
   870→      : 0;
   871→
   872→  // آخر 6 أشهر
   873→  const last6Months: MonthBucket[] = [];
   874→  for (let i = 5; i >= 0; i--) {
   875→    const d = new Date(thisYear, thisMonth - i, 1);
   876→    const m = d.getMonth();
   877→    const y = d.getFullYear();
   878→    const monthOrders = orders.filter((o) => {
   879→      const od = new Date(o.createdAt);
   880→      return od.getMonth() === m && od.getFullYear() === y;
   881→    });
   882→    last6Months.push({
   883→      label: MONTH_NAMES_AR[m],
   884→      revenue: monthOrders.reduce((s, o) => s + o.total, 0),
   885→      count: monthOrders.length,
   886→    });
   887→  }
   888→
   889→  // منحنى يومي (آخر 14 يوم)
   890→  const dailyTrend: DayBucket[] = [];
   891→  for (let i = 13; i >= 0; i--) {
   892→    const dayDate = new Date(now);
   893→    dayDate.setDate(now.getDate() - i);
   894→    dayDate.setHours(0, 0, 0, 0);
   895→    const nextDay = new Date(dayDate);
   896→    nextDay.setDate(dayDate.getDate() + 1);
   897→    const dayOrders = orders.filter((o) => {
   898→      const od = new Date(o.createdAt);
   899→      return od >= dayDate && od < nextDay;
   900→    });
   901→    dailyTrend.push({
   902→      label: dayDate.toLocaleDateString("ar-DZ-u-nu-latn", {
   903→        day: "2-digit",
   904→        month: "2-digit",
   905→      }),
   906→      revenue: dayOrders.reduce((s, o) => s + o.total, 0),
   907→      count: dayOrders.length,
   908→    });
   909→  }
   910→
   911→  // أفضل العملاء
   912→  const customerMap: Record<string, TopCustomer> = {};
   913→  orders.forEach((o) => {
   914→    const key = o.customer.phone || o.customer.name;
   915→    if (!customerMap[key]) {
   916→      customerMap[key] = {
   917→        name: o.customer.name,
   918→        phone: o.customer.phone,
   919→        count: 0,
   920→        total: 0,
   921→      };
   922→    }
   923→    customerMap[key].count += 1;
   924→    customerMap[key].total += o.total;
   925→  });
   926→  const topCustomers = Object.values(customerMap)
   927→    .sort((a, b) => b.count - a.count || b.total - a.total)
   928→    .slice(0, 3);
   929→
   930→  // خريطة النشاط الأسبوعي
   931→  const today = new Date();
   932→  today.setHours(0, 0, 0, 0);
   933→  const dayOfWeek = today.getDay();
   934→  const currentWeekStartOffset = (dayOfWeek + 1) % 7;
   935→  const currentWeekStart = new Date(today);
   936→  currentWeekStart.setDate(today.getDate() - currentWeekStartOffset);
   937→
   938→  const weeklyHeatmap: WeekBucket[] = [];
   939→  for (let w = 3; w >= 0; w--) {
   940→    const weekStart = new Date(currentWeekStart);
   941→    weekStart.setDate(currentWeekStart.getDate() - w * 7);
   942→    const weekEnd = new Date(weekStart);
   943→    weekEnd.setDate(weekStart.getDate() + 6);
   944→    const days: { date: Date; count: number }[] = [];
   945→    let total = 0;
   946→    for (let d = 0; d < 7; d++) {
   947→      const dayDate = new Date(weekStart);
   948→      dayDate.setDate(weekStart.getDate() + d);
   949→      const dayCount = orders.filter((o) => {
   950→        const od = new Date(o.createdAt);
   951→        return od.toDateString() === dayDate.toDateString();
   952→      }).length;
   953→      days.push({ date: dayDate, count: dayCount });
   954→      total += dayCount;
   955→    }
   956→    const weekNum = 4 - w;
   957→    const rangeLabel = `${weekStart.toLocaleDateString("ar-DZ-u-nu-latn", {
   958→      day: "2-digit",
   959→      month: "2-digit",
   960→    })} - ${weekEnd.toLocaleDateString("ar-DZ-u-nu-latn", {
   961→      day: "2-digit",
   962→      month: "2-digit",
   963→    })}`;
   964→    weeklyHeatmap.push({ label: `أ${weekNum}`, rangeLabel, days, total });
   965→  }
   966→
   967→  return {
   968→    monthlyRevenue,
   969→    monthlyCount,
   970→    monthlyRevenueTrend,
   971→    monthlyCountTrend,
   972→    avgOrderValue,
   973→    deliveryRate,
   974→    last6Months,
   975→    topCustomers,
   976→    weeklyHeatmap,
   977→    dailyTrend,
   978→  };
   979→}
