    94→/** توليد معرّف فريد قصير */
    95→function uid(prefix = "id"): string {
    96→  return `${prefix}_${Math.random().toString(36).slice(2, 9)}`;
    97→}
    98→
    99→/** نسخة عميقة آمنة (للإعدادات القابلة للتسلسل JSON) */
   100→function deepClone<T>(value: T): T {
   101→  return JSON.parse(JSON.stringify(value)) as T;
   102→}
   103→
   104→/** مقارنة عميقة لكائنين قابلين للتسلسل */
   105→function deepEqual(a: unknown, b: unknown): boolean {
   106→  return JSON.stringify(a) === JSON.stringify(b);
   107→}
   108→
   109→/** تحويل قيمة نصية إلى رقم بأمان */
   110→function toNumber(value: string): number {
   111→  const n = Number(value);
   112→  return Number.isFinite(n) ? n : 0;
   113→}

   119→interface FieldProps {
   120→  label: string;
   121→  value: string | number;
   122→  onChange: (v: string) => void;
   123→  type?: "text" | "number";
   124→  placeholder?: string;
   125→  dir?: "rtl" | "ltr";
   126→  className?: string;
   127→  hint?: string;

   129→
   130→function Field({
   131→  label,
   132→  value,
   133→  onChange,
   134→  type = "text",
   135→  placeholder,
   136→  dir,
   137→  className,
   138→  hint,
   139→}: FieldProps) {
   140→  return (
   141→    <div className={`space-y-1.5 ${className ?? ""}`}>
   142→      <Label className="text-xs font-medium text-muted-foreground">
   143→        {label}
   144→      </Label>
   145→      <Input
   146→        type={type}
   147→        value={value}
   148→        onChange={(e) => onChange(e.target.value)}
   149→        placeholder={placeholder}
   150→        dir={dir}
   151→        className="h-9 text-sm"
   152→      />
   153→      {hint && <p className="text-xs text-muted-foreground/80">{hint}</p>}
   154→    </div>
   155→  );
   156→}

   175→export function AdminSettings() {
   176→  const adminCode = useAppStore((s) => s.adminCode);
   177→  const adminHeaders: Record<string, string> = { "x-admin-code": adminCode };
   178→  const [settings, setSettings] = useState<AppSettings | null>(null);
   179→  const [original, setOriginal] = useState<AppSettings | null>(null);
   180→  const [loading, setLoading] = useState(true);
   181→  const [saving, setSaving] = useState(false);
   182→  const [resetting, setResetting] = useState(false);
   183→  const [activeTab, setActiveTab] = useState("services");
   184→
   185→  // تحميل الإعدادات عند التركيب
   186→  const loadSettings = useCallback(async () => {
   187→    setLoading(true);
   188→    try {
   189→      const res = await fetch("/api/settings");
   190→      if (!res.ok) throw new Error("فشل تحميل الإعدادات");
   191→      const data = (await res.json()) as AppSettings;
   192→      const safe: AppSettings = {
   193→        services: data.services ?? DEFAULT_SETTINGS.services,
   194→        deliveryOptions: data.deliveryOptions ?? DEFAULT_SETTINGS.deliveryOptions,
   195→        general: { ...DEFAULT_SETTINGS.general, ...(data.general ?? {}) },
   196→      };
   197→      setSettings(safe);
   198→      setOriginal(deepClone(safe));
   199→    } catch (e) {
   200→      toast.error("تعذّر تحميل الإعدادات", {
   201→        description: (e as Error).message,
   202→      });
   203→      // استخدام الافتراضي كنسخة احتياطية
   204→      setSettings(deepClone(DEFAULT_SETTINGS));
   205→      setOriginal(deepClone(DEFAULT_SETTINGS));
   206→    } finally {
   207→      setLoading(false);
   208→    }
   209→  }, []);
   210→
   211→  useEffect(() => {
   212→    void loadSettings();
   213→  }, [loadSettings]);
   214→
   215→  // مؤشر التغييرات غير المحفوظة
   216→  const hasChanges = useMemo(() => {
   217→    if (!settings || !original) return false;
   218→    return !deepEqual(settings, original);
   219→  }, [settings, original]);

   223→  /** تحديث حقل في الإعدادات العامة */
   224→  function updateGeneral<K extends keyof AppSettings["general"]>(
   225→    key: K,
   226→    value: AppSettings["general"][K],
   227→  ) {
   228→    setSettings((prev) =>
   229→      prev
   230→        ? { ...prev, general: { ...prev.general, [key]: value } }
   231→        : prev,
   232→    );
   233→  }
   234→
   235→  /** تحديث حقل في خدمة معيّنة */
   236→  function updateService(
   237→    serviceIdx: number,
   238→    patch: Partial<ServiceSpec>,
   239→  ) {
   240→    setSettings((prev) => {
   241→      if (!prev) return prev;
   242→      const services = [...prev.services];
   243→      services[serviceIdx] = { ...services[serviceIdx], ...patch };
   244→      return { ...prev, services };
   245→    });
   246→  }
   247→
   248→  /** تحديث حقل في قسم معيّن داخل خدمة */
   249→  function updateSection(
   250→    serviceIdx: number,
   251→    sectionIdx: number,
   252→    patch: Partial<SpecSection>,
   253→  ) {
   254→    setSettings((prev) => {
   255→      if (!prev) return prev;
   256→      const services = prev.services.map((s, i) => {
   257→        if (i !== serviceIdx) return s;
   258→        const sections = s.sections.map((sec, j) =>
   259→          j === sectionIdx ? { ...sec, ...patch } : sec,
   260→        );
   261→        return { ...s, sections };
   262→      });
   263→      return { ...prev, services };
   264→    });
   265→  }
   266→
   267→  /** تحديث حقل في خيار معيّن */
   268→  function updateOption(
   269→    serviceIdx: number,
   270→    sectionIdx: number,
   271→    optionIdx: number,
   272→    patch: Partial<SpecOption>,
   273→  ) {
   274→    setSettings((prev) => {
   275→      if (!prev) return prev;
   276→      const services = prev.services.map((s, i) => {
   277→        if (i !== serviceIdx) return s;
   278→        const sections = s.sections.map((sec, j) => {
   279→          if (j !== sectionIdx) return sec;
   280→          const options = sec.options.map((opt, k) =>
   281→            k === optionIdx ? { ...opt, ...patch } : opt,
   282→          );
   283→          return { ...sec, options };
   284→        });
   285→        return { ...s, sections };
   286→      });
   287→      return { ...prev, services };
   288→    });
   289→  }

   292→  function addOption(serviceIdx: number, sectionIdx: number) {
   293→    setSettings((prev) => {
   294→      if (!prev) return prev;
   295→      const services = prev.services.map((s, i) => {
   296→        if (i !== serviceIdx) return s;
   297→        const sections = s.sections.map((sec, j) => {
   298→          if (j !== sectionIdx) return sec;
   299→          const newOpt: SpecOption = {
   300→            id: uid("opt"),
   301→            label: "خيار جديد",
   302→            emoji: "✨",
   303→            description: "",
   304→          };
   305→          return { ...sec, options: [...sec.options, newOpt] };
   306→        });
   307→        return { ...s, sections };
   308→      });
   309→      return { ...prev, services };
   310→    });
   311→    toast.success("تمت إضافة خيار جديد");
   312→  }

   457→  // ======== الحفظ ========

   459→  async function handleSave() {
   460→    if (!settings) return;
   461→    setSaving(true);
   462→    try {
   463→      const res = await fetch("/api/settings", {
   464→        method: "PUT",
   465→        headers: { "Content-Type": "application/json", ...adminHeaders },
   466→        body: JSON.stringify(settings),
   467→      });
   468→      if (!res.ok) {
   469→        const err = await res.json().catch(() => ({}));
   470→        throw new Error(err.error || "فشل الحفظ");
   471→      }
   472→      setOriginal(deepClone(settings));
   473→      toast.success("تم حفظ الإعدادات بنجاح", {
   474→        description: "ستظهر التغييرات للعملاء فوراً",
   475→        icon: <CheckCircle2 className="h-4 w-4 text-emerald-500" />,
   476→      });
   477→    } catch (e) {
   478→      toast.error("فشل حفظ الإعدادات", {
   479→        description: (e as Error).message,
   480→      });
   481→    } finally {
   482→      setSaving(false);
   483→    }
   484→  }
   485→
   486→  async function handleReset() {
   487→    setResetting(true);
   488→    try {
   489→      const res = await fetch("/api/settings", { method: "DELETE", headers: adminHeaders });
   490→      if (!res.ok) {
   491→        const err = await res.json().catch(() => ({}));
   492→        throw new Error(err.error || "فشل إعادة التعيين");
   493→      }
   494→      const fresh = deepClone(DEFAULT_SETTINGS);
   495→      setSettings(fresh);
   496→      setOriginal(deepClone(fresh));
   497→      toast.success("تمت إعادة التعيين للإعدادات الافتراضية", {
   498→        description: "عادت كل القيم إلى الوضع الأصلي",
   499→        icon: <RotateCcw className="h-4 w-4 text-amber-500" />,
   500→      });
   501→      setActiveTab("services");
   502→    } catch (e) {
   503→      toast.error("فشلت إعادة التعيين", {
   504→        description: (e as Error).message,
   505→      });
   506→    } finally {
   507→      setResetting(false);
   508→    }
   509→  }

   608→          <TabsTrigger
