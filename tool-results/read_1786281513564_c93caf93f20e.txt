   162→// ===== Sound Notification Helper =====
   163→function playNotificationSound() {
   164→  if (typeof window === "undefined") return;
   165→  try {
   166→    const ctx = new (window.AudioContext || (window as any).webkitAudioContext)();
   167→    const oscillator = ctx.createOscillator();
   168→    const gain = ctx.createGain();
   169→    oscillator.connect(gain);
   170→    gain.connect(ctx.destination);
   171→    oscillator.type = "sine";
   172→    oscillator.frequency.setValueAtTime(587, ctx.currentTime);
   173→    oscillator.frequency.setValueAtTime(784, ctx.currentTime + 0.1);
   174→    oscillator.frequency.setValueAtTime(1047, ctx.currentTime + 0.2);
   175→    gain.gain.setValueAtTime(0.3, ctx.currentTime);
   176→    gain.gain.exponentialRampToValueAtTime(0.01, ctx.currentTime + 0.5);
   177→    oscillator.start(ctx.currentTime);
   178→    oscillator.stop(ctx.currentTime + 0.5);
   179→  } catch {}
   180→}

   216→function TodayAchievementGauge({ achievement }: { achievement: { total: number; completed: number; percentage: number } }) {

   218→  const pct = achievement.percentage;
   219→  const color = pct > 70 ? "emerald" : pct >= 40 ? "amber" : "rose";
   220→  const strokeColor =
   221→    color === "emerald"
   222→      ? "#10b981"
   223→      : color === "amber"
   224→        ? "#f59e0b"
   225→        : "#f43f5e";
   226→  const trackColor =
   227→    color === "emerald"
   228→      ? "rgba(16,185,129,0.15)"
   229→      : color === "amber"
   230→        ? "rgba(245,158,11,0.15)"
   231→        : "rgba(244,63,94,0.15)";

   233→    color === "emerald"
   234→      ? "text-emerald-600 dark:text-emerald-400"
   235→      : color === "amber"
   236→        ? "text-amber-600 dark:text-amber-400"
   237→        : "text-rose-600 dark:text-rose-400";

   239→    color === "emerald"
   240→      ? "from-emerald-50 to-emerald-100/50 dark:from-emerald-950/30 dark:to-emerald-900/10"

   246→    const timer = setTimeout(() => setAnimatedPct(pct), 100);

   250→  const radius = 54;

   263→      {/* دائرة التقدم */}

   267→          className="w-full h-full -rotate-90"

   270→          {/* المسار الخلفي */}

   272→            cx="64" cy="64" r={radius}
   273→            fill="none"
   274→            stroke={trackColor}
   275→            strokeWidth="10"
   276→            strokeLinecap="round"
   277→          />
   278→          {/* مسار التقدم المتحرك */}
   279→          <circle
   280→            cx="64" cy="64" r={radius}
   281→            fill="none"
   282→            stroke={strokeColor}

   321→function QuickCustomerSearch({ orders }: { orders: PrintOrderLite[] }) {
   322→  const [query, setQuery] = useState("");
   323→  const [results, setResults] = useState<Array<{ name: string; phone: string; orderCount: number }>>([]);
   324→  const [selected, setSelected] = useState<string | null>(null);
   325→  const debounceRef = useRef<ReturnType<typeof setTimeout>>(undefined);
   326→
   327→  useEffect(() => {
   328→    if (!query.trim()) {
   329→      setResults([]);
   330→      return;
   331→    }
   332→    if (debounceRef.current) clearTimeout(debounceRef.current);
   333→    debounceRef.current = setTimeout(() => {
   334→      const q = query.toLowerCase().trim();
   335→      const customerMap = new Map<string, { name: string; phone: string; orderCount: number }>();
   336→      orders.forEach((o) => {
   337→        const phone = o.customer?.phone || "";
   338→        const name = o.customer?.name || "";
   339→        if (phone.includes(q) || name.toLowerCase().includes(q)) {
   340→          const key = phone || name;
   341→          const existing = customerMap.get(key);
   342→          if (existing) {
   343→            existing.orderCount += 1;
   344→          } else {
   345→            customerMap.set(key, { name, phone, orderCount: 1 });
   346→          }
   347→        }
   348→      });
   349→      setResults(Array.from(customerMap.values()).slice(0, 8));
   350→    }, 300);
   351→    return () => { if (debounceRef.current) clearTimeout(debounceRef.current); };
   352→  }, [query, orders]);
   353→
   354→  return (
   355→    <div className="mb-5 relative">
   356→      <div className="relative">
   357→        <Search className="absolute right-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground/70" />
   358→        <Input
   359→          value={query}
   360→          onChange={(e) => { setQuery(e.target.value); setSelected(null); }}
   361→          placeholder="ابحث عن عميل بالاسم أو الهاتف..."
   362→          className="pr-10 text-sm h-10 rounded-lg focus:ring-ring focus:border-ring bg-background focus-outline input-glow"
   363→        />
   364→      </div>
   365→      {results.length > 0 && !selected && (
   366→        <div className="absolute top-full left-0 right-0 z-20 mt-1 bg-card border border-border rounded-xl shadow-lg overflow-hidden cluster">
   367→          {results.map((c, idx) => (
   368→            <button
   369→              key={`${c.phone}-${idx}`}
   370→              onClick={() => setSelected(c.phone || c.name)}
   371→              className="w-full flex items-center justify-between px-4 py-3 text-right hover:bg-accent/50 transition-colors"
   372→            >
   373→              <div className="min-w-0">
   374→                <div className="text-sm font-medium text-foreground truncate">{c.name || "—"}</div>
   375→                <div className="text-xs text-muted-foreground tabular-nums" dir="ltr">{c.phone || "—"}</div>
   376→              </div>
   377→              <span className="text-[10px] font-semibold px-2 py-0.5 rounded-full bg-primary/10 text-primary shrink-0 tabular-nums">
   378→                {c.orderCount} طلب
   379→              </span>
   380→            </button>
   381→          ))}
   382→        </div>
   383→      )}
   384→      {selected && (
   385→        <div className="mt-2 flex items-center gap-2 px-3 py-2 rounded-lg bg-primary/5 border border-primary/20 text-sm">
   386→          <span className="text-foreground">{results.find(c => (c.phone || c.name) === selected)?.name}</span>
   387→          <span className="text-muted-foreground tabular-nums" dir="ltr">{selected}</span>
   388→          <button onClick={() => { setSelected(null); setQuery(""); }} className="mr-auto text-muted-foreground hover:text-foreground">
   389→            <X className="h-3.5 w-3.5" />
   390→          </button>
   391→        </div>
   392→      )}
   393→    </div>
   394→  );
   395→}
