   289→  function fallbackAnswer(query: string) {
   290→    const q = query.toLowerCase().trim();
   291→    let bestMatch: FAQItem | null = null;
   292→    let bestScore = 0;
   293→    for (const faq of FAQS) {
   294→      let score = 0;
   295→      for (const kw of faq.keywords) {
   296→        if (q.includes(kw.toLowerCase())) score += kw.length > 3 ? 3 : 2;
   297→      }
   298→      const qWords = faq.question.toLowerCase().split(/\s+/);
   299→      for (const w of qWords) {
   300→        if (w.length > 2 && q.includes(w)) score += 1;
   301→      }
   302→      if (score > bestScore) { bestScore = score; bestMatch = faq; }
   303→    }
   304→    if (bestMatch && bestScore >= 2) {
   305→      const suggestions = FAQS.filter((f) => f.id !== bestMatch!.id && f.category === bestMatch!.category).slice(0, 3);
   306→      return { text: bestMatch.answer, suggestions };
   307→    }
   308→    return { text: "لم أجد إجابة دقيقة. جرّب إعادة صياغة السؤال أو تواصل معنا عبر واتساب 📱", suggestions: FAQS.slice(0, 4) };
   309→  }

   311→  async function askQuestion(q: string) {
   312→    if (!q.trim()) return;
   313→
   314→    // كشف أوامر خاصة: بحث أسعار
   315→    const searchMatch = q.match(/بحث\s+(أسعار|أسعار?|سعر)\s+(.+)/i);
   316→    if (searchMatch) {
   317→      const userMsg: Message = { role: "user", text: q };
   318→      setMessages((prev) => [...prev, userMsg]);
   319→      setInput("");
   320→      setTyping(true);
   321→      const query = `أسعار ${searchMatch[2].trim()} طباعة الجزائر 2025`;
   322→      await performWebSearch(query);
   323→      setTyping(false);
   324→      return;
   325→    }
   326→
   327→    // كشف أوامر خاصة: تصميم
   328→    const imgMatch = q.match(/(?:صمم|صميم|أنشئ|ارسم)\s+(?:لي\s+)?(.+)/i);
   329→    if (imgMatch) {
   330→      const userMsg: Message = { role: "user", text: q };
   331→      setMessages((prev) => [...prev, userMsg]);
   332→      setInput("");
   333→      setTyping(true);
   334→      const description = imgMatch[1].trim();
   335→      if (description.length >= 5) {
   336→        await generateImage(description);
   337→      } else {
   338→        setMessages((prev) => [
   339→          ...prev,
   340→          { role: "assistant", text: "❌ الوصف قصير جداً. اكتب تفاصيل أكثر عن التصميم المطلوب.\n\nمثال: *صمم لي بطاقة أعمال احترافية بلون ذهبي*", isAI: true },
   341→        ]);
   342→      }
   343→      setTyping(false);
   344→      return;
   345→    }
   346→
   347→    const userMsg: Message = { role: "user", text: q };
   348→    const newMessages = [...messages, userMsg];
   349→    setMessages(newMessages);
   350→    setInput("");
   351→    setTyping(true);
   352→
   353→    if (mode === "ai") {
   354→      const result = await callAIChat(q, newMessages);
   355→      setTyping(false);
   356→      if (result && result.success) {
   357→        setMessages((prev) => [...prev, { role: "assistant", text: result.response, isAI: true }]);
   358→      } else {
   359→        setMode("fallback");
   360→        const fallback = fallbackAnswer(q);
   361→        setMessages((prev) => [
   362→          ...prev,
   363→          { role: "assistant", text: fallback.text + "\n\n⚠️ تم التبديل للوضع الاحتياطي", suggestions: fallback.suggestions },
   364→        ]);
   365→      }
   366→    } else {
   367→      await new Promise((r) => setTimeout(r, 500));
   368→      setTyping(false);
   369→      const fallback = fallbackAnswer(q);
   370→      setMessages((prev) => [...prev, { role: "assistant", text: fallback.text, suggestions: fallback.suggestions }]);
   371→    }
   372→  }
