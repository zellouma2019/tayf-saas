   227→// ===== حساب حالة التجربة =====
   228→function getTrialInfo(shop: ShopStat): { label: string; color: string; daysLeft: number | null } {
   229→  if (!shop.trialDays || !shop.trialStartsAt) {
   230→    return { label: "بلا حدود", color: "bg-emerald-50 dark:bg-emerald-950/30 text-emerald-600", daysLeft: null };
   231→  }
   232→  const start = new Date(shop.trialStartsAt);
   233→  const end = new Date(start.getTime() + shop.trialDays * 86400000);
   234→  const now = new Date();
   235→  const daysLeft = Math.ceil((end.getTime() - now.getTime()) / 86400000);
   236→  if (daysLeft <= 0) {
   237→    return { label: "انتهت", color: "bg-rose-50 dark:bg-rose-950/30 text-rose-600", daysLeft: 0 };
   238→  }
   239→  if (daysLeft <= 3) {
   240→    return { label: `${daysLeft} يوم متبقي`, color: "bg-amber-50 dark:bg-amber-950/30 text-amber-600", daysLeft };
   241→  }
   242→  return { label: `${daysLeft} يوم متبقي`, color: "bg-primary/10 text-primary", daysLeft };
   243→}
